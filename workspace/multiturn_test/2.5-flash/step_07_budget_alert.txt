    // 获取所有支出类别选项 (用于填充类别预算的下拉菜单)
    function getAllExpenseCategories() {
        return Array.from(document.querySelectorAll('#category optgroup[label="支出"] option')).map(opt => opt.value);
    }

    // --- 数据操作与存储 ---

    // 从 localStorage 加载交易记录
    function loadTransactions() {
        const storedTransactions = localStorage.getItem('transactions');
        if (storedTransactions) {
            transactions = JSON.parse(storedTransactions);
        } else {
            transactions = []; // 如果没有数据，初始化为空数组
        }
    }

    // 保存交易记录到 localStorage
    function saveTransactions() {
        localStorage.setItem('transactions', JSON.stringify(transactions));
    }

    // 从 localStorage 加载类别预算
    function loadCategoryBudgets() {
        const storedBudgets = localStorage.getItem('categoryBudgets');
        if (storedBudgets) {
            // 合并默认预算和存储的预算，确保新增类别也有默认预算 (0)
            categoryBudgets = { ...categoryBudgets, ...JSON.parse(storedBudgets) };
        }
        // 确保所有HTML中定义的支出类别都在 categoryBudgets 中有记录
        // 如果没有，就给一个默认值 0
        const expenseCategories = getAllExpenseCategories();
        expenseCategories.forEach(cat => {
            if (!(cat in categoryBudgets)) {
                categoryBudgets[cat] = 0;
            }
        });
    }

    // 保存类别预算到 localStorage
    function saveCategoryBudgets() {
        localStorage.setItem('categoryBudgets', JSON.stringify(categoryBudgets));
    }

    // --- 核心功能函数 ---

    // 计算并显示总余额
    function updateBalance() {
        let totalBalance = 0;
        transactions.forEach(transaction => {
            if (transaction.type === 'income') {
                totalBalance += transaction.amount;
            } else {
                totalBalance -= transaction.amount;
            }
        });
        balanceDisplay.textContent = formatAmount(totalBalance);
    }

    // 渲染交易列表
    function renderTransactions() {
        transactionsBody.innerHTML = ''; // 清空现有列表
        transactions.forEach((transaction, index) => {
            const row = transactionsBody.insertRow();
            const typeCell = row.insertCell(0);
            const amountCell = row.insertCell(1);
            const categoryCell = row.insertCell(2);
            const dateCell = row.insertCell(3);
            const descriptionCell = row.insertCell(4);
            const actionCell = row.insertCell(5);

            typeCell.textContent = transaction.type === 'income' ? '收入' : '支出';
            typeCell.classList.add(transaction.type === 'income' ? 'income-text' : 'expense-text');
            
            amountCell.textContent = formatAmount(transaction.amount);
            amountCell.classList.add(transaction.type === 'income' ? 'income-text' : 'expense-text');

            categoryCell.textContent = transaction.category;
            dateCell.textContent = transaction.date;
            descriptionCell.textContent = transaction.description;

            const deleteButton = document.createElement('button');
            deleteButton.textContent = '删除';
            deleteButton.classList.add('delete-btn'); // 添加一个类，方便样式控制
            deleteButton.onclick = () => deleteTransaction(index);
            actionCell.appendChild(deleteButton);
        });
    }

    // 删除交易记录
    function deleteTransaction(index) {
        if (confirm('确定要删除这笔记录吗？')) {
            transactions.splice(index, 1); // 从数组中删除
            saveTransactions(); // 删除后保存
            initApp(); // 重新加载和更新所有数据
        }
    }

    // 计算并显示月度统计
    function updateMonthlySummary() {
        const currentMonth = getCurrentMonthString();
        let monthlyIncome = 0;
        let monthlyExpense = 0;

        transactions.forEach(transaction => {
            // 检查交易日期是否属于当前月份
            if (transaction.date.startsWith(currentMonth)) {
                if (transaction.type === 'income') {
                    monthlyIncome += transaction.amount;
                } else {
                    monthlyExpense += transaction.amount;
                }
            }
        });

        const monthlyBalance = monthlyIncome - monthlyExpense;
        const remainingBudget = MONTHLY_BUDGET - monthlyExpense;

        monthlyIncomeDisplay.textContent = formatAmount(monthlyIncome);
        monthlyExpenseDisplay.textContent = formatAmount(monthlyExpense);
        monthlyBalanceDisplay.textContent = formatAmount(monthlyBalance);

        // 预算提醒
        budgetAlert.classList.remove('over', 'within'); // 清除之前的样式
        if (remainingBudget < 0) {
            budgetAlert.textContent = `🚨 总预算超支 ${formatAmount(Math.abs(remainingBudget))} 元！请注意。`;
            budgetAlert.classList.add('over');
        } else {
            budgetAlert.textContent = `本月总预算剩余 ${formatAmount(remainingBudget)} 元。`;
            budgetAlert.classList.add('within');
        }
    }

    // 根据选择的类型更新类别下拉框的可用选项
    function updateCategoryOptions(type) {
        const categorySelect = document.getElementById('category');
        const currentSelectedValue = categorySelect.value;
        
        Array.from(categorySelect.children).forEach(optgroup => {
            if (optgroup.tagName === 'OPTGROUP') {
                if (optgroup.label === '收入' && type === 'expense') {
                    optgroup.style.display = 'none';
                } else if (optgroup.label === '支出' && type === 'income') {
                    optgroup.style.display = 'none';
                } else {
                    optgroup.style.display = ''; // 显示对应类别
                }
            }
        });

        let foundCurrent = false;
        for (let i = 0; i < categorySelect.options.length; i++) {
            const option = categorySelect.options[i];
            if (option.value === currentSelectedValue && option.parentNode.style.display !== 'none') {
                foundCurrent = true;
                break;
            }
        }
        
        if (!foundCurrent) {
            for (let i = 0; i < categorySelect.options.length; i++) {
                const option = categorySelect.options[i];
                if (option.parentNode.style.display !== 'none') {
                    categorySelect.selectedIndex = i;
                    break;
                }
            }
        }
    }

    // --- 图表相关函数 ---

    function renderExpenseChart() {
        const currentMonth = getCurrentMonthString();
        const expenseData = {}; // 用于存储 {类别: 总金额}

        transactions.forEach(transaction => {
            if (transaction.type === 'expense' && transaction.date.startsWith(currentMonth)) {
                const category = transaction.category;
                expenseData[category] = (expenseData[category] || 0) + transaction.amount;
            }
        });

        const labels = Object.keys(expenseData);
        const data = Object.values(expenseData);

        // 如果没有支出数据，隐藏图表并显示提示
        if (labels.length === 0) {
            expenseChartCanvas.style.display = 'none';
            noChartDataMessage.style.display = 'block';
            if (expenseChartInstance) {
                expenseChartInstance.destroy(); // 销毁旧图表实例
                expenseChartInstance = null;
            }
            return;
        } else {
            expenseChartCanvas.style.display = 'block';
            noChartDataMessage.style.display = 'none';
        }

        // 如果图表实例已存在，更新数据
        if (expenseChartInstance) {
            expenseChartInstance.data.labels = labels;
            expenseChartInstance.data.datasets[0].data = data;
            // 确保颜色也更新，即使类别数量没变，顺序可能变了
            expenseChartInstance.data.datasets[0].backgroundColor = labels.map((_, i) => chartColors[i % chartColors.length]);
            expenseChartInstance.update();
        } else {
            // 否则，创建一个新的图表实例
            const ctx = expenseChartCanvas.getContext('2d');
            expenseChartInstance = new Chart(ctx, {
                type: 'pie', // 饼图
                data: {
                    labels: labels,
                    datasets: [{
                        data: data,
                        backgroundColor: labels.map((_, i) => chartColors[i % chartColors.length]), // 循环使用颜色
                        hoverOffset: 4
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false, // 允许手动设置canvas尺寸，但在这里我们用CSS控制容器
                    plugins: {
                        legend: {
                            position: 'right', // 图例放在右边
                            labels: {
                                font: {
                                    size: 12,
                                    family: 'Noto Sans SC',
                                }
                            }
                        },
                        title: {
                            display: false, // 标题已经在H2中显示
                            text: '本月支出类别分析'
                        },
                        tooltip: {
                            callbacks: {
                                label: function(context) {
                                    let label = context.label || '';
                                    if (label) {
                                        label += ': ';
                                    }
                                    if (context.parsed !== null) {
                                        label += formatAmount(context.parsed) + ' 元';
                                    }
                                    return label;
                                }
                            }
                        }
                    }
                }
            });
        }
    }

    // --- 类别预算相关函数 ---

    // 填充类别预算设置的下拉菜单
    function populateBudgetCategoryDropdown() {
        const expenseCategories = getAllExpenseCategories();
        budgetCategorySelect.innerHTML = ''; // 清空现有选项
        if (expenseCategories.length === 0) {
            budgetCategorySelect.innerHTML = '<option value="">暂无支出类别</option>';
            budgetCategorySelect.disabled = true;
            budgetAmountInput.disabled = true;
            return;
        }
        budgetCategorySelect.disabled = false;
        budgetAmountInput.disabled = false;

        expenseCategories.forEach(category => {
            const option = document.createElement('option');
            option.value = category;
            option.textContent = category;
            budgetCategorySelect.appendChild(option);
        });
        // 默认选中第一个类别，并显示其当前预算
        if (expenseCategories.length > 0) {
            const initialCategory = budgetCategorySelect.value;
            budgetAmountInput.value = categoryBudgets[initialCategory] !== undefined ? categoryBudgets[initialCategory] : 0;
        }
    }

    // 渲染本月类别预算与支出概览
    function renderCategoryBudgetOverview() {
        const currentMonth = getCurrentMonthString();
        const monthlyCategorySpending = {}; // {类别: 总金额}

        transactions.forEach(transaction => {
            if (transaction.type === 'expense' && transaction.date.startsWith(currentMonth)) {
                monthlyCategorySpending[transaction.category] = (monthlyCategorySpending[transaction.category] || 0) + transaction.amount;
            }
        });

        categoryBudgetBody.innerHTML = ''; // 清空现有列表

        const allExpenseCategories = getAllExpenseCategories();

        if (allExpenseCategories.length === 0) {
            const row = categoryBudgetBody.insertRow();
            const cell = row.insertCell(0);
            cell.colSpan = 4;
            cell.textContent = '暂无支出类别可设置预算。';
            cell.classList.add('no-data-message');
            return;
        }
        
        let hasData = false;
        allExpenseCategories.forEach(category => {
            const budget = categoryBudgets[category] !== undefined ? categoryBudgets[category] : 0;
            const spent = monthlyCategorySpending[category] || 0;
            const remaining = budget - spent;

            // 只有当有预算设置或有实际支出时才显示该行
            if (budget > 0 || spent > 0) {
                hasData = true;
                const row = categoryBudgetBody.insertRow();
                row.insertCell(0).textContent = category;
                row.insertCell(1).textContent = formatAmount(budget);
                const spentCell = row.insertCell(2);
                spentCell.textContent = formatAmount(spent);
                const statusCell = row.insertCell(3);

                if (budget > 0 && spent > budget) {
                    statusCell.textContent = `超支 ${formatAmount(spent - budget)}`;
                    statusCell.classList.add('over-budget-category');
                } else if (budget > 0 && remaining >= 0) {
                    statusCell.textContent = `剩余 ${formatAmount(remaining)}`;
                    statusCell.classList.add('within-budget-category');
                } else if (budget === 0 && spent > 0) {
                    statusCell.textContent = `已支出 ${formatAmount(spent)} (无预算)`;
                    statusCell.classList.add('no-budget-set');
                } else {
                    statusCell.textContent = '无支出';
                    statusCell.classList.add('no-budget-set');
                }
            }
        });

        if (!hasData) {
            const row = categoryBudgetBody.insertRow();
            const cell = row.insertCell(0);
            cell.colSpan = 4;
            cell.textContent = '本月暂无类别预算或支出数据。';
            cell.classList.add('no-data-message');
        }
    }


    // --- 事件监听 ---

    // 表单提交处理
    transactionForm.addEventListener('submit', (event) => {
        event.preventDefault(); // 阻止表单默认提交行为（页面刷新）

        const type = document.querySelector('input[name="transaction-type"]:checked').value;
        const amount = parseFloat(document.getElementById('amount').value);
        const category = document.getElementById('category').value;
        const date = document.getElementById('date').value;
        const description = document.getElementById('description').value.trim();

        if (isNaN(amount) || amount <= 0) {
            alert('请输入有效的金额！');
            return;
        }

        const newTransaction = {
            type,
            amount,
            category,
            date,
            description,
            id: Date.now() // 为每笔交易生成一个唯一ID，方便将来操作
        };

        transactions.push(newTransaction);
        saveTransactions(); // 添加后保存
        transactionForm.reset(); // 清空表单
        // 重置类型选择和类别
        document.getElementById('type-income').checked = true;
        updateCategoryOptions('income');
        // 重新设置日期为今天
        const today = new Date();
        const year = today.getFullYear();
        const month = String(today.getMonth() + 1).padStart(2, '0');
        const day = String(today.getDate()).padStart(2, '0');
        dateInput.value = `${year}-${month}-${day}`;

        initApp(); // 重新加载和更新所有数据
    });

    // 监听类型（收入/支出）选择的变化
    document.querySelectorAll('input[name="transaction-type"]').forEach(radio => {
        radio.addEventListener('change', (event) => {
            updateCategoryOptions(event.target.value);
        });
    });

    // 类别预算表单提交处理
    budgetForm.addEventListener('submit', (event) => {
        event.preventDefault();
        const category = budgetCategorySelect.value;
        const amount = parseFloat(budgetAmountInput.value);

        if (!category) {
            alert('请选择一个类别！');
            return;
        }
        if (isNaN(amount) || amount < 0) {
            alert('请输入有效的预算金额（非负数）！');
            return;
        }

        categoryBudgets[category] = amount;
        saveCategoryBudgets();
        alert(`“${category}”的预算已更新为 ${formatAmount(amount)} 元。`);
        initApp(); // 重新加载和更新所有数据，包括预算概览
    });

    // 监听类别预算选择框的变化，更新预算金额输入框
    budgetCategorySelect.addEventListener('change', () => {
        const selectedCategory = budgetCategorySelect.value;
        if (selectedCategory) {
            budgetAmountInput.value = categoryBudgets[selectedCategory] !== undefined ? categoryBudgets[selectedCategory] : 0;
        }
    });

    // --- 初始化函数 ---

    // 应用初始化
    function initApp() {
        loadTransactions();          // 1. 加载所有记录
        loadCategoryBudgets();       // 2. 加载类别预算
        updateBalance();             // 3. 更新总余额
        renderTransactions();        // 4. 渲染交易列表
        updateMonthlySummary();      // 5. 更新月度统计和总预算提醒
        renderExpenseChart();        // 6. 渲染月度支出饼图
        populateBudgetCategoryDropdown(); // 7. 填充类别预算设置的下拉菜单
        renderCategoryBudgetOverview(); // 8. 渲染本月类别预算概览
    }

    // 页面加载完成后执行
    document.addEventListener('DOMContentLoaded', () => {
        // 设置日期输入框的默认值为今天
        const today = new Date();
        const year = today.getFullYear();
        const month = String(today.getMonth() + 1).padStart(2, '0');
        const day = String(today.getDate()).padStart(2, '0');
        dateInput.value = `${year}-${month}-${day}`;
        
        // 默认选中收入，并根据类型调整类别选择
        document.getElementById('type-income').checked = true;
        updateCategoryOptions('income');

        initApp(); // 启动应用
    });
</script>
