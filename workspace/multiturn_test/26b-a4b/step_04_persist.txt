在之前的代码里，你会看到这两行关键代码：
1. `let transactions = JSON.parse(localStorage.getItem('myFinanceData')) || [];` （**读取**：打开网页时，去浏览器的“小抽屉”里找之前的账单）
2. `localStorage.setItem('myFinanceData', JSON.stringify(transactions));` （**保存**：每次你添加或删除记录时，立刻把新账单塞进浏览器的“小抽屉”）

这个“小抽屉”在编程里叫 **`localStorage` (本地存储)**。它不需要服务器，数据就保存在你电脑浏览器的缓存文件里，关了浏览器、关了电脑，数据都会在。

### ⚠️ 为什么你可能会觉得数据“丢了”？

如果你发现数据没了，通常只有这三个原因：
1. **使用了“无痕模式/隐身模式” (Incognito/Private Window)**：无痕模式为了隐私，在你关掉窗口的一瞬间，会把所有的“小抽屉”全部清空。
2. **清理了浏览器缓存/历史记录**：如果你在浏览器设置里点了“清除浏览数据”并勾选了“Cookie 和其他网站数据”，那“小抽屉”也会被清空。
3. **换了浏览器**：你在 Chrome 里记的账，在 Edge 里是看不见的，因为每个浏览器的“抽屉”是独立的。

### 🚀 进阶方案：给你的数据买个“保险”

既然你不想用服务器，但又担心浏览器清理缓存导致数据丢失，我为你增加一个**“终极保险”功能**：**数据备份与恢复**。

我可以为你增加两个按钮：
1. **“备份数据”**：点击后，它会把你的所有账单生成一个 `.json` 文件下载到你的电脑里。这就是你的“账本副本”。
2. **“恢复数据”**：如果你换了电脑，或者不小心清理了浏览器，你只需要点击这个按钮，把刚才那个文件传上来，所有的账单就瞬间回来了！

<script>
    /* --- JavaScript 逻辑部分 --- */

    const list = document.getElementById('transaction-list');
    const balanceDisplay = document.getElementById('total-balance');
    const mIncomeDisplay = document.getElementById('month-income');
    const mExpenseDisplay = document.getElementById('month-expense');
    const mRemainingDisplay = document.getElementById('month-remaining');
    const budgetWarning = document.getElementById('budget-warning');
    const amountInput = document.getElementById('amount');
    const categoryInput = document.getElementById('category');
    const dateInput = document.getElementById('date');
    const noteInput = document.getElementById('note');

    const MONTHLY_BUDGET = 5000; 

    // 【核心】从 localStorage 读取
    let transactions = JSON.parse(localStorage.getItem('myFinanceData')) || [];

    window.onload = function() {
        const today = new Date().toISOString().split('T')[0];
        dateInput.value = today;
        init();
    };

    function init() {
        list.innerHTML = '';
        let totalAll = 0;
        let monthIncome = 0;
        let monthExpense = 0;
        
        const now = new Date();
        const currentMonthStr = now.getFullYear() + '-' + String(now.getMonth() + 1).padStart(2, '0');

        // 按日期排序
        transactions.sort((a, b) => new Date(b.date) - new Date(a.date));

        transactions.forEach((t, index) => {
            totalAll += t.amount;
            if (t.date.startsWith(currentMonthStr)) {
                if (t.amount > 0) {
                    monthIncome += t.amount;
                } else {
                    monthExpense += Math.abs(t.amount);
                }
            }

            const li = document.createElement('li');
            const amountClass = t.amount >= 0 ? 'amount-plus' : 'amount-minus';
            const amountSign = t.amount >= 0 ? '+' : '';
            li.innerHTML = `
                <div class="record-info">
                    <div class="record-main">
                        <span class="record-category">${t.category}</span>
                        <span class="record-date">${t.date}</span>
                    </div>
                    <div class="record-note">${t.note || ''}</div>
                </div>
                <div style="display: flex; align-items: center;">
                    <span class="amount-text ${amountClass}">${amountSign}${t.amount.toFixed(2)}</span>
                    <span class="delete-btn" onclick="deleteTransaction(${index})">&times;</span>
                </div>
            `;
            list.appendChild(li);
        });

        balanceDisplay.innerText = `¥ ${totalAll.toFixed(2)}`;
        mIncomeDisplay.innerText = `¥ ${monthIncome.toFixed(2)}`;
        mExpenseDisplay.innerText = `¥ ${monthExpense.toFixed(2)}`;
        
        const remaining = MONTHLY_BUDGET - monthExpense;
        mRemainingDisplay.innerText = `¥ ${remaining.toFixed(2)}`;

        if (monthExpense > MONTHLY_BUDGET) {
            budgetWarning.style.display = 'block';
            mRemainingDisplay.style.color = '#e74c3c';
        } else {
            budgetWarning.style.display = 'none';
            mRemainingDisplay.style.color = '#333';
        }

        // 【核心】每次改动，都把最新的数据存入 localStorage
        localStorage.setItem('myFinanceData', JSON.stringify(transactions));
    }

    function addTransaction() {
        const amount = parseFloat(amountInput.value);
        const category = categoryInput.value;
        const date = dateInput.value;
        const note = noteInput.value;

        if (isNaN(amount) || date === '') {
            alert('请填写正确的金额和日期');
            return;
        }

        transactions.push({ amount, category, date, note });
        amountInput.value = '';
        noteInput.value = '';
        init();
    }

    function deleteTransaction(index) {
        if(confirm('确定要删除这条记录吗？')) {
            transactions.splice(index, 1);
            init();
        }
    }

    /* --- 【新增功能】备份与恢复逻辑 --- */

    // 备份功能：将数据变成一个文件下载
    function exportData() {
        if (transactions.length === 0) {
            alert("目前没有数据可以备份！");
            return;
        }
        const dataStr = JSON.stringify(transactions, null, 2); // 格式化一下，让文件好看点
        const blob = new Blob([dataStr], { type: "application/json" });
        const url = URL.createObjectURL(blob);
        
        const link = document.createElement('a');
        link.href = url;
        link.download = `my_finance_backup_${new Date().toISOString().split('T')[0]}.json`;
        link.click();
        URL.revokeObjectURL(url);
    }

    // 恢复功能：读取用户上传的文件
    function importData(event) {
        const file = event.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = function(e) {
            try {
                const importedData = JSON.parse(e.target.result);
                if (Array.isArray(importedData)) {
                    if (confirm('这将覆盖你当前的所有记录，确定吗？')) {
                        transactions = importedData;
                        init();
                        alert('数据恢复成功！');
                    }
                } else {
                    alert('文件格式不正确！');
                }
            } catch (err) {
                alert('读取文件失败，请确保这是一个正确的备份文件。');
            }
        };
        reader.readAsText(file);
    }
