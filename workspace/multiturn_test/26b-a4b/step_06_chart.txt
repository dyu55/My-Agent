这个需求非常专业！有了图表，你的记账本就不再只是一个“记事本”，而是一个真正的“财务分析工具”了。

为了实现这个功能，我们不能自己从头写绘图算法（那太难了），我们需要引入一个世界上最流行的绘图库 —— **Chart.js**。它非常强大，能做出非常丝滑、漂亮的动画效果，而且对移动端非常友好。

我为你做了以下升级：
1.  **引入 Chart.js**：通过一行代码引入这个“绘图大师”。
2.  **新增“消费分布”卡片**：在统计数据下方增加了一个圆环图（Doughnut Chart）。圆环图比饼图看起来更现代、更高级。
3.  **自动分类计算逻辑**：我写了一段逻辑，它会自动扫描你本月的记录，把属于“餐饮”的钱加在一起，把属于“交通”的钱加在一起，然后交给绘图大师去画图。
4.  **色彩联动**：图表的颜色会根据你的类别自动分配，保持整体视觉的和谐。

<script>
    /* --- JavaScript 逻辑 --- */

    const list = document.getElementById('transaction-list');
    const balanceDisplay = document.getElementById('total-balance');
    const mIncomeDisplay = document.getElementById('month-income');
    const mExpenseDisplay = document.getElementById('month-expense');
    const mRemainingDisplay = document.getElementById('month-remaining');
    const budgetWarning = document.getElementById('budget-warning');
    
    const amountInput = document.getElementById('amount');
    const categoryInput = document.getElementById('category');
    const dateInput = document.getElementById('date');
    const noteInput = document.getElementById('note');

    const MONTHLY_BUDGET = 5000; 
    let transactions = JSON.parse(localStorage.getItem('myFinanceData')) || [];
    
    // 【新增】用于存储图表实例，防止重复创建导致内存泄漏或闪烁
    let myChart = null;

    window.onload = function() {
        const today = new Date().toISOString().split('T')[0];
        dateInput.value = today;
        init();
    };

    function init() {
        list.innerHTML = '';
        let totalAll = 0;
        let monthIncome = 0;
        let monthExpense = 0;
        
        // 用于统计分类支出的对象
        // 格式如: { "🍎 餐饮": 150, "🚗 交通": 50 }
        const categoryDataMap = {};

        const now = new Date();
        const currentMonthStr = now.getFullYear() + '-' + String(now.getMonth() + 1).padStart(2, '0');

        // 按日期排序
        transactions.sort((a, b) => new Date(b.date) - new Date(a.date));

        transactions.forEach((t, index) => {
            totalAll += t.amount;
            
            // 统计本月数据
            if (t.date.startsWith(currentMonthStr)) {
                if (t.amount > 0) {
                    monthIncome += t.amount;
                } else {
                    const absAmount = Math.abs(t.amount);
                    monthExpense += absAmount;
                    
                    // 【新增】累加该类别的支出金额
                    categoryDataMap[t.category] = (categoryDataMap[t.category] || 0) + absAmount;
                }
            }

            // 渲染列表
            const li = document.createElement('li');
            const isPlus = t.amount >= 0;
            li.innerHTML = `
                <div class="record-left">
                    <div class="record-cat">${t.category}</div>
                    <div class="record-date">${t.date}</div>
                    ${t.note ? `<div class="record-note">${t.note}</div>` : ''}
                </div>
                <div class="record-right">
                    <span class="amount-val ${isPlus ? 'amount-plus' : 'amount-minus'}">${isPlus ? '+' : ''}${t.amount.toFixed(2)}</span>
                    <span class="del-icon" onclick="deleteTransaction(${index})">&times;</span>
                </div>
            `;
            list.appendChild(li);
        });

        // 更新文字显示
        balanceDisplay.innerText = `¥ ${totalAll.toFixed(2)}`;
        mIncomeDisplay.innerText = `¥ ${monthIncome.toFixed(2)}`;
        mExpenseDisplay.innerText = `¥ ${monthExpense.toFixed(2)}`;
        
        const remaining = MONTHLY_BUDGET - monthExpense;
        mRemainingDisplay.innerText = `¥ ${remaining.toFixed(2)}`;
        mRemainingDisplay.style.color = remaining < 0 ? 'var(--danger)' : 'var(--text-main)';

        if (monthExpense > MONTHLY_BUDGET) {
            budgetWarning.style.display = 'block';
        } else {
            budgetWarning.style.display = 'none';
        }

        // 【核心】更新图表
        updateChart(categoryDataMap);

        localStorage.setItem('myFinanceData', JSON.stringify(transactions));
    }

    // 【新增】更新图表的函数
    function updateChart(categoryMap) {
        const ctx = document.getElementById('categoryChart').getContext('2d');
        
        // 准备数据
        const labels = Object.keys(categoryMap);
        const dataValues = Object.values(categoryMap);

        // 如果这个月没有任何支出，显示提示或清空图表
        if (labels.length === 0) {
            if (myChart) myChart.destroy();
            return;
        }

        // 如果之前已经有一个图表了，先把它销毁，否则新旧数据会重叠
        if (myChart) {
            myChart.destroy();
        }

        // 创建新图表
        myChart = new Chart(ctx, {
            type: 'doughnut', // 圆环图
            data: {
                labels: labels,
                datasets: [{
                    data: dataValues,
                    backgroundColor: [
                        '#7da684', '#e9967a', '#f4d03f', '#5dade2', '#af7ac5', '#ec7063', '#aab7b8'
                    ],
                    borderWidth: 0,
                    hoverOffset: 10
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false, // 让图表填充容器
                plugins: {
                    legend: {
                        position: 'bottom',
                        labels: {
                            usePointStyle: true,
                            padding: 20,
                            font: { size: 12 }
                        }
                    }
                },
                cutout: '60%' // 圆环中间的空心程度
            }
        });
    }

    function addTransaction() {
        const amount = parseFloat(amountInput.value);
        const category = categoryInput.value;
        const date = dateInput.value;
        const note = noteInput.value;

        if (isNaN(amount) || date === '') {
            alert('请填写金额和日期哦~');
            return;
        }

        transactions.push({ amount, category, date, note });
        amountInput.value = '';
        noteInput.value = '';
        init();
    }

    function deleteTransaction(index) {
        if(confirm('确定要删除这条记录吗？')) {
            transactions.splice(index, 1);
            init();
        }
    }

    function exportData() {
        if (transactions.length === 0) return alert("没有数据可以备份");
        const dataStr = JSON.stringify(transactions, null, 2);
        const blob = new Blob([dataStr], { type: "application/json" });
        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = `家账本备份_${new Date().toLocaleDateString()}.json`;
        link.click();
    }

    function importData(event) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function(e) {
            try {
                const data = JSON.parse(e.target.result);
                if (confirm('导入将覆盖当前数据，确定吗？')) {
                    transactions = data;
                    init();
                }
            } catch (err) { alert('文件格式错误'); }
        };
        reader.readAsText(file);
    }
</script>

### 📈 现在的变化：

1.  **直观的视觉分析**：
    *   我在统计卡片下方新增了一个“**消费占比分析**”圆环图。
    *   当你输入一笔“🍎 餐饮”支出后，图表会自动增加一个绿色的切片。
    *   当你输入一笔“🛍️ 购物”支出后，图表会自动增加一个橘色的切片。
    *   通过切片的大小，你瞬间就能明白：**“哎呀，这个月我的钱都花在吃上面了！”**

2.  **智能数据处理**：
    *   图表是**动态**的。它只统计**本月**的**支出**（负数）。
    *   如果你输入了一笔工资（收入），图表是不会变化的，这保证了分析的准确性——我们分析的是“钱花哪了”，而不是“钱从哪来”。

3.  **技术细节（给你的小知识）**：
    *   **`myChart.destroy()`**：这是非常关键的一步。在编程中，如果你在一个位置反复画新图而不把旧图擦掉，旧图的颜色和数据会堆积在一起，导致显示错乱。这行代码保证了每次你添加新账单时，旧图会被彻底清理，只画一张全新的。
    *   **`responsive: true`**：这确保了无论你在 iPhone 还是安卓手机上打开，图表都会自动调整大小，不会超出屏幕。
