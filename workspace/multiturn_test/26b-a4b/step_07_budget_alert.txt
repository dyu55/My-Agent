这已经完全进入了“专业级财务应用”的逻辑了！你现在要求的不是简单的“总额预警”，而是**“精细化预算管理”**。

为了实现这个功能，我们需要在逻辑上做一个巨大的转变：
1.  **引入“预算配置”数据结构**：我们要新增一个专门存储“每个类别预算是多少”的对象，并把它也存进 `localStorage`。
2.  **新增“预算设置”界面**：用户需要一个地方来输入和修改这些数字。
3.  **多维度预警逻辑**：
    *   **总预算预警**（你之前已经有的）。
    *   **分类预警**：程序要对比“本月该类别的实际支出” vs “该类别的设定预算”。
4.  **动态警告列表**：如果某个类别超支了，我们就在页面上弹出一个醒目的“警告列表”。

由于功能变复杂了，我把代码进行了模块化处理，让它看起来整洁且易于维护。

    /* --- JavaScript 逻辑 --- */

    // 预设的类别和默认预算
    const DEFAULT_CATEGORIES = ["🍎 餐饮", "🚗 交通", "🛍️ 购物", "💵 工资", "🎬 娱乐", "🏠 居家", "✨ 其他"];

    const list = document.getElementById('transaction-list');
    const balanceDisplay = document.getElementById('total-balance');

    const mExpenseDisplay = document.getElementById('month-expense');
    const mRemainingDisplay = document.getElementById('month-remaining');

    const budgetSettingsList = document.getElementById('budget-settings-list');
    
    const amountInput = document.getElementById('amount');
    const categoryInput = document.getElementById('category');
    const dateInput = document.getElementById('date');
    const noteInput = document.getElementById('note');

    const TOTAL_MONTHLY_BUDGET = 5000; // 总月预算

    // 数据加载

    let budgets = JSON.parse(localStorage.getItem('myCategoryBudgets')) || { ...DEFAULT_BUDGETS };
    
    let myChart = null;

    window.onload = function() {
        const today = new Date().toISOString().split('T')[0];
        dateInput.value = today;
        renderBudgetSettings(); // 初始化预算设置界面
        init();
    };

    // 渲染预算设置界面
    function renderBudgetSettings() {
        budgetSettingsList.innerHTML = '';
        DEFAULT_CATEGORIES.forEach(cat => {
            const div = document.createElement('div');
            div.className = 'budget-setting-item';
            div.innerHTML = `
                <label>${cat}</label>
                <input type="number" class="budget-input" data-category="${cat}" value="${budgets[cat] || 0}">
            `;
            budgetSettingsList.appendChild(div);
        });
    }

    // 保存预算
    function saveBudgets() {
        const inputs = document.querySelectorAll('.budget-input');
        inputs.forEach(input => {
            const cat = input.getAttribute('data-category');
            budgets[cat] = parseFloat(input.value) || 0;
        });
        localStorage.setItem('myCategoryBudgets', JSON.stringify(budgets));
        init(); // 重新计算并刷新界面
        alert('预算设置已保存！');
    }

    function init() {
        list.innerHTML = '';
        warningArea.innerHTML = ''; // 清空警告
        
        let totalAll = 0;
        let monthIncome = 0;
        let monthExpense = 0;
        
        // 用于统计分类支出的对象
        const categoryExpenses = {};
        // 用于存放超支的警告消息
        const warnings = [];

        const now = new Date();
        const currentMonthStr = now.getFullYear() + '-' + String(now.getMonth() + 1).padStart(2, '0');

        transactions.sort((a, b) => new Date(b.date) - new Date(a.date));

        transactions.forEach((t, index) => {
            totalAll += t.amount;
            
            if (t.date.startsWith(currentMonthStr)) {
                if (t.amount > 0) {
                    monthIncome += t.amount;
                } else {
                    const absAmount = Math.abs(t.amount);
                    monthExpense += absAmount;
                    categoryExpenses[t.category] = (categoryExpenses[t.category] || 0) + absAmount;
                }
            }

            // 渲染列表项
            const li = document.createElement('li');
            const isPlus = t.amount >= 0;
            li.innerHTML = `
                <div class="record-left">
                    <div class="record-cat">${t.category}</div>
                    <div class="record-date">${t.date}</div>
                    ${t.note ? `<div class="record-note">${t.note}</div>` : ''}
                </div>
                <div class="record-right">
                    <span class="amount-val ${isPlus ? 'amount-plus' : 'amount-minus'}">${isPlus ? '+' : ''}${t.amount.toFixed(2)}</span>
                    <span class="del-icon" onclick="deleteTransaction(${index})">&times;</span>
                </div>
            `;
            list.appendChild(li);
        });

        // --- 核心逻辑：检查分类预算是否超支 ---
        for (const cat in budgets) {
            const spent = categoryExpenses[cat] || 0;
            const limit = budgets[cat];
            // 只有当预算大于0且实际支出大于预算时，才触发警告
            if (limit > 0 && spent > limit) {
                warnings.push(`⚠️ ${cat} 已超支! (已花 ${spent.toFixed(0)} / 预算 ${limit})`);
            }
        }

        // 如果总预算也超了
        if (monthExpense > TOTAL_MONTHLY_BUDGET) {
            warnings.unshift(`🚨 总预算已超支! (已花 ${monthExpense.toFixed(0)})`);
        }

        // 显示警告
        warnings.forEach(msg => {
            const div = document.createElement('div');
            div.className = 'warning-item';
            div.innerText = msg;
            warningArea.appendChild(div);
        });

        // 更新文字
        balanceDisplay.innerText = `¥ ${totalAll.toFixed(2)}`;
        mIncomeDisplay.innerText = `¥ ${monthIncome.toFixed(2)}`;
        mExpenseDisplay.innerText = `¥ ${monthExpense.toFixed(2)}`;
        
        const remaining = TOTAL_MONTHLY_BUDGET - monthExpense;
        mRemainingDisplay.innerText = `¥ ${remaining.toFixed(2)}`;
        mRemainingDisplay.style.color = remaining < 0 ? 'var(--danger)' : 'var(--text-main)';

        // 更新图表
        updateChart(categoryExpenses);

        localStorage.setItem('myFinanceData', JSON.stringify(transactions));
    }

    function updateChart(categoryMap) {
        const ctx = document.getElementById('categoryChart').getContext('2d');
        const labels = Object.keys(categoryMap);
        const dataValues = Object.values(categoryMap);

        if (myChart) myChart.destroy();
        if (labels.length === 0) return;

        myChart = new Chart(ctx, {
            type: 'doughnut',
            data: {
                labels: labels,
                datasets: [{
                    data: dataValues,
                    backgroundColor: ['#7da684', '#e9967a', '#f4d03f', '#5dade2', '#af7ac5', '#ec7063', '#aab7b8'],
                    borderWidth: 0
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: { legend: { position: 'bottom', labels: { usePointStyle: true, padding: 15, font: { size: 11 } } } },
                cutout: '65%'
            }
        });
    }

    function addTransaction() {
        const amount = parseFloat(amountInput.value);
        const category = categoryInput.value;
        const date = dateInput.value;
        const note = noteInput.value;
        if (isNaN(amount) || date === '') { alert('请填写金额和日期哦~'); return; }
        transactions.push({ amount, category, date, note });
        amountInput.value = ''; noteInput.value = '';
        init();
    }

    function deleteTransaction(index) {
        if(confirm('确定要删除吗？')) { transactions.splice(index, 1); init(); }
    }

    function exportData() {
        const dataStr = JSON.stringify({ transactions, budgets }, null, 2);
        const blob = new Blob([dataStr], { type: "application/json" });
        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = `家账本备份_${new Date().toLocaleDateString()}.json`;
        link.click();
    }

    function importData(event) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function(e) {
            try {
                const imported = JSON.parse(e.target.result);
                if (confirm('导入将覆盖当前所有数据，确定吗？')) {
                    if (imported.transactions) transactions = imported.transactions;
                    if (imported.budgets) budgets = imported.budgets;
                    init();
                    renderBudgetSettings();
                    alert('恢复成功！');
                }
            } catch (err) { alert('文件错误'); }
        };
        reader.readAsText(file);
    }
</script>

### 🌟 本次升级的核心变化：

#### 1. 建立“预算管理系统”
*   **新增卡片**：我在页面中间增加了一个“⚙️ 分类预算设置”卡片。
*   **可编辑性**：你可以直接在每个分类后面输入数字（比如餐饮输入 1500）。
*   **持久化存储**：点击“保存预算设置”后，这些数字会存入 `localStorage`。即使你关掉浏览器，下次打开时，你的预算设置依然还在。

#### 2. 强大的“多级预警”逻辑
现在的程序在每次记账时，会进行两轮“安检”：
*   **第一轮（分类安检）**：它会对比“🍎 餐饮”实际花了多少钱 vs 你设定的“🍎 餐饮”预算。如果超了，它会生成一个像下面这样的警告：
    > `⚠️ 🍎 餐饮 已超支! (已花 1600 / 预算 1500)`
*   **第二轮（总额安检）**：它会看你这个月的总支出是否超过了总预算（5000）。
*   **展示方式**：所有的警告都会放在“统计卡片”下方，并带有一个**红色闪烁效果**，确保你一眼就能看到哪里出问题了。

#### 3. 数据备份的升级
*   因为现在你的数据不仅包含“账单”，还包含了“预算设置”，所以我升级了备份逻辑。当你点击“备份数据”时，它会把**账单+预算**一起打包成一个文件。
*   当你“恢复数据”时，所有的设置也会一起恢复。
