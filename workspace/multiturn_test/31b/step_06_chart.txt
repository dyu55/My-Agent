<script>
    const BUDGET = 5000; 
    const CATEGORIES = {
        "支出": ["🍱 餐饮", "🚌 交通", "🛍️ 购物", "🎬 娱乐", "🏠 房租水电", "💊 医疗", "🌸 其他"],
        "收入": ["💰 工资", "🧧 红包", "📈 投资", "🎁 奖金", "✨ 其他"]
    };

    let myChart = null; // 用于存储图表实例

    document.getElementById('date').valueAsDate = new Date();
    updateCategories();

    let records = JSON.parse(localStorage.getItem('myFamilyFinance')) || [];

    function updateCategories() {
        const type = document.getElementById('type').value;
        const catSelect = document.getElementById('category');
        catSelect.innerHTML = '';
        CATEGORIES[type].forEach(cat => {
            const opt = document.createElement('option');
            opt.value = cat;
            opt.innerHTML = cat;
            catSelect.appendChild(opt);
        });
    }

    function addRecord() {
        const date = document.getElementById('date').value;
        const type = document.getElementById('type').value;
        const category = document.getElementById('category').value;
        const amount = parseFloat(document.getElementById('amount').value);
        const note = document.getElementById('note').value;

        if (!date || isNaN(amount) || amount <= 0) {
            alert("请填好日期和正确的金额哦 🌸");
            return;
        }

        const record = { date, type, category, amount: type === "收入" ? amount : -amount, note };
        records.push(record);
        localStorage.setItem('myFamilyFinance', JSON.stringify(records));
        
        document.getElementById('amount').value = '';
        document.getElementById('note').value = '';
        render();
    }

    function clearAll() {
        if (confirm("确定要清空所有家庭账单吗？")) {
            records = [];
            localStorage.removeItem('myFamilyFinance');
            render();
        }
    }

    function render() {
        const tbody = document.getElementById('table-body');
        tbody.innerHTML = '';
        let totalBalance = 0;
        let mIncome = 0;
        let mExpense = 0;

        const now = new Date();
        const currentMonth = now.getMonth();
        const currentYear = now.getFullYear();

        // 用于统计各类别支出的对象
        const expenseMap = {};

        const sortedRecords = [...records].sort((a, b) => new Date(b.date) - new Date(a.date));

        sortedRecords.forEach(r => {
            totalBalance += r.amount;
            const rDate = new Date(r.date);
            if (rDate.getMonth() === currentMonth && rDate.getFullYear() === currentYear) {
                if (r.amount > 0) {
                    mIncome += r.amount;
                } else {
                    const absAmt = Math.abs(r.amount);
                    mExpense += absAmt;
                    // 累加到对应类别
                    expenseMap[r.category] = (expenseMap[r.category] || 0) + absAmt;
                }
            }

            tbody.innerHTML += `<tr>
                <td>${r.date}</td>
                <td>${r.category}</td>
                <td>${r.note}</td>
                <td class="${r.amount > 0 ? 'amt-positive' : 'amt-negative'}">
                    ${r.amount > 0 ? '+' : ''}${r.amount.toFixed(2)}
                </td>
            </tr>`;
        });

        document.getElementById('total-balance').innerText = `￥${totalBalance.toFixed(2)}`;
        document.getElementById('month-income').innerText = `￥${mIncome.toFixed(2)}`;
        document.getElementById('month-expense').innerText = `￥${mExpense.toFixed(2)}`;
        document.getElementById('month-net').innerText = `￥${(mIncome - mExpense).toFixed(2)}`;

        const warningDiv = document.getElementById('budget-warning');
        if (mExpense > BUDGET) {
            warningDiv.style.display = 'block';
            document.getElementById('warning-text').innerText = `本月已支出 ￥${mExpense.toFixed(2)}，超出预算 ￥${(mExpense - BUDGET).toFixed(2)}。记得省着花哦 🌿`;
        } else {
            warningDiv.style.display = 'none';
        }

        updateChart(expenseMap);
    }

    function updateChart(expenseMap) {
        const ctx = document.getElementById('expenseChart').getContext('2d');
        
        const labels = Object.keys(expenseMap);
        const data = Object.values(expenseMap);

        // 如果没有支出数据，显示一个简单的提示
        if (labels.length === 0) {
            if(myChart) myChart.destroy();
            return;
        }

        // 销毁旧图表，创建新图表
        if (myChart) {
            myChart.destroy();
        }

        myChart = new Chart(ctx, {
            type: 'pie', // 饼图
            data: {
                labels: labels,
                datasets: [{
                    data: data,
                    backgroundColor: [
                        '#a8bca1', '#dfcfbe', '#e9967a', '#8eb69b', 
                        '#d68a8a', '#b8c5d6', '#f2d5a6', '#c9ada7'
                    ],
                    borderWidth: 2,
                    borderColor: '#ffffff'
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        position: 'bottom',
                        labels: {
                            color: '#5d576b',
                            font: { size: 12 }
                        }
                    }
                }
            }
        });
    }

    render();
</script>

### 📈 本次更新详解：

**1. 引入了 Chart.js**
我在代码顶部添加了 `<script src="https://cdn.jsdelivr.net/npm/chart.js"></script>`。这就像是在你的网页里安装了一个“图表插件”，让网页拥有了绘制复杂图形的能力。

**2. 数据处理逻辑 (Data Mapping)**
为了画图，程序现在在执行 `render()`（渲染）时，会额外做一件事：
*   它会创建一个 `expenseMap`（支出地图）。
*   比如，当你录入一笔“🍱 餐饮 20元”和另一笔“🍱 餐饮 30元”时，它会在地图里记录：`{"🍱 餐饮": 50}`。
*   最后把这个地图的“类别”作为饼图的标签，“金额”作为饼图的大小。

**3. 视觉设计**
*   **饼图 (Pie Chart)**：我选择了饼图，因为它能最直观地显示**比例**。
*   **色系同步**：图表的颜色我特意设置成了之前的莫兰迪暖色系（鼠尾草绿、浅杏色、珊瑚橙等），确保图表和整体温馨的风格统一，不会显得突兀。
*   **自动刷新**：每当你点击“保存记录”或“清空数据”时，图表会自动销毁并重新绘制，确保你看到的是最新的数据。

### 💡 使用小建议：
*   **数据越多，图表越准**：当你录入不同类别的支出后，你会发现饼图被分成了很多块，颜色各异，一眼就能看出哪个“大块头”在吞噬你的预算。
*   **手机端查看**：在手机上，图表会由于响应式设计自动缩放在中间，非常方便查看。
