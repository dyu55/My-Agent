<script>
    // --- 配置 ---
    const BUDGET = 5000; 
    const CATEGORIES = {
        "支出": ["🍱 餐饮", "🚌 交通", "🛍️ 购物", "🎬 娱乐", "🏠 房租水电", "💊 医疗", "🌸 其他"],
        "收入": ["💰 工资", "🧧 红包", "📈 投资", "🎁 奖金", "✨ 其他"]
    };

    document.getElementById('date').valueAsDate = new Date();
    updateCategories();

    let records = JSON.parse(localStorage.getItem('myFamilyFinance')) || [];

    function updateCategories() {
        const type = document.getElementById('type').value;
        const catSelect = document.getElementById('category');
        catSelect.innerHTML = '';
        CATEGORIES[type].forEach(cat => {
            const opt = document.createElement('option');
            opt.value = cat;
            opt.innerHTML = cat;
            catSelect.appendChild(opt);
        });
    }

    function addRecord() {
        const date = document.getElementById('date').value;
        const type = document.getElementById('type').value;
        const category = document.getElementById('category').value;
        const amount = parseFloat(document.getElementById('amount').value);
        const note = document.getElementById('note').value;

        if (!date || isNaN(amount) || amount <= 0) {
            alert("请填好日期和正确的金额哦 🌸");
            return;
        }

        const record = { date, type, category, amount: type === "收入" ? amount : -amount, note };
        records.push(record);
        localStorage.setItem('myFamilyFinance', JSON.stringify(records));
        
        document.getElementById('amount').value = '';
        document.getElementById('note').value = '';
        render();
    }

    function clearAll() {
        if (confirm("确定要清空所有家庭账单吗？")) {
            records = [];
            localStorage.removeItem('myFamilyFinance');
            render();
        }
    }

    function render() {
        const tbody = document.getElementById('table-body');
        tbody.innerHTML = '';
        let totalBalance = 0;
        let mIncome = 0;
        let mExpense = 0;

        const now = new Date();
        const currentMonth = now.getMonth();
        const currentYear = now.getFullYear();

        const sortedRecords = [...records].sort((a, b) => new Date(b.date) - new Date(a.date));

        sortedRecords.forEach(r => {
            totalBalance += r.amount;
            const rDate = new Date(r.date);
            if (rDate.getMonth() === currentMonth && rDate.getFullYear() === currentYear) {
                if (r.amount > 0) mIncome += r.amount;
                else mExpense += Math.abs(r.amount);
            }

            tbody.innerHTML += `<tr>
                <td>${r.date}</td>
                <td>${r.category}</td>
                <td>${r.note}</td>
                <td class="${r.amount > 0 ? 'amt-positive' : 'amt-negative'}">
                    ${r.amount > 0 ? '+' : ''}${r.amount.toFixed(2)}
                </td>
            </tr>`;
        });

        document.getElementById('total-balance').innerText = `￥${totalBalance.toFixed(2)}`;
        document.getElementById('month-income').innerText = `￥${mIncome.toFixed(2)}`;
        document.getElementById('month-expense').innerText = `￥${mExpense.toFixed(2)}`;
        document.getElementById('month-net').innerText = `￥${(mIncome - mExpense).toFixed(2)}`;

        const warningDiv = document.getElementById('budget-warning');
        if (mExpense > BUDGET) {
            warningDiv.style.display = 'block';
            document.getElementById('warning-text').innerText = `本月已支出 ￥${mExpense.toFixed(2)}，超出预算 ￥${(mExpense - BUDGET).toFixed(2)}。记得省着花哦 🌿`;
        } else {
            warningDiv.style.display = 'none';
        }
    }

    render();
</script>
</body>

### 🎨 视觉升级亮点：

1.  **温馨配色 (Warm Palette)**：
    *   使用了**奶油米色 (`#fdfbf7`)** 作为背景，避免了纯白色的刺眼。
    *   主色调改为**鼠尾草绿 (`#a8bca1`)**，这是一种给人感觉平静、自然的颜色，非常适合家庭记账。
    *   收入和支出的颜色不再是深红深绿，而是采用了**柔和的莫兰迪色调**。
2.  **圆润设计 (Soft Edges)**：
    *   所有的卡片、按钮、输入框都使用了**大圆角 (20px - 24px)**，视觉上更亲切，没有棱角，像现代的手机 App。
3.  **响应式适配 (Mobile Friendly)**：
    *   **电脑端**：依然保持左边录入，右边查看，效率最高。
    *   **手机端**：当屏幕宽度较小时，界面会自动调整。统计面板会排在最上面，录入表单会自动移到下方，确保在手机上不需要左右滑动就能完整操作。
4.  **细节增强 (Micro-interactions)**：
    *   **按钮动效**：按钮在鼠标悬停时会微微上浮，点击时会下沉，有真实的按压感。
    *   **类别图标**：在类别中加入了 Emoji（如 🍱 餐饮, 🚌 交通），让账单看起来不再枯燥，更有生活气息。
    *   **温和警告**：超支警告变成了带有虚线边框的柔和红色卡片，并配上了温情的提醒文字。

### 📱 怎么在手机上使用？
因为这是一个本地文件，最简单的方法是：
1.  将这个 `.html` 文件通过微信、QQ 或邮件发到你的手机上。
2.  在手机上用**浏览器**（如 Safari 或 Chrome）打开它。
3.  **小技巧**：在手机浏览器中选择 **“添加到主屏幕”**，它就会像一个真正的 App 一样出现在你的手机桌面，以后点击就能直接记账！
