### 💻 第二步：把代码粘贴进去

现在你会看到一个编辑界面，中间有一个 `main.py` 文件。请把里面的所有内容删除，然后**完整地复制并粘贴**下面的这段代码：

```python
import streamlit as st
import pandas as pd
from datetime import date
import os

# --- 页面设置 ---
st.set_page_config(page_title="我的个人财务管家", layout="wide")
st.title("💰 我的个人财务管家")

# --- 数据存储设置 ---
# 我们使用一个简单的 CSV 文件来保存你的数据，这样即使网页刷新，数据也不会丢失
DATA_FILE = "finance_data.csv"

def load_data():
    if os.path.exists(DATA_FILE):
        return pd.read_csv(DATA_FILE)
    else:
        # 如果文件不存在，创建一个空的表格
        return pd.DataFrame(columns=["日期", "类型", "类别", "金额", "备注"])

def save_data(df):
    df.to_csv(DATA_FILE, index=False)

# 加载现有数据
df = load_data()

# --- 侧边栏：输入表单 ---
st.sidebar.header("📝 添加新记录")

with st.sidebar.form("finance_form", clear_on_submit=True):
    # 1. 日期选择
    record_date = st.date_input("日期", date.today())
    
    # 2. 类型选择 (收入/支出)
    record_type = st.selectbox("类型", ["支出", "收入"])
    
    # 3. 类别选择 (根据类型动态变化)
    categories = {
        "支出": ["餐饮", "交通", "购物", "娱乐", "房租/水电", "医疗", "其他支出"],
        "收入": ["工资", "奖金", "投资", "红包", "其他收入"]
    }
    record_category = st.selectbox("类别", categories[record_type])
    
    # 4. 金额输入
    record_amount = st.number_input("金额", min_value=0.0, format="%.2f")
    
    # 5. 备注输入
    record_note = st.text_input("备注 (可选)")
    
    # 提交按钮
    submit_button = st.form_submit_button("提交记录")

# 处理提交逻辑
if submit_button:
    if record_amount <= 0:
        st.sidebar.error("请输入大于 0 的金额！")
    else:
        # 将新记录添加到 DataFrame
        new_data = pd.DataFrame({
            "日期": [record_date],
            "类型": [record_type],
            "类别": [record_category],
            "金额": [record_amount if record_type == "收入" else -record_amount], # 支出记为负数
            "备注": [record_note]
        })
        df = pd.concat([df, new_data], ignore_index=True)
        save_data(df)
        st.sidebar.success("✅ 记录已保存！")

# --- 主界面：数据显示与统计 ---
col1, col2 = st.columns(2)

# 计算总额
total_balance = df["金额"].sum()
total_income = df[df["金额"] > 0]["金额"].sum()
total_expense = abs(df[df["金额"] < 0]["金额"].sum())

with col1:
    st.metric("当前余额", f"￥{total_balance:.2f}")
with col2:
    st.metric("总收入", f"￥{total_income:.2f}", delta_color="normal")
    st.metric("总支出", f"￥{total_expense:.2f}", delta_color="inverse")

st.divider()

# 显示历史记录表格
st.subheader("📅 历史账单明细")
if not df.empty:
    # 按照日期倒序排列，最新的在最上面
    df['日期'] = pd.to_datetime(df['日期']).dt.date
    df_display = df.sort_values(by="日期", ascending=False)
    st.dataframe(df_display, use_container_width=True)
    
    # 添加一个删除所有数据的按钮（慎用）
    if st.button("🗑️ 清空所有数据"):
        if os.path.exists(DATA_FILE):
            os.remove(DATA_FILE)
            st.rerun()
else:
    st.info("目前还没有记录，快去左侧表单添加一笔吧！")
```

### 💡 这个工具是怎么工作的？（给不懂编程的你）

为了让你以后能跟我沟通修改需求，我简单解释一下这个工具的原理：

*   **左侧表单 (The Form)**：我使用了 `st.sidebar.form`。它像一个收集箱，当你填完所有信息点击“提交”时，它才会一次性把数据发送给程序。
*   **数据存储 (The Storage)**：因为你没有数据库，所以我让程序在你的 Replit 文件夹里创建了一个名为 `finance_data.csv` 的文件。它就像一个简单的 Excel 表格，每次你提交，程序就把新的一行写在文件末尾。
*   **计算逻辑 (The Logic)**：
    *   如果你选的是“支出”，程序会自动在金额前加一个**负号**（例如：-50元）。
    *   这样在计算总余额时，程序只需要把所有金额**相加**，就能得出你的剩余钱数。
*   **界面显示 (The UI)**：`st.metric` 是用来显示那种大数字看板的，`st.dataframe` 则是把保存的 CSV 文件转换成一个可以筛选、排序的精美表格。
