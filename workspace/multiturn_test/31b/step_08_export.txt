<script>
    const CATEGORIES = {
        "支出": ["🍱 餐饮", "🚌 交通", "🛍️ 购物", "🎬 娱乐", "🏠 房租水电", "💊 医疗", "🌸 其他"],

    function saveBudgets() {
        const inputs = document.querySelectorAll('.budget-input');
        inputs.forEach(input => {
            budgets[input.dataset.cat] = parseFloat(input.value) || 0;

        localStorage.setItem('myFamilyBudgets', JSON.stringify(budgets));

    }

    function updateCategories() {
        const type = document.getElementById('type').value;
        const catSelect = document.getElementById('category');
        catSelect.innerHTML = '';
        CATEGORIES[type].forEach(cat => {

    function addRecord() {
        const date = document.getElementById('date').value;
        const type = document.getElementById('type').value;
        const category = document.getElementById('category').value;
        const amount = parseFloat(document.getElementById('amount').value);
        const note = document.getElementById('note').value;

        if (!date || isNaN(amount) || amount <= 0) {
            alert("请填好日期和正确的金额哦 🌸");
            return;
        }

        const record = { date, type, category, amount: type === "收入" ? amount : -amount, note };
        records.push(record);
        localStorage.setItem('myFamilyFinance', JSON.stringify(records));
        
        document.getElementById('amount').value = '';
        document.getElementById('note').value = '';
        render();
    }

    function clearAll() {
        if (confirm("确定要清空所有账单吗？")) {
            records = [];
            localStorage.removeItem('myFamilyFinance');
            render();
        }
    }

    // --- 核心导出功能 ---
    function exportToCSV() {
        if (records.length === 0) {
            alert("目前还没有记录可以导出哦 🌸");
            return;
        }

        // 1. 创建 CSV 表头 (包含 BOM 字符 \uFEFF 解决 Excel 中文乱码)
        let csvContent = "\uFEFF"; 
        csvContent += "日期,类型,类别,金额,备注\n";

        // 2. 遍历所有记录并添加到字符串
        records.forEach(r => {
            // 处理备注中可能存在的逗号，防止 CSV 分列错位
            const safeNote = r.note ? `"${r.note.replace(/"/g, '""')}"` : '""';
            csvContent += `${r.date},${r.type},${r.category},${r.amount},${safeNote}\n`;
        });

        // 3. 创建 Blob 文件对象并触发下载
        const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement("a");
        
        const dateStr = new Date().toISOString().split('T')[0];
        link.setAttribute("href", url);
        link.setAttribute("download", `家庭账单导出_${dateStr}.csv`);
        link.style.visibility = 'hidden';
        
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    }

    function render() {
        const tbody = document.getElementById('table-body');
        tbody.innerHTML = '';
        let totalBalance = 0;
        let mIncome = 0;
        let mExpense = 0;
        const expenseMap = {};

        const now = new Date();
        const currentMonth = now.getMonth();
        const currentYear = now.getFullYear();

        const sortedRecords = [...records].sort((a, b) => new Date(b.date) - new Date(a.date));

        sortedRecords.forEach(r => {
            totalBalance += r.amount;
            const rDate = new Date(r.date);
            if (rDate.getMonth() === currentMonth && rDate.getFullYear() === currentYear) {
                if (r.amount > 0) {
                    mIncome += r.amount;
                } else {
                    const absAmt = Math.abs(r.amount);
                    mExpense += absAmt;
                    expenseMap[r.category] = (expenseMap[r.category] || 0) + absAmt;
                }
            }

            tbody.innerHTML += `<tr>
                <td>${r.date}</td>
                <td>${r.category}</td>
                <td>${r.note}</td>
                <td class="${r.amount > 0 ? 'amt-positive' : 'amt-negative'}">
                    ${r.amount > 0 ? '+' : ''}${r.amount.toFixed(2)}
                </td>
            </tr>`;
        });

        document.getElementById('total-balance').innerText = `￥${totalBalance.toFixed(2)}`;
        document.getElementById('month-income').innerText = `￥${mIncome.toFixed(2)}`;
        document.getElementById('month-expense').innerText = `￥${mExpense.toFixed(2)}`;
        document.getElementById('month-net').innerText = `￥${(mIncome - mExpense).toFixed(2)}`;

        updateBudgetHealth(expenseMap);
        updateChart(expenseMap);
    }

    function updateBudgetHealth(expenseMap) {
        const statusList = document.getElementById('budget-status-list');
        statusList.innerHTML = '';
        CATEGORIES["支出"].forEach(cat => {
            const spent = expenseMap[cat] || 0;
            const limit = budgets[cat] || 0;
            if (limit === 0) return;
            let statusClass = 'status-safe';
            let statusText = '充足';
            let icon = '✅';
            if (spent > limit) {
                statusClass = 'status-danger';
                statusText = '超支';
                icon = '⚠️';
            } else if (spent > limit * 0.8) {
                statusClass = 'status-warning';
                statusText = '临界';
                icon = '🔔';
            }
            statusList.innerHTML += `
                <div class="status-tag ${statusClass}">
                    ${icon} ${cat}: ${spent.toFixed(0)} / ${limit} (${statusText})
                </div>
            `;
        });
    }

    function updateChart(expenseMap) {
        const ctx = document.getElementById('expenseChart').getContext('2d');
        const labels = Object.keys(expenseMap);
        const data = Object.values(expenseMap);
        if (labels.length === 0) {
            if(myChart) myChart.destroy();
            return;
        }
        if (myChart) myChart.destroy();
        myChart = new Chart(ctx, {
            type: 'pie',
            data: {
                labels: labels,
                datasets: [{
                    data: data,
                    backgroundColor: ['#a8bca1', '#dfcfbe', '#e9967a', '#8eb69b', '#d68a8a', '#b8c5d6', '#f2d5a6', '#c9ada7'],
                    borderWidth: 2, borderColor: '#ffffff'
                }]
            },
            options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'bottom' } } }
        });
    }

    initBudgetSetup();
    render();
</script>

### 🛠️ 导出功能是如何工作的？

1.  **CSV 格式**：我为你创建了一个 `.csv` 文件（Comma-Separated Values）。这种文件是用逗号分隔数据的文本文件，**Excel 能够完美识别并将其转换为表格**。
2.  **解决乱码 (BOM 标记)**：在 JavaScript 中，我特意在文件的开头加上了 `\uFEFF`。这是一个不可见的标记，告诉 Excel：“嘿，这个文件是用 UTF-8 编码的，请正确显示中文！”
3.  **处理特殊字符**：如果你在“备注”里写了逗号（例如：“买水果,一共3件”），如果不处理，Excel 会把这个备注拆分成两列。我在代码中加入了 `"${r.note}"` 的引号包裹处理，确保即使有逗号，备注依然会被完整地放在一个格子里。
4.  **全字段导出**：导出的表格包含：`日期` $\rightarrow$ `类型` $\rightarrow$ `类别` $\rightarrow$ `金额` $\rightarrow$ `备注`。完全满足报税或年度审计的需求。
