### 3. `app.js` (逻辑更新)

我们将添加筛选逻辑和撤销/重做栈管理。

```javascript
// app.js

document.addEventListener('DOMContentLoaded', () => {
    console.log("全家记账本应用已加载！");

    // --- DOM 元素引用 ---
    // 表单相关
    const recordForm = document.getElementById('record-form');
    const amountInput = document.getElementById('amount');
    const categorySelect = document.getElementById('category');
    const dateInput = document.getElementById('date');
    const memoInput = document.getElementById('memo');
    const memberSelect = document.getElementById('member');
    const newMemberNameInput = document.getElementById('new-member-name');
    const addMemberBtn = document.getElementById('add-member-btn');
    const addRecordBtn = document.getElementById('add-record-btn');

    // 错误信息
    const amountError = document.getElementById('amount-error');
    const categoryError = document.getElementById('category-error');
    const dateError = document.getElementById('date-error');
    const memberError = document.getElementById('member-error');

    // 记录列表及总览
    const recordList = document.getElementById('record-list');
    const noRecordsMessage = document.getElementById('no-records-message');
    const totalIncomeSpan = document.getElementById('total-income');
    const totalExpenseSpan = document.getElementById('total-expense');
    const netBalanceSpan = document.getElementById('net-balance');

    // 筛选区相关
    const filterFromDateInput = document.getElementById('filter-from-date');
    const filterToDateInput = document.getElementById('filter-to-date');
    const filterCategoryCheckboxes = document.querySelectorAll('.category-checkboxes input[type="checkbox"]');
    const filterMemberSelect = document.getElementById('filter-member');
    const filterMinAmountInput = document.getElementById('filter-min-amount');
    const filterMaxAmountInput = document.getElementById('filter-max-amount');
    const filterKeywordInput = document.getElementById('filter-keyword');
    const applyFilterBtn = document.getElementById('apply-filter-btn');
    const clearFilterBtn = document.getElementById('clear-filter-btn');

    // --- Local Storage Keys ---
    const LOCAL_STORAGE_RECORDS_KEY = 'familyLedger.records';
    const LOCAL_STORAGE_MEMBERS_KEY = 'familyLedger.members';

    // --- 数据数组 ---
    let records = []; // 存储所有记账记录
    let members = ['爸爸', '妈妈', '孩子']; // 存储家庭成员

    // --- 筛选条件对象 ---
    let currentFilters = {
        fromDate: '',
        toDate: '',
        categories: [], // 存储选中的类别
        member: '',
        minAmount: '',
        maxAmount: '',
        keyword: ''
    };

    // --- 撤销/重做栈 ---
    const MAX_HISTORY_LENGTH = 50; // 最大历史记录步数
    let history = []; // 存储records状态的快照
    let historyPointer = -1; // 指向history数组中当前records状态的索引

    // --- 初始化函数 ---
    function initializeApp() {
        loadData();
        renderMembers(); // 渲染添加表单和筛选表单的成员列表
        pushState(records); // 将初始状态推入历史栈
        renderFilteredRecords(); // 首次加载时渲染（可能已有记录）
        updateSummary();
        // 设置日期输入框的最大值为今天
        const today = new Date().toISOString().split('T')[0];
        dateInput.max = today;
        filterFromDateInput.max = today;
        filterToDateInput.max = today;
        
        setupFilterListeners(); // 设置筛选器事件监听
    }

    // --- 数据加载与保存 (localStorage) ---
    function loadData() {
        const storedRecords = localStorage.getItem(LOCAL_STORAGE_RECORDS_KEY);
        if (storedRecords) {
            records = JSON.parse(storedRecords);
        }

        const storedMembers = localStorage.getItem(LOCAL_STORAGE_MEMBERS_KEY);
        if (storedMembers) {
            members = JSON.parse(storedMembers);
        }
    }

    function saveRecords() {
        localStorage.setItem(LOCAL_STORAGE_RECORDS_KEY, JSON.stringify(records));
    }

    function saveMembers() {
        localStorage.setItem(LOCAL_STORAGE_MEMBERS_KEY, JSON.stringify(members));
    }

    // --- 成员管理 ---
    function renderMembers() {
        // 更新添加记录表单的成员选择
        memberSelect.innerHTML = '<option value="">请选择成员</option>';
        // 更新筛选表单的成员选择
        filterMemberSelect.innerHTML = '<option value="">所有成员</option>';

        members.forEach(member => {
            const option = document.createElement('option');
            option.value = member;
            option.textContent = member;
            memberSelect.appendChild(option);

            const filterOption = document.createElement('option');
            filterOption.value = member;
            filterOption.textContent = member;
            filterMemberSelect.appendChild(filterOption);
        });
        // 重新设置筛选成员选择框的选中值
        filterMemberSelect.value = currentFilters.member;
    }

    addMemberBtn.addEventListener('click', () => {
        const newMemberName = newMemberNameInput.value.trim();
        if (newMemberName && !members.includes(newMemberName)) {
            members.push(newMemberName);
            saveMembers();
            renderMembers();
            newMemberNameInput.value = ''; // 清空输入框
        } else if (newMemberName && members.includes(newMemberName)) {
            alert('该成员已存在！');
        } else {
            alert('请输入成员姓名！');
        }
    });

    // --- 表单校验 ---
    function showError(element, message) {
        element.textContent = message;
        element.classList.add('visible');
    }

    function hideError(element) {
        element.textContent = '';
        element.classList.remove('visible');
    }

    function validateForm() {
        let isValid = true;

        // 1. 金额校验
        const amount = parseFloat(amountInput.value);
        if (isNaN(amount) || amount <= 0) {
            showError(amountError, '请输入正数金额');
            isValid = false;
        } else {
            hideError(amountError);
        }

        // 2. 类别校验
        if (categorySelect.value === '') {
            showError(categoryError, '请选择类别');
            isValid = false;
        } else {
            hideError(categoryError);
        }

        // 3. 日期校验
        const selectedDate = new Date(dateInput.value);
        const today = new Date();
        today.setHours(0, 0, 0, 0); // 清除时间，只比较日期

        if (!dateInput.value) { // 检查日期是否为空
            showError(dateError, '请选择日期');
            isValid = false;
        } else if (selectedDate > today) {
            showError(dateError, '日期不能是未来日期');
            isValid = false;
        } else {
            hideError(dateError);
        }

        // 4. 成员校验
        if (memberSelect.value === '') {
            showError(memberError, '请选择成员');
            isValid = false;
        } else {
            hideError(memberError);
        }

        return isValid;
    }

    // --- 添加记录 ---
    recordForm.addEventListener('submit', (e) => {
        e.preventDefault(); // 阻止表单默认提交行为

        if (!validateForm()) {
            return; // 校验失败，停止提交
        }

        const amount = parseFloat(amountInput.value);
        const category = categorySelect.value;
        const date = dateInput.value; // 日期字符串 'YYYY-MM-DD'
        const memo = memoInput.value.trim();
        const member = memberSelect.value;

        // 判断是收入还是支出
        const type = (category === '工资') ? 'income' : 'expense';

        const newRecord = {
            id: Date.now(), // 简单生成唯一ID
            amount: amount,
            category: category,
            date: date,
            memo: memo,
            member: member,
            type: type
        };

        records.push(newRecord);
        saveRecords();
        pushState(records); // 将新状态推入历史栈
        recordForm.reset(); // 清空表单
        dateInput.max = new Date().toISOString().split('T')[0]; // 重设max日期
        renderFilteredRecords(); // 重新渲染（可能应用了筛选）
        updateSummary();
    });

    // --- 渲染记录列表 ---
    function renderRecords(recordsToRender) {
        recordList.innerHTML = ''; // 清空现有列表

        if (recordsToRender.length === 0) {
            noRecordsMessage.classList.remove('hidden');
        } else {
            noRecordsMessage.classList.add('hidden');
            // recordsToRender.sort((a, b) => new Date(b.date) - new Date(a.date)); // 排序应该在筛选前进行，或者在获取原始记录时

            recordsToRender.forEach(record => {
                const li = document.createElement('li');
                li.setAttribute('data-id', record.id); // 用于删除

                const amountClass = record.type === 'income' ? 'income' : 'expense';
                const displayAmount = record.type === 'income' ? `+${record.amount.toFixed(2)}` : `-${record.amount.toFixed(2)}`;
                const displayMemo = record.memo ? ` (${record.memo})` : ''; // 前面加空格

                li.innerHTML = `
                    <span class="description">${record.category}${displayMemo}</span>
                    <span class="member">${record.member}</span>
                    <span class="date">${record.date}</span>
                    <span class="amount ${amountClass}">${displayAmount} 元</span>
                    <button class="delete-btn">删除</button>
                `;
                recordList.appendChild(li);
            });
        }
    }

    // --- 删除记录 ---
    recordList.addEventListener('click', (e) => {
        if (e.target.classList.contains('delete-btn')) {
            const recordId = parseInt(e.target.closest('li').getAttribute('data-id'));
            records = records.filter(record => record.id !== recordId);
            saveRecords();
            pushState(records); // 将新状态推入历史栈
            renderFilteredRecords(); // 重新渲染（可能应用了筛选）
            updateSummary();
        }
    });

    // --- 更新总览 ---
    function updateSummary() {
        let totalIncome = 0;
        let totalExpense = 0;

        records.forEach(record => {
            if (record.type === 'income') {
                totalIncome += record.amount;
            } else {
                totalExpense += record.amount;
            }
        });

        const netBalance = totalIncome - totalExpense;

        totalIncomeSpan.textContent = totalIncome.toFixed(2);
        totalExpenseSpan.textContent = totalExpense.toFixed(2);
        netBalanceSpan.textContent = netBalance.toFixed(2);
    }

    // --- 筛选功能 ---
    function getFilteredRecords() {
        let filtered = [...records]; // 从完整的记录开始筛选

        // 1. 日期范围
        if (currentFilters.fromDate) {
            const fromDate = new Date(currentFilters.fromDate + 'T00:00:00'); // 确保比较的是日期而不是时间
            filtered = filtered.filter(record => new Date(record.date + 'T00:00:00') >= fromDate);
        }
        if (currentFilters.toDate) {
            const toDate = new Date(currentFilters.toDate + 'T23:59:59'); // 确保比较到当天结束
            filtered = filtered.filter(record => new Date(record.date + 'T00:00:00') <= toDate);
        }

        // 2. 类别多选
        if (currentFilters.categories.length > 0) {
            filtered = filtered.filter(record => currentFilters.categories.includes(record.category));
        }

        // 3. 成员单选
        if (currentFilters.member) {
            filtered = filtered.filter(record => record.member === currentFilters.member);
        }

        // 4. 金额范围
        const minAmount = parseFloat(currentFilters.minAmount);
        const maxAmount = parseFloat(currentFilters.maxAmount);
        if (!isNaN(minAmount)) {
            filtered = filtered.filter(record => record.amount >= minAmount);
        }
        if (!isNaN(maxAmount)) {
            filtered = filtered.filter(record => record.amount <= maxAmount);
        }

        // 5. 关键词搜索备注
        if (currentFilters.keyword) {
            const keyword = currentFilters.keyword.toLowerCase();
            filtered = filtered.filter(record => 
                (record.memo && record.memo.toLowerCase().includes(keyword)) ||
                (record.category && record.category.toLowerCase().includes(keyword)) // 也可以搜索类别
            );
        }

        // 排序：始终按日期降序
        return filtered.sort((a, b) => new Date(b.date) - new Date(a.date));
    }

    function renderFilteredRecords() {
        const filteredRecords = getFilteredRecords();
        renderRecords(filteredRecords);
    }

    function updateFilterStateFromUI() {
        currentFilters.fromDate = filterFromDateInput.value;
        currentFilters.toDate = filterToDateInput.value;
        currentFilters.member = filterMemberSelect.value;
        currentFilters.minAmount = filterMinAmountInput.value;
        currentFilters.maxAmount = filterMaxAmountInput.value;
        currentFilters.keyword = filterKeywordInput.value.trim();

        currentFilters.categories = Array.from(filterCategoryCheckboxes)
            .filter(cb => cb.checked && cb.value !== 'all')
            .map(cb => cb.value);
        
        // 如果'全部'被选中，清空categories
        const allChecked = document.getElementById('cat-all').checked;
        if (allChecked) {
            currentFilters.categories = [];
        }
    }

    function applyFilters() {
        updateFilterStateFromUI();
        renderFilteredRecords();
    }

    function clearFilters() {
        // 重置筛选UI
        filterFromDateInput.value = '';
        filterToDateInput.value = '';
        filterMemberSelect.value = '';
        filterMinAmountInput.value = '';
        filterMaxAmountInput.value = '';
        filterKeywordInput.value = '';
        
        // 重置类别复选框
        filterCategoryCheckboxes.forEach(cb => {
            cb.checked = (cb.value === 'all'); // 默认选中“全部”
        });

        // 重置筛选状态对象
        currentFilters = {
            fromDate: '', toDate: '', categories: [], member: '', minAmount: '', maxAmount: '', keyword: ''
        };
        renderFilteredRecords();
    }

    function setupFilterListeners() {
        // 日期范围
        filterFromDateInput.addEventListener('change', applyFilters);
        filterToDateInput.addEventListener('change', applyFilters);

        // 类别多选
        filterCategoryCheckboxes.forEach(checkbox => {
            checkbox.addEventListener('change', (e) => {
                if (e.target.value === 'all') {
                    // 如果选中“全部”，则取消其他所有，并确保“全部”选中
                    filterCategoryCheckboxes.forEach(cb => {
                        if (cb.value !== 'all') cb.checked = false;
                    });
                    e.target.checked = true; // 确保"全部"选中
                } else {
                    // 如果选中其他类别，则取消“全部”
                    document.getElementById('cat-all').checked = false;
                    // 如果所有非“全部”都被取消，则选中“全部”
                    const anyCategorySelected = Array.from(filterCategoryCheckboxes).some(cb => cb.checked && cb.value !== 'all');
                    if (!anyCategorySelected) {
                        document.getElementById('cat-all').checked = true;
                    }
                }
                applyFilters();
            });
        });

        // 成员单选
        filterMemberSelect.addEventListener('change', applyFilters);

        // 金额范围
        filterMinAmountInput.addEventListener('input', applyFilters);
        filterMaxAmountInput.addEventListener('input', applyFilters);

        // 备注关键词搜索
        filterKeywordInput.addEventListener('input', applyFilters); // 实时搜索

        // 按钮
        applyFilterBtn.addEventListener('click', applyFilters);
        clearFilterBtn.addEventListener('click', clearFilters);

        // 初始化时，确保 "全部" 类别被选中
        document.getElementById('cat-all').checked = true;
    }

    // --- 撤销/重做功能 ---
    function pushState(newRecords) {
        // 清除在当前指针之后的所有历史（即重做栈中的内容失效）
        if (historyPointer < history.length - 1) {
            history = history.slice(0, historyPointer + 1);
        }
        
        // 深拷贝当前记录状态并推入历史栈
        history.push(JSON.parse(JSON.stringify(newRecords)));
        historyPointer++;

        // 限制历史栈大小
        if (history.length > MAX_HISTORY_LENGTH) {
            history.shift(); // 移除最旧的记录
            historyPointer--; // 对应调整指针
        }
        // console.log("History pushed. Pointer:", historyPointer, "History length:", history.length);
    }

    function applyHistoryState(state) {
        records = JSON.parse(JSON.stringify(state)); // 恢复到指定历史状态
        saveRecords(); // 撤销/重做后也要更新localStorage
        renderFilteredRecords(); // 重新渲染列表（考虑筛选）
        updateSummary();
    }

    function undo() {
        if (historyPointer > 0) {
            historyPointer--;
            applyHistoryState(history[historyPointer]);
            // console.log("Undo. Pointer:", historyPointer);
        } else {
            console.log("Cannot undo further.");
        }
    }

    function redo() {
        if (historyPointer < history.length - 1) {
            historyPointer++;
            applyHistoryState(history[historyPointer]);
            // console.log("Redo. Pointer:", historyPointer);
        } else {
            console.log("Cannot redo further.");
        }
    }

    // 键盘事件监听 Ctrl+Z 和 Ctrl+Y
    document.addEventListener('keydown', (e) => {
        if (e.ctrlKey || e.metaKey) { // 兼容Mac的Cmd键
            if (e.key === 'z' || e.key === 'Z') {
                e.preventDefault(); // 阻止浏览器默认的撤销行为
                undo();
            } else if (e.key === 'y' || e.key === 'Y') {
                e.preventDefault(); // 阻止浏览器默认的重做行为
                redo();
            }
        }
    });

    // --- 页面加载时执行初始化 ---
    initializeApp();
});
```

### 如何运行和测试：

1.  将上述代码分别保存为 `index.html`, `style.css`, `app.js` 到同一个文件夹中。
2.  用浏览器打开 `index.html` 文件。

**新功能测试点：**

**1. 筛选区：**
*   **日期范围**: 选择开始和结束日期，验证只有在该范围内的记录显示。
*   **类别多选**:
    *   选中“吃饭”和“交通”，看是否只显示这两种类别的记录。
    *   选中“全部”时，其他类别自动取消选中。
    *   选中任一类别时，“全部”自动取消选中。
    *   当所有其他类别都取消选中时，“全部”自动选中。
*   **成员单选**: 选择一个成员，看是否只显示该成员的记录。
*   **金额范围**: 输入最小/最大金额，看是否符合条件的记录显示。
*   **关键词搜索备注**: 在备注或类别中输入关键词，看是否能实时过滤。
*   **条件叠加**: 尝试同时使用日期、类别、成员、金额和关键词，验证所有条件都生效。
*   **清除按钮**: 点击“清除筛选”，所有筛选条件应该被重置，并显示所有记录。

**2. 撤销重做：**
*   **添加记录**: 添加几条记录，然后按 `Ctrl+Z` (或 Mac 上的 `Cmd+Z`)，记录应该一条条被撤销。
*   **删除记录**: 删除一条记录，然后按 `Ctrl+Z`，被删除的记录应该重新出现。
*   **重做**: 在撤销几次后，按 `Ctrl+Y` (或 Mac 上的 `Cmd+Y`)，被撤销的记录应该一条条重新出现。
*   **新操作打断重做**: 撤销几次后，执行一个新的添加或删除操作，此时 `Ctrl+Y` 应该失效（重做栈被清空），只能继续 `Ctrl+Z`。
*   **LocalStorage 交互**: 撤销/重做操作后，刷新页面，数据应该与撤销/重做后的最终状态一致（因为 `applyHistoryState` 中调用了 `saveRecords`）。
*   **限制**: 连续执行超过 50 个操作，看看最旧的记录是否会被自动清除（这在控制台不容易直接观察，但逻辑上已实现）。

这个版本包含了所有要求的功能，并做了一些细节优化，例如日期输入框的最大值限制为今天，筛选器类别多选的逻辑优化，以及撤销重做时对 `localStorage` 的更新。
