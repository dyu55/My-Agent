**主要改动点：**

*   为“添加新成员”按钮添加了样式，并调整了 `member` 字段和按钮的布局，使它们可以排在一行。

### 3. 更新 `script.js` (逻辑)

这是最复杂的部分。我们需要：
1.  获取所有相关的表单元素和错误提示 `<span>` 元素。
2.  初始化交易列表和账户余额（暂时都从0开始）。
3.  实现一个验证函数，检查所有表单字段。
4.  实现一个提交表单的事件监听器，并在验证通过后添加交易。
5.  实现添加新成员的功能。
6.  实现删除交易的功能（虽然还没要求，但现在框架搭好，先预留）。
7.  更新总览数据。

将以下代码替换你 `script.js` 的全部内容：

```javascript
// script.js

document.addEventListener('DOMContentLoaded', () => {
    console.log('JavaScript 文件已成功加载并准备就绪！');

    // --- 获取 DOM 元素 ---
    const transactionForm = document.getElementById('transaction-form');
    const typeInput = document.getElementById('type');
    const amountInput = document.getElementById('amount');
    const categoryInput = document.getElementById('category');
    const dateInput = document.getElementById('date');
    const memberInput = document.getElementById('member');
    const descriptionInput = document.getElementById('description');
    const addMemberBtn = document.getElementById('add-member-btn');

    const transactionList = document.getElementById('transaction-list');
    const totalIncomeSpan = document.getElementById('total-income');
    const totalExpenseSpan = document.getElementById('total-expense');
    const currentBalanceSpan = document.getElementById('current-balance');

    // 错误信息显示元素
    const amountError = document.getElementById('amount-error');
    const categoryError = document.getElementById('category-error');
    const dateError = document.getElementById('date-error');
    const memberError = document.getElementById('member-error');

    // --- 数据存储 (暂时在内存中，页面刷新会丢失) ---
    let transactions = [];
    let members = ['爸爸', '妈妈', '孩子']; // 初始成员列表

    // --- 初始化函数 ---
    function init() {
        renderMembers(); // 渲染成员下拉列表
        updateSummary(); // 更新总览
        renderTransactions(); // 渲染所有交易（目前为空）
    }

    // --- 辅助函数 ---

    /**
     * 渲染成员下拉列表
     */
    function renderMembers() {
        memberInput.innerHTML = '<option value="">请选择成员</option>'; // 清空并添加默认选项
        members.forEach(member => {
            const option = document.createElement('option');
            option.value = member;
            option.textContent = member;
            memberInput.appendChild(option);
        });
    }

    /**
     * 显示错误信息
     * @param {HTMLElement} errorElement 错误信息span元素
     * @param {string} message 错误信息内容
     */
    function showError(errorElement, message) {
        errorElement.textContent = message;
        errorElement.classList.add('visible');
    }

    /**
     * 隐藏错误信息
     * @param {HTMLElement} errorElement 错误信息span元素
     */
    function hideError(errorElement) {
        errorElement.textContent = '';
        errorElement.classList.remove('visible');
    }

    /**
     * 表单验证函数
     * @returns {boolean} 如果表单验证通过返回true，否则返回false
     */
    function validateForm() {
        let isValid = true;

        // 1. 金额验证
        const amount = parseFloat(amountInput.value);
        if (isNaN(amount) || amount <= 0) {
            showError(amountError, '金额必须是正数且不能为空。');
            isValid = false;
        } else {
            hideError(amountError);
        }

        // 2. 类别验证
        if (categoryInput.value === '') {
            showError(categoryError, '请选择一个类别。');
            isValid = false;
        } else {
            hideError(categoryError);
        }

        // 3. 日期验证
        const transactionDate = new Date(dateInput.value);
        const today = new Date();
        today.setHours(0, 0, 0, 0); // 清除时间，只比较日期
        if (!dateInput.value) { // 检查日期是否为空
            showError(dateError, '日期不能为空。');
            isValid = false;
        } else if (transactionDate > today) {
            showError(dateError, '日期不能是未来的日期。');
            isValid = false;
        } else {
            hideError(dateError);
        }

        // 4. 成员验证
        if (memberInput.value === '') {
            showError(memberError, '请选择一个成员。');
            isValid = false;
        } else {
            hideError(memberError);
        }

        return isValid;
    }

    /**
     * 更新账户总览 (总收入, 总支出, 当前余额)
     */
    function updateSummary() {
        let income = 0;
        let expense = 0;

        transactions.forEach(transaction => {
            if (transaction.type === 'income') {
                income += transaction.amount;
            } else {
                expense += transaction.amount;
            }
        });

        totalIncomeSpan.textContent = `¥ ${income.toFixed(2)}`;
        totalExpenseSpan.textContent = `¥ ${expense.toFixed(2)}`;
        currentBalanceSpan.textContent = `¥ ${(income - expense).toFixed(2)}`;
    }

    /**
     * 渲染所有交易记录到列表中
     */
    function renderTransactions() {
        transactionList.innerHTML = ''; // 清空现有列表
        transactions.forEach((transaction, index) => {
            const listItem = document.createElement('li');
            listItem.classList.add('transaction-item', transaction.type); // 添加收支类型类
            
            // 格式化日期
            const displayDate = new Date(transaction.date).toLocaleDateString('zh-CN', {
                year: 'numeric',
                month: '2-digit',
                day: '2-digit'
            });

            // 构建显示文本
            let descriptionText = transaction.description ? ` (${transaction.description})` : '';
            const sign = transaction.type === 'income' ? '+' : '-';
            
            listItem.innerHTML = `
                <span>
                    ${transaction.category} - ${transaction.member} - ${displayDate}${descriptionText}
                </span>
                <span class="amount">${sign} ¥ ${transaction.amount.toFixed(2)}</span>
                <button class="delete-btn" data-index="${index}">X</button>
            `;
            transactionList.appendChild(listItem);
        });
        // 确保新添加的删除按钮也能绑定事件
        addDeleteEventListeners();
    }

    /**
     * 为所有删除按钮添加事件监听器
     */
    function addDeleteEventListeners() {
        document.querySelectorAll('.delete-btn').forEach(button => {
            button.onclick = (event) => {
                const index = parseInt(event.target.dataset.index);
                deleteTransaction(index);
            };
        });
    }

    /**
     * 删除交易
     * @param {number} index 要删除交易的索引
     */
    function deleteTransaction(index) {
        if (confirm('确定要删除这笔交易吗？')) {
            transactions.splice(index, 1); // 从数组中移除
            updateSummary();
            renderTransactions(); // 重新渲染列表
        }
    }


    // --- 事件监听器 ---

    // 提交表单
    transactionForm.addEventListener('submit', (event) => {
        event.preventDefault(); // 阻止表单默认提交行为

        if (validateForm()) {
            const newTransaction = {
                id: Date.now(), // 简单生成一个唯一ID
                type: typeInput.value,
                amount: parseFloat(amountInput.value),
                category: categoryInput.value,
                date: dateInput.value, // 存储为 'YYYY-MM-DD' 格式
                member: memberInput.value,
                description: descriptionInput.value
            };

            transactions.push(newTransaction);
            console.log('新交易已添加:', newTransaction);

            updateSummary(); // 更新总览
            renderTransactions(); // 重新渲染交易列表

            // 清空表单
            transactionForm.reset();
            // 确保类别和成员选回默认值
            categoryInput.value = ''; 
            memberInput.value = '';
            // 隐藏所有错误信息
            hideError(amountError);
            hideError(categoryError);
            hideError(dateError);
            hideError(memberError);

        } else {
            console.log('表单验证失败，请检查输入。');
        }
    });

    // 添加新成员按钮点击事件
    addMemberBtn.addEventListener('click', () => {
        const newMemberName = prompt('请输入新成员的名字:');
        if (newMemberName && newMemberName.trim() !== '' && !members.includes(newMemberName.trim())) {
            members.push(newMemberName.trim());
            renderMembers(); // 重新渲染成员列表
            memberInput.value = newMemberName.trim(); // 选中新添加的成员
            hideError(memberError); // 清除成员错误提示
        } else if (newMemberName && members.includes(newMemberName.trim())) {
             alert('该成员已存在！');
        }
    });

    // 实时清除错误信息（可选，但用户体验更好）
    amountInput.addEventListener('input', () => hideError(amountError));
    categoryInput.addEventListener('change', () => hideError(categoryError));
    dateInput.addEventListener('change', () => hideError(dateError));
    memberInput.addEventListener('change', () => hideError(memberError));


    // --- 页面初始化 ---
    init();
});
```

**主要改动点和解释：**

1.  **DOM 元素获取：**
    *   获取了所有新增的表单字段的 `input` 和 `select` 元素。
    *   获取了所有错误提示 `<span>` 元素，方便JS操作它们的文本和可见性。
    *   获取了“添加新成员”按钮。

2.  **数据存储：**
    *   `transactions = []`: 一个数组，用于存储所有的交易对象。每个交易对象将包含 `id`, `type`, `amount`, `category`, `date`, `member`, `description` 等属性。
    *   `members = ['爸爸', '妈妈', '孩子']`: 一个数组，存储家庭成员的名字，用于填充成员下拉列表。

3.  **`init()` 函数：** 在页面加载完成后调用，用于初始化成员列表、账户总览和交易列表。

4.  **`renderMembers()`：**
    *   清空 `memberInput` 的现有选项。
    *   遍历 `members` 数组，为每个成员创建一个 `<option>` 元素并添加到 `memberInput`。

5.  **`showError()` / `hideError()`：**
    *   这两个函数封装了显示和隐藏错误信息的逻辑，通过修改 `textContent` 和添加/移除 `visible` 类来实现。

6.  **`validateForm()`：** 这是核心的验证逻辑。
    *   **金额：** 检查是否是数字，是否大于0。
    *   **类别：** 检查是否选择了非空选项。
    *   **日期：** 检查是否为空，是否是未来的日期。将输入的日期和当前日期都设置为午夜零点，确保只比较日期部分。
    *   **成员：** 检查是否选择了非空选项。
    *   如果任何验证失败，它会调用 `showError()` 显示提示，并设置 `isValid = false`。
    *   最后返回 `isValid`。

7.  **`updateSummary()`：**
    *   遍历 `transactions` 数组，计算总收入和总支出。
    *   更新 `totalIncomeSpan`, `totalExpenseSpan`, `currentBalanceSpan` 的文本内容，并使用 `toFixed(2)` 格式化为两位小数。

8.  **`renderTransactions()`：**
    *   清空 `transactionList`。
    *   遍历 `transactions` 数组，为每个交易创建一个 `<li>` 元素。
    *   `<li>` 元素会根据交易类型 (`income` 或 `expense`) 添加相应的CSS类，以便应用不同的颜色。
    *   格式化日期以友好的方式显示。
    *   动态生成交易的描述（类别 - 成员 - 日期 - 备注）。
    *   添加删除按钮，并通过 `data-index` 属性存储交易在 `transactions` 数组中的索引，方便后续删除。
    *   调用 `addDeleteEventListeners()` 确保新的删除按钮绑定事件。

9.  **`addDeleteEventListeners()` / `deleteTransaction()`：**
    *   `addDeleteEventListeners()` 遍历所有 `.delete-btn` 并为它们添加点击事件。
    *   `deleteTransaction()` 接收一个索引，从 `transactions` 数组中移除对应元素，然后重新更新总览和渲染列表。

10. **表单提交事件监听器 (`transactionForm.addEventListener('submit')`)：**
    *   `event.preventDefault()`: 阻止页面刷新。
    *   调用 `validateForm()`。如果通过，则：
        *   创建一个新的 `transaction` 对象，收集所有表单数据。
        *   将其添加到 `transactions` 数组。
        *   调用 `updateSummary()` 和 `renderTransactions()` 来更新UI。
        *   `transactionForm.reset()`: 清空表单字段。
        *   手动将 `category` 和 `member` `select` 元素的值重置为空（`reset()` 不会重置 `select` 到默认选项）。
        *   隐藏所有错误信息。

11. **添加新成员功能 (`addMemberBtn.addEventListener('click')`)：**
    *   弹出一个 `prompt` 框让用户输入新成员名字。
    *   进行简单验证（不为空，不重复）。
    *   将新成员添加到 `members` 数组。
    *   调用 `renderMembers()` 刷新成员下拉列表，并自动选中新成员。

12. **实时清除错误信息：**
    *   为 `amount`, `category`, `date`, `member` 的输入框添加了 `input` 或 `change` 事件监听器，当用户开始修改输入时，会自动隐藏对应的错误提示。这提供了更好的用户体验。
