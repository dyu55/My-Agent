### 3. 更新 `script.js` (逻辑)

这将是最大的改动。我们需要：
1.  获取所有筛选和搜索相关的DOM元素。
2.  扩展 `init()` 函数来填充筛选类别的下拉列表。
3.  实现一个新的 `filterTransactions()` 函数，它将根据所有当前的筛选条件来过滤 `transactions` 数组。
4.  修改 `renderTransactions()` 函数，使其现在渲染的是过滤后的交易列表。
5.  为所有筛选和搜索输入框添加事件监听器，当它们的值改变时，就重新应用筛选并渲染。
6.  实现“清除所有筛选”按钮的逻辑。

将以下代码替换你 `script.js` 的全部内容：

```javascript
// script.js

document.addEventListener('DOMContentLoaded', () => {
    console.log('JavaScript 文件已成功加载并准备就绪！');

    // --- 获取 DOM 元素 ---
    const transactionForm = document.getElementById('transaction-form');
    const typeInput = document.getElementById('type');
    const amountInput = document.getElementById('amount');
    const categoryInput = document.getElementById('category');
    const dateInput = document.getElementById('date');
    const memberInput = document.getElementById('member');
    const descriptionInput = document.getElementById('description');
    const addMemberBtn = document.getElementById('add-member-btn');

    const transactionList = document.getElementById('transaction-list');
    const totalIncomeSpan = document.getElementById('total-income');
    const totalExpenseSpan = document.getElementById('total-expense');
    const currentBalanceSpan = document.getElementById('current-balance');

    // 错误信息显示元素
    const amountError = document.getElementById('amount-error');
    const categoryError = document.getElementById('category-error');
    const dateError = document.getElementById('date-error');
    const memberError = document.getElementById('member-error');

    // 筛选和搜索元素
    const searchKeywordInput = document.getElementById('search-keyword');
    const filterStartDateInput = document.getElementById('filter-start-date');
    const filterEndDateInput = document.getElementById('filter-end-date');
    const filterCategorySelect = document.getElementById('filter-category');
    const filterMemberSelect = document.getElementById('filter-member');
    const filterMinAmountInput = document.getElementById('filter-min-amount');
    const filterMaxAmountInput = document.getElementById('filter-max-amount');
    const applyFiltersBtn = document.getElementById('apply-filters-btn');
    const clearFiltersBtn = document.getElementById('clear-filters-btn');


    // --- 数据存储 (暂时在内存中，页面刷新会丢失) ---
    let transactions = [];
    let members = ['爸爸', '妈妈', '孩子']; // 初始成员列表
    const categories = ['吃饭', '交通', '购物', '房租', '水电', '娱乐', '医疗', '教育', '工资', '其他']; // 所有类别

    // --- 初始化函数 ---
    function init() {
        renderMembers(); // 渲染成员下拉列表 (添加表单和筛选表单)
        renderFilterCategories(); // 渲染筛选类别下拉列表
        updateSummary(); // 更新总览
        renderTransactions(); // 渲染所有交易（目前为空）
    }

    // --- 辅助函数 ---

    /**
     * 渲染成员下拉列表 (用于添加表单和筛选表单)
     */
    function renderMembers() {
        // 添加表单的成员列表
        memberInput.innerHTML = '<option value="">请选择成员</option>';
        members.forEach(member => {
            const option = document.createElement('option');
            option.value = member;
            option.textContent = member;
            memberInput.appendChild(option);
        });

        // 筛选表单的成员列表
        filterMemberSelect.innerHTML = '<option value="">所有成员</option>';
        members.forEach(member => {
            const option = document.createElement('option');
            option.value = member;
            option.textContent = member;
            filterMemberSelect.appendChild(option);
        });
    }

    /**
     * 渲染筛选类别的下拉列表
     */
    function renderFilterCategories() {
        filterCategorySelect.innerHTML = ''; // 清空现有选项
        categories.forEach(cat => {
            const option = document.createElement('option');
            option.value = cat;
            option.textContent = cat;
            filterCategorySelect.appendChild(option);
        });
    }

    /**
     * 显示错误信息
     * @param {HTMLElement} errorElement 错误信息span元素
     * @param {string} message 错误信息内容
     */
    function showError(errorElement, message) {
        errorElement.textContent = message;
        errorElement.classList.add('visible');
    }

    /**
     * 隐藏错误信息
     * @param {HTMLElement} errorElement 错误信息span元素
     */
    function hideError(errorElement) {
        errorElement.textContent = '';
        errorElement.classList.remove('visible');
    }

    /**
     * 表单验证函数 (用于添加新交易)
     * @returns {boolean} 如果表单验证通过返回true，否则返回false
     */
    function validateForm() {
        let isValid = true;

        // 1. 金额验证
        const amount = parseFloat(amountInput.value);
        if (isNaN(amount) || amount <= 0) {
            showError(amountError, '金额必须是正数且不能为空。');
            isValid = false;
        } else {
            hideError(amountError);
        }

        // 2. 类别验证
        if (categoryInput.value === '') {
            showError(categoryError, '请选择一个类别。');
            isValid = false;
        } else {
            hideError(categoryError);
        }

        // 3. 日期验证
        const transactionDate = new Date(dateInput.value);
        const today = new Date();
        today.setHours(0, 0, 0, 0); // 清除时间，只比较日期
        if (!dateInput.value) { // 检查日期是否为空
            showError(dateError, '日期不能为空。');
            isValid = false;
        } else if (transactionDate > today) {
            showError(dateError, '日期不能是未来的日期。');
            isValid = false;
        } else {
            hideError(dateError);
        }

        // 4. 成员验证
        if (memberInput.value === '') {
            showError(memberError, '请选择一个成员。');
            isValid = false;
        } else {
            hideError(memberError);
        }

        return isValid;
    }

    /**
     * 更新账户总览 (总收入, 总支出, 当前余额)
     */
    function updateSummary() {
        let income = 0;
        let expense = 0;

        transactions.forEach(transaction => {
            if (transaction.type === 'income') {
                income += transaction.amount;
            } else {
                expense += transaction.amount;
            }
        });

        totalIncomeSpan.textContent = `¥ ${income.toFixed(2)}`;
        totalExpenseSpan.textContent = `¥ ${expense.toFixed(2)}`;
        currentBalanceSpan.textContent = `¥ ${(income - expense).toFixed(2)}`;
    }

    /**
     * 根据当前筛选条件过滤交易记录
     * @returns {Array} 过滤后的交易数组
     */
    function filterTransactions() {
        let filtered = [...transactions]; // 复制一份原始交易记录，避免修改原数组

        const keyword = searchKeywordInput.value.toLowerCase();
        const startDate = filterStartDateInput.value ? new Date(filterStartDateInput.value + 'T00:00:00') : null;
        const endDate = filterEndDateInput.value ? new Date(filterEndDateInput.value + 'T23:59:59') : null; // 结束日期到当天的最后一秒
        
        // 获取所有选中的类别
        const selectedCategories = Array.from(filterCategorySelect.selectedOptions).map(option => option.value);
        const selectedMember = filterMemberSelect.value;
        const minAmount = parseFloat(filterMinAmountInput.value);
        const maxAmount = parseFloat(filterMaxAmountInput.value);

        filtered = filtered.filter(transaction => {
            const transactionDate = new Date(transaction.date + 'T00:00:00'); // 统一格式，方便比较

            // 1. 关键词搜索 (备注)
            if (keyword && !transaction.description.toLowerCase().includes(keyword)) {
                return false;
            }

            // 2. 日期范围筛选
            if (startDate && transactionDate < startDate) {
                return false;
            }
            if (endDate && transactionDate > endDate) {
                return false;
            }

            // 3. 类别筛选 (多选)
            if (selectedCategories.length > 0 && !selectedCategories.includes(transaction.category)) {
                return false;
            }

            // 4. 成员筛选
            if (selectedMember && transaction.member !== selectedMember) {
                return false;
            }

            // 5. 金额范围筛选
            if (!isNaN(minAmount) && transaction.amount < minAmount) {
                return false;
            }
            if (!isNaN(maxAmount) && transaction.amount > maxAmount) {
                return false;
            }

            return true; // 所有条件都满足
        });

        return filtered;
    }


    /**
     * 渲染交易记录到列表中 (现在渲染的是过滤后的交易)
     */
    function renderTransactions() {
        const filteredTransactions = filterTransactions(); // 获取过滤后的交易

        transactionList.innerHTML = ''; // 清空现有列表
        if (filteredTransactions.length === 0) {
            const noRecordsItem = document.createElement('li');
            noRecordsItem.classList.add('transaction-item');
            noRecordsItem.style.justifyContent = 'center';
            noRecordsItem.style.color = '#777';
            noRecordsItem.textContent = '暂无符合条件的交易记录。';
            transactionList.appendChild(noRecordsItem);
            return;
        }

        filteredTransactions.forEach((transaction, index) => {
            const listItem = document.createElement('li');
            listItem.classList.add('transaction-item', transaction.type); // 添加收支类型类
            
            // 格式化日期
            const displayDate = new Date(transaction.date).toLocaleDateString('zh-CN', {
                year: 'numeric',
                month: '2-digit',
                day: '2-digit'
            });

            // 构建显示文本
            let descriptionText = transaction.description ? ` (${transaction.description})` : '';
            const sign = transaction.type === 'income' ? '+' : '-';
            
            listItem.innerHTML = `
                <span>
                    ${transaction.category} - ${transaction.member} - ${displayDate}${descriptionText}
                </span>
                <span class="amount">${sign} ¥ ${transaction.amount.toFixed(2)}</span>
                <button class="delete-btn" data-id="${transaction.id}">X</button>
            `;
            transactionList.appendChild(listItem);
        });
        // 确保新添加的删除按钮也能绑定事件
        addDeleteEventListeners();
    }

    /**
     * 为所有删除按钮添加事件监听器
     */
    function addDeleteEventListeners() {
        // 注意：每次重新渲染时会移除旧的事件监听，这里需要重新绑定
        document.querySelectorAll('.delete-btn').forEach(button => {
            button.onclick = (event) => {
                const transactionId = parseInt(event.target.dataset.id); // 使用ID来删除
                deleteTransaction(transactionId);
            };
        });
    }

    /**
     * 删除交易
     * @param {number} id 要删除交易的ID
     */
    function deleteTransaction(id) {
        if (confirm('确定要删除这笔交易吗？')) {
            transactions = transactions.filter(transaction => transaction.id !== id); // 通过ID过滤掉要删除的交易
            updateSummary();
            renderTransactions(); // 重新渲染列表（会自动应用筛选）
        }
    }


    // --- 事件监听器 ---

    // 提交表单
    transactionForm.addEventListener('submit', (event) => {
        event.preventDefault(); // 阻止表单默认提交行为

        if (validateForm()) {
            const newTransaction = {
                id: Date.now(), // 简单生成一个唯一ID
                type: typeInput.value,
                amount: parseFloat(amountInput.value),
                category: categoryInput.value,
                date: dateInput.value, // 存储为 'YYYY-MM-DD' 格式
                member: memberInput.value,
                description: descriptionInput.value
            };

            transactions.push(newTransaction);
            console.log('新交易已添加:', newTransaction);

            updateSummary(); // 更新总览
            renderTransactions(); // 重新渲染交易列表 (会自动应用筛选)

            // 清空表单
            transactionForm.reset();
            categoryInput.value = ''; // 确保类别选回默认值
            memberInput.value = ''; // 确保成员选回默认值
            hideError(amountError);
            hideError(categoryError);
            hideError(dateError);
            hideError(memberError);

        } else {
            console.log('表单验证失败，请检查输入。');
        }
    });

    // 添加新成员按钮点击事件
    addMemberBtn.addEventListener('click', () => {
        const newMemberName = prompt('请输入新成员的名字:');
        if (newMemberName && newMemberName.trim() !== '') {
            const trimmedName = newMemberName.trim();
            if (!members.includes(trimmedName)) {
                members.push(trimmedName);
                renderMembers(); // 重新渲染成员列表 (包括添加表单和筛选表单)
                memberInput.value = trimmedName; // 选中新添加的成员
                hideError(memberError); // 清除成员错误提示
            } else {
                alert('该成员已存在！');
            }
        }
    });
```

**`script.js` 的主要改动点：**

1.  **新增DOM元素获取：** 获取了所有用于筛选和搜索的 `input`、`select` 和 `button` 元素。
2.  **`categories` 数组：** 定义了一个 `categories` 数组，用于方便地填充表单和筛选器的类别选项。
3.  **`renderMembers()` 改进：** 现在它会同时渲染“添加新交易”表单的成员下拉列表和“筛选成员”的下拉列表。
4.  **`renderFilterCategories()`：** 填充“筛选类别”的多选下拉列表。
5.  **`filterTransactions()` 函数 (核心逻辑)：**
    *   接收所有筛选条件作为输入（关键词、日期范围、类别、成员、金额范围）。
    *   使用 `Array.prototype.filter()` 方法，遍历所有 `transactions`。
    *   每个 `if` 条件都检查一类筛选规则，如果交易不符合该规则，就返回 `false` (即过滤掉这笔交易)。
    *   **关键词：** 检查 `description` 字段是否包含关键词（不区分大小写）。
    *   **日期范围：** 将交易日期和筛选日期都转换为 `Date` 对象进行比较。注意处理日期边界（开始日期从当天0点开始，结束日期到当天23:59:59）。
    *   **类别：** 如果有选中的类别，则检查交易的类别是否包含在 `selectedCategories` 数组中。
    *   **成员：** 如果选择了成员，则检查交易的 `member` 是否与 `selectedMember` 匹配。
    *   **金额范围：** 检查交易金额是否在最小/最大金额之间。
    *   最后返回通过所有筛选条件的交易数组。
6.  **`renderTransactions()` 改进：**
    *   现在不再直接渲染 `transactions` 数组，而是先调用 `filterTransactions()` 获取过滤后的结果。
    *   如果过滤结果为空，会显示“暂无符合条件的交易记录。”的提示。
    *   **删除按钮的 `data-id`：** 将删除按钮的 `data-index` 改为 `data-id="${transaction.id}"`。因为筛选后，数组索引会变动，使用唯一的 `id` 更可靠地定位要删除的交易。
7.  **`deleteTransaction()` 改进：** 现在通过 `id` 而不是索引来删除交易，确保即使在筛选状态下也能正确删除。
8.  **筛选和搜索的事件监听器：**
    *   `applyFiltersBtn` 的点击事件直接调用 `renderTransactions()`，因为 `renderTransactions()` 内部会调用 `filterTransactions()`。
    *   `clearFiltersBtn` 的点击事件会清空所有筛选输入框的值，然后重新调用 `renderTransactions()`。
    *   （可选）注释掉的部分展示了如何实现“实时筛选”——即用户每次输入或选择变化时就立即更新列表。如果记录量很大，这可能会有性能问题，但对于家庭记账通常是可行的。目前我先保留了“应用筛选”按钮的方式，更明确。
9.  **初始测试数据：** 在 `init()` 之后添加了一些示例交易数据，这样你在首次打开页面时就能看到一些记录，方便测试。

### 如何查看和测试

1.  **保存所有文件。**
2.  **在浏览器中打开 `index.html`。**
3.  **你应该能看到一些初始的交易记录。**
4.  **测试关键词搜索：** 在“关键词搜索”框中输入“超市”或“工资”，然后点击“应用筛选”，看结果是否正确。
5.  **测试日期范围：** 选择一个开始日期和结束日期，然后点击“应用筛选”。
6.  **测试类别多选：** 按住 `Ctrl` (Windows) / `Cmd` (macOS) 键，在“类别”下拉框中选择多个类别，然后点击“应用筛选”。
7.  **测试成员筛选：** 选择一个成员，点击“应用筛选”。
8.  **测试金额范围：** 输入最小金额（如100），或最大金额（如500），或两者都输入，然后点击“应用筛选”。
9.  **组合筛选：** 尝试同时输入关键词，选择日期范围，选择类别，选择成员和金额范围，点击“应用筛选”，观察结果是否符合所有条件。
10. **清除筛选：** 点击“清除所有筛选”按钮，所有条件都应该被清空，并且显示所有交易记录。
11. **删除交易：** 尝试删除一条记录，看看它是否从列表中消失，并且总览数据是否更新。
