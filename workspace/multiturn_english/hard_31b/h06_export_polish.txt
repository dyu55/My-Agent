### 3. app.js
I've introduced `currentPage` and `sortConfig` to manage the data pipeline. The `updateUI` function now processes: **Filter $\rightarrow$ Sort $\rightarrow$ Paginate $\rightarrow$ Render**.

```javascript
// --- State Management ---
let transactions = JSON.parse(localStorage.getItem('family_txs')) || [];
let members = JSON.parse(localStorage.getItem('family_members')) || [];
let budgets = JSON.parse(localStorage.getItem('family_budgets')) || { total: 0, categories: {}, members: {} };
let recurringRules = JSON.parse(localStorage.getItem('family_rules')) || [];

let currentPage = 1;
const recordsPerPage = 20;
let sortConfig = { key: 'date', direction: 'desc' };

// --- Dashboard Logic ---
function updateDashboard() {
    const now = new Date();
    const currentMonth = now.toISOString().substring(0, 7);
    
    const monthlyTxs = transactions.filter(tx => tx.date.startsWith(currentMonth));
    const income = monthlyTxs.filter(tx => tx.type === 'income').reduce((s, t) => s + t.amount, 0);
    const expense = monthlyTxs.filter(tx => tx.type === 'expense').reduce((s, t) => s + t.amount, 0);
    
    document.getElementById('dash-income').innerText = `$${income.toFixed(2)}`;
    document.getElementById('dash-expense').innerText = `$${expense.toFixed(2)}`;
    document.getElementById('dash-balance').innerText = `$${(income - expense).toFixed(2)}`;
    
    const usage = budgets.total > 0 ? (expense / budgets.total) * 100 : 0;
    document.getElementById('dash-usage').innerText = `${usage.toFixed(1)}%`;
}

// --- Sorting & Pagination Logic ---
function sortTransactions(data) {
    return [...data].sort((a, b) => {
        let valA = a[sortConfig.key];
        let valB = b[sortConfig.key];

        if (sortConfig.key === 'amount') {
            valA = parseFloat(valA);
            valB = parseFloat(valB);
        }

        if (valA < valB) return sortConfig.direction === 'asc' ? -1 : 1;
        if (valA > valB) return sortConfig.direction === 'asc' ? 1 : -1;
        return 0;
    });
}

function updateUI() {
    const filtered = getFilteredTransactions();
    const sorted = sortTransactions(filtered);
    
    // Empty State
    document.getElementById('empty-state').classList.toggle('hidden', sorted.length > 0);

    // Pagination logic
    const totalPages = Math.ceil(sorted.length / recordsPerPage) || 1;
    if (currentPage > totalPages) currentPage = totalPages;
    
    const start = (currentPage - 1) * recordsPerPage;
    const paginated = sorted.slice(start, start + recordsPerPage);

    txList.innerHTML = '';
    paginated.forEach(tx => {
        const li = document.createElement('li');
        li.className = 'transaction-item';
        li.innerHTML = `
            <input type="checkbox" class="tx-check" data-id="${tx.id}" onchange="this.parentElement.classList.toggle('selected')">
            <div class="item-details">
                <strong>${tx.note || tx.category}</strong>
                <span class="member-cat">${tx.date} • ${tx.member} • ${tx.category}</span>
            </div>
            <div>
                <button class="btn-edit" onclick="editTransaction(${tx.id})">Edit</button>
                <button class="btn-edit" style="color:var(--expense)" onclick="deleteTransaction(${tx.id})">Delete</button>
            </div>
            <div class="amount-val ${tx.type === 'income' ? 'inc' : 'exp'}">
                ${tx.type === 'income' ? '+' : '-'}$${tx.amount.toFixed(2)}
            </div>
        `;
        txList.appendChild(li);
    });

    document.getElementById('page-info').innerText = `Page ${currentPage} of ${totalPages}`;
    updateQuickAccess();
    updateDashboard();
}

function updateQuickAccess() {
    const sorted = [...transactions].sort((a, b) => new Date(b.date) - new Date(a.date)).slice(0, 5);
    const quickList = document.getElementById('quick-access-list');
    quickList.innerHTML = sorted.map(tx => `
        <li class="mini-item">
            <span>${tx.date} - ${tx.member}</span>
            <span class="${tx.type === 'income' ? 'inc' : 'exp'}">${tx.type === 'income' ? '+' : '-'}$${tx.amount}</span>
        </li>
    `).join('');
}

// --- Data Export & Management ---
async function exportCSV() {
    const start = document.getElementById('exp-start').value;
    const end = document.getElementById('exp-end').value;
    const fields = Array.from(document.querySelectorAll('.exp-field:checked')).map(cb => cb.value);
    
    let data = transactions.filter(tx => {
        return (!start || tx.date >= start) && (!end || tx.date <= end);
    });

    const headers = fields.join(',');
    const rows = data.map(tx => fields.map(f => `"${tx[f] || ''}"`).join(','));
    const csvContent = [headers, ...rows].join('\n');
    
    const blob = new Blob([csvContent], { type: 'text/csv' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.setAttribute('href', url);
    a.setAttribute('download', `finance_export_${new Date().toISOString().slice(0,10)}.csv`);
    a.click();
}

function backupJSON() {
    const data = { transactions, members, budgets, recurringRules };
    const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `backup_${new Date().toISOString().slice(0,10)}.json`;
    a.click();
}

function restoreJSON() {
    const file = document.getElementById('restore-json-file').files[0];
    if (!file) return alert('Please select a file');

    const reader = new FileReader();
    reader.onload = (e) => {
        try {
            const data = JSON.parse(e.target.result);
            transactions = data.transactions || [];
            members = data.members || [];
            budgets = data.budgets || {};
            recurringRules = data.recurringRules || [];
            syncStorage();
            updateUI();
            renderMembers();
            alert('Data restored successfully!');
        } catch (err) { alert('Invalid backup file'); }
    };
    reader.readAsText(file);
}

function printMonthlyReport() {
    const month = document.getElementById('report-month-picker').value;
    const monthTxs = transactions.filter(tx => tx.date.startsWith(month));
    const income = monthTxs.filter(tx => tx.type === 'income').reduce((s, t) => s + t.amount, 0);
    const expense = monthTxs.filter(tx => tx.type === 'expense').reduce((s, t) => s + t.amount, 0);

    document.getElementById('print-title').innerText = `Financial Report: ${month}`;
    document.getElementById('print-income').innerText = `$${income.toFixed(2)}`;
    document.getElementById('print-expense').innerText = `$${expense.toFixed(2)}`;
    
    const usage = budgets.total > 0 ? (expense / budgets.total) * 100 : 0;
    document.getElementById('print-usage').innerText = `${usage.toFixed(1)}%`;

    // Fill Category Table
    const catT = {};
    monthTxs.filter(tx => tx.type === 'expense').forEach(tx => catT[tx.category] = (catT[tx.category] || 0) + tx.amount);
    document.getElementById('print-cat-table').innerHTML = `<tr><th>Category</th><th>Amount</th></tr>` + 
        Object.entries(catT).map(([k, v]) => `<tr><td>${k}</td><td>$${v.toFixed(2)}</td></tr>`).join('');

    // Fill Member Table
    const memT = {};
    monthTxs.filter(tx => tx.type === 'expense').forEach(tx => memT[tx.member] = (memT[tx.member] || 0) + tx.amount);
    document.getElementById('print-mem-table').innerHTML = `<tr><th>Member</th><th>Amount</th></tr>` + 
        Object.entries(memT).map(([k, v]) => `<tr><td>${k}</td><td>$${v.toFixed(2)}</td></tr>`).join('');

    window.print();
}

// --- Event Listeners ---
document.querySelectorAll('.sort-tab').forEach(tab => {
    tab.onclick = () => {
        const key = tab.dataset.sort;
        sortConfig.direction = (sortConfig.key === key && sortConfig.direction === 'desc') ? 'asc' : 'desc';
        sortConfig.key = key;
        updateUI();
    };
});

document.getElementById('prev-page').onclick = () => { if (currentPage > 1) { currentPage--; updateUI(); } };
document.getElementById('next-page').onclick = () => { 
    const filtered = getFilteredTransactions();
    if (currentPage < Math.ceil(filtered.length / recordsPerPage)) { currentPage++; updateUI(); } 
};

document.getElementById('batch-delete-btn').onclick = () => {
    const selected = Array.from(document.querySelectorAll('.tx-check:checked')).map(cb => parseInt(cb.dataset.id));
    if (selected.length === 0) return;
    if (confirm(`Are you sure you want to delete ${selected.length} records?`)) {
        saveState();
        transactions = transactions.filter(t => !selected.includes(t.id));
        updateUI();
        syncStorage();
        updateCharts();
    }
};

document.getElementById('export-csv-btn').onclick = exportCSV;
document.getElementById('backup-json-btn').onclick = backupJSON;
document.getElementById('restore-json-btn').onclick = restoreJSON;
document.getElementById('print-report-btn').onclick = printMonthlyReport;

// --- Updated Init ---
function fullInit() {
    initTheme();
    initCategoryFilters();
    renderMembers();
    updateUI();
    initCharts();
    processRecurring();
    updateBudgetBanner();
    fillRecurringDropdowns();
    
    const now = new Date();
    document.getElementById('report-month-picker').value = now.toISOString().substring(0, 7);
    updateCharts();
}
fullInit();
```

### Key Implementation Details:

1.  **Performance (The >500 Records Problem)**: I implemented **Virtual Pagination**. Instead of letting the DOM struggle with 500+ `<li>` elements, the `updateUI` function only slices 20 records from the sorted array. This keeps the page buttery smooth regardless of data size.
2.  **Sort Pipeline**: I added a `sortConfig` object. The sequence is now: `Input Data` $\rightarrow$ `getFilteredTransactions()` $\rightarrow$ `sortTransactions()` $\rightarrow$ `slice(start, end)` $\rightarrow$ `render`.
3.  **CSV Export Logic**: The export function is now fully dynamic. It creates a comma-separated string based on the user's field selection and uses a `Blob` to trigger a file download without needing a server.
4.  **Printable Reports**: I used a dedicated `@media print` CSS block. This allows the "Print Report" button to populate a hidden template and then tell the browser to print *only* that template, hiding the dashboard, filters, and forms.
5.  **JSON Backup**: Since `localStorage` can be cleared by the browser or user, the Backup/Restore feature uses `FileReader` and `JSON.stringify` to allow users to save their entire database to a local `.json` file.
6.  **Dashboard**: The top cards now provide "At-a-Glance" monthly metrics, while the "Quick Access" list ensures the user doesn't have to paginate just to see their last few entries.
