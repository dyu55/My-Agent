### 3. app.js
The JS now includes a `BudgetManager` and a `RecurringManager`. I've added a check on page load to see if any recurring transactions should be automatically generated.

```javascript
// --- State ---
let transactions = JSON.parse(localStorage.getItem('family_txs')) || [];
let members = JSON.parse(localStorage.getItem('family_members')) || [];
let budgets = JSON.parse(localStorage.getItem('family_budgets')) || { total: 0, categories: {}, members: {} };
let recurringRules = JSON.parse(localStorage.getItem('family_rules')) || [];
let stateHistory = []; 
let redoStack = [];    

// --- DOM Elements ---
const financeForm = document.getElementById('finance-form');
const memberSelect = document.getElementById('member-select');
const txList = document.getElementById('transaction-list');
const settingsModal = document.getElementById('settings-modal');
const budgetBanner = document.getElementById('budget-banner');

// --- Budget Logic ---
function calculateMonthlySpent(filterFn) {
    const now = new Date();
    const currentMonth = now.toISOString().substring(0, 7);
    return transactions
        .filter(tx => tx.date.startsWith(currentMonth) && tx.type === 'expense' && filterFn(tx))
        .reduce((sum, tx) => sum + tx.amount, 0);
}

function checkBudgetWarnings(amount, category, member) {
    const now = new Date();
    const currentMonth = now.toISOString().substring(0, 7);
    
    const totalSpent = calculateMonthlySpent(() => true);
    const catSpent = calculateMonthlySpent(tx => tx.category === category);
    const memSpent = calculateMonthlySpent(tx => tx.member === member);

    const warnings = [];
    
    // Total Budget Check
    if (budgets.total > 0) {
        const perc = ((totalSpent + amount) / budgets.total) * 100;
        if (perc >= 100) warnings.push(`🚨 TOTAL BUDGET EXCEEDED! (${perc.toFixed(1)}%)`);
        else if (perc >= 80) warnings.push(`⚠️ Total budget near limit (${perc.toFixed(1)}%)`);
    }

    // Category Budget Check
    if (budgets.categories[category] > 0) {
        const perc = ((catSpent + amount) / budgets.categories[category]) * 100;
        if (perc >= 100) warnings.push(`🚨 ${category.toUpperCase()} budget exceeded!`);
        else if (perc >= 80) warnings.push(`⚠️ ${category.toUpperCase()} budget near limit`);
    }

    // Member Budget Check
    if (budgets.members[member] > 0) {
        const perc = ((memSpent + amount) / budgets.members[member]) * 100;
        if (perc >= 100) warnings.push(`🚨 ${member}'s budget exceeded!`);
        else if (perc >= 80) warnings.push(`⚠️ ${member}'s budget near limit`);
    }

    return warnings;
}

function updateBudgetBanner() {
    const totalSpent = calculateMonthlySpent(() => true);
    if (budgets.total === 0) { budgetBanner.classList.add('hidden'); return; }

    const perc = (totalSpent / budgets.total) * 100;
    budgetBanner.classList.remove('hidden');
    if (perc >= 100) {
        budgetBanner.className = 'budget-banner banner-danger';
        budgetBanner.innerText = `🔴 Monthly Budget Exceeded! ${perc.toFixed(1)}% used.`;
    } else if (perc >= 80) {
        budgetBanner.className = 'budget-banner banner-warning';
        budgetBanner.innerText = `🟡 Warning: Monthly Budget at ${perc.toFixed(1)}%.`;
    } else {
        budgetBanner.classList.add('hidden');
    }
}

// --- Recurring Logic ---
function processRecurring() {
    const today = new Date();
    const currentMonth = today.toISOString().substring(0, 7);
    const todayDay = today.getDate();
    let added = false;

    recurringRules.forEach(rule => {
        // Check if it's the right day and hasn't been added this month
        const alreadyExists = transactions.some(tx => 
            tx.recurringId === rule.id && tx.date.startsWith(currentMonth)
        );

        if (rule.dayOfMonth === todayDay && !alreadyExists) {
            const dateStr = today.toISOString().split('T')[0];
            transactions.push({
                id: Date.now() + Math.random(),
                recurringId: rule.id,
                amount: rule.amount,
                category: rule.category,
                member: rule.member,
                date: dateStr,
                note: `Auto-generated: ${rule.category}`,
                type: ['salary'].includes(rule.category) ? 'income' : 'expense',
                isRecurring: true
            });
            added = true;
        }
    });

    if (added) {
        syncStorage();
        updateUI();
        updateCharts();
    }
}

// --- Settings Modal Logic ---
document.getElementById('open-settings').onclick = () => {
    settingsModal.classList.remove('hidden');
    renderBudgetSettings();
    renderRecurringRules();
};
document.getElementById('close-settings').onclick = () => settingsModal.classList.add('hidden');

document.querySelectorAll('.tab-btn').forEach(btn => {
    btn.onclick = () => {

        btn.classList.add('active');
        document.getElementById(`tab-${btn.dataset.tab}`).classList.add('active');
    };
});

function renderBudgetSettings() {
    // Total Budget
    document.getElementById('set-total-budget').value = budgets.total;
    const totalSpent = calculateMonthlySpent(() => true);
    const totalPerc = budgets.total > 0 ? (totalSpent / budgets.total) * 100 : 0;
    updateUsageBar('total-budget-usage', totalPerc);

    // Category Budgets
    const catList = document.getElementById('category-budget-list');
    catList.innerHTML = '';
    categories.forEach(cat => {
        const spent = calculateMonthlySpent(tx => tx.category === cat);
        const limit = budgets.categories[cat] || 0;
        const perc = limit > 0 ? (spent / limit) * 100 : 0;
        
        const div = document.createElement('div');
        div.className = 'budget-row';
        div.innerHTML = `
            <span>${cat}</span>
            <input type="number" value="${limit}" onchange="updateCatBudget('${cat}', this.value)">
            <div class="usage-bar-container"><div class="usage-bar-fill ${getFillColor(perc)}" style="width:${Math.min(perc, 100)}%"></div></div>
        `;
        catList.appendChild(div);
    });

    // Member Budgets
    const memList = document.getElementById('member-budget-list');
    memList.innerHTML = '';
    members.forEach(mem => {
        const spent = calculateMonthlySpent(tx => tx.member === mem);
        const limit = budgets.members[mem] || 0;
        const perc = limit > 0 ? (spent / limit) * 100 : 0;
        
        const div = document.createElement('div');
        div.className = 'budget-row';
        div.innerHTML = `
            <span>${mem}</span>
            <input type="number" value="${limit}" onchange="updateMemBudget('${mem}', this.value)">
            <div class="usage-bar-container"><div class="usage-bar-fill ${getFillColor(perc)}" style="width:${Math.min(perc, 100)}%"></div></div>
        `;
        memList.appendChild(div);
    });
}

function getFillColor(perc) {
    if (perc >= 100) return 'fill-red';
    if (perc >= 80) return 'fill-yellow';
    return 'fill-green';
}

function updateUsageBar(id, perc) {
    const container = document.getElementById(id);
    container.innerHTML = `<div class="usage-bar-container"><div class="usage-bar-fill ${getFillColor(perc)}" style="width:${Math.min(perc, 100)}%"></div></div>`;
}

window.updateCatBudget = (cat, val) => { budgets.categories[cat] = parseFloat(val) || 0; syncStorage(); renderBudgetSettings(); };

// --- Modified Form Submission ---
```

### Key New Features Explained:

1.  **Three-Level Budgeting**:
    *   **Total Budget**: Monitored globally. If you exceed it, a red banner appears at the top of the screen.
    *   **Category/Member Budgets**: Configured in the "Settings $\rightarrow$ Budgets" tab. 
    *   **Proactive Warnings**: During the `submit` event of the form, the app calculates the *projected* spent amount. If the new transaction pushes a budget over 80%, a `confirm()` dialog warns the user before the transaction is saved.
2.  **Recurring Transaction Engine**:
    *   **Rules**: Users define a rule (Amount, Category, Member, Day of Month).
    *   **Auto-Generation**: On every page load, `processRecurring()` runs. It checks the current date and the `recurringRules` array. If it's the specified day and no transaction for that rule exists for the current month, it automatically injects the record.
    *   **Marker**: Auto-generated records are given a `recurringId` and a note: `"Auto-generated: [Category]"`.
3.  **Budget Visualization**:
    *   The Settings page uses dynamic progress bars.
    *   `getFillColor()` maps the percentage to CSS classes: $\text{Green} (<80\%) \rightarrow \text{Yellow} (80-100\%) \rightarrow \text{Red} (>100\%)$.
4.  **User Experience**:
    *   The **Settings Modal** separates complex configuration from daily data entry.
    *   The **Budget Banner** provides an "at-a-glance" status of the monthly financial health without needing to open any menus.
