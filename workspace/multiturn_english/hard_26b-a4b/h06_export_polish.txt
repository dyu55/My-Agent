### 3. app.js
The core complexity is now in the `DataManager` and `Exporter` logic.

```javascript
/**
 * GLOBAL STATE
 **/
let state = {
    transactions: JSON.parse(localStorage.getItem('transactions')) || [],
    members: JSON.parse(localStorage.getItem('members')) || ['Admin'],
    budgets: JSON.parse(localStorage.getItem('budgets')) || { total: 5000, categories: {}, members: {} },
    rules: JSON.parse(localStorage.getItem('rules')) || [],
    
    // View State
    currentPage: 1,
    itemsPerPage: 20,
    sortBy: 'date',
    sortDir: 'desc', // 'asc' or 'desc'
    currentMonth: new Date().toISOString().slice(0, 7),
    
    // Filters
    filters: { category: [], member: '', search: '', start: '', end: '', min: null, max: null }
};

/**
 * DATA MANAGEMENT & SORTING
 **/
function getProcessedTransactions() {
    let list = [...state.transactions];

    // 1. Apply Filters
    list = list.filter(t => {
        const dateMatch = (!state.filters.start || t.date >= state.filters.start) &&
                          (!state.filters.end || t.date <= state.filters.end);
        const catMatch = state.filters.category.length === 0 || state.filters.category.includes(t.category);
        const memMatch = !state.filters.member || t.member === state.filters.member;
        const searchMatch = !state.filters.search || t.note?.toLowerCase().includes(state.filters.search.toLowerCase());
        const amtMatch = (state.filters.min === null || t.amount >= state.filters.min) &&
                         (state.filters.max === null || t.amount <= state.filters.max);
        return dateMatch && catMatch && memMatch && searchMatch && amtMatch;
    });

    // 2. Apply Sorting
    list.sort((a, b) => {
        let valA = a[state.sortBy];
        let valB = b[state.sortBy];
        
        if (state.sortBy === 'amount') {
            return state.sortDir === 'asc' ? valA - valB : valB - valA;
        }
        // String comparison for category/member/date
        return state.sortDir === 'asc' 
            ? String(valA).localeCompare(String(valB))
            : String(valB).localeCompare(String(valA));
    });

    return list;
}

function setSort(field) {
    if (state.sortBy === field) {
        state.sortDir = state.sortDir === 'asc' ? 'desc' : 'asc';
    } else {
        state.sortBy = field;
        state.sortDir = 'desc';
    }
    updateSortIcons();
    render();
}

/**
 * EXPORT & BACKUP
 **/
function exportToCSV() {
    const data = getProcessedTransactions();
    const selectedFields = Array.from(document.querySelectorAll('.csv-field:checked')).map(cb => cb.value);
    
    if (selectedFields.length === 0) return alert("Select at least one field");

    const header = selectedFields.join(',');
    const rows = data.map(t => selectedFields.map(field => `"${t[field] || ''}"`).join(','));
    const csvContent = "data:text/csv;charset=utf-8," + [header, ...rows].join("\n");

    const link = document.createElement("a");
    link.setAttribute("href", encodeURI(csvContent));
    link.setAttribute("download", `export_${state.currentMonth}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
}

function backupJSON() {
    const dataStr = JSON.stringify(state);
    const dataUri = 'data:application/json;charset=utf-8,' + encodeURIComponent(dataStr);
    const date = new Date().toISOString().split('T')[0];
    
    const link = document.createElement("a");
    link.setAttribute("href", dataUri);
    link.setAttribute("download", `finance_backup_${date}.json`);
    link.click();
}

function restoreJSON(event) {
    const file = event.target.files[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = (e) => {
        try {
            const imported = JSON.parse(e.target.result);
            if (confirm("This will overwrite your current data. Continue?")) {
                state = { ...state, ...imported };
                saveToLocalStorage();
                location.reload();
            }
        } catch (err) {
            alert("Invalid JSON file.");
        }
    };
    reader.readAsText(file);
}

/**
 * PRINTING ENGINE
 **/
function generatePrintReport() {
    const report = document.getElementById('print-report');
    const monthTxs = state.transactions.filter(t => t.date.startsWith(state.currentMonth));
    
    // Summary
    const income = monthTxs.filter(t => t.amount > 0).reduce((a, b) => a + b.amount, 0);
    const expense = monthTxs.filter(t => t.amount < 0).reduce((a, b) => a + Math.abs(b.amount), 0);

    document.getElementById('report-title').textContent = `Financial Report: ${state.currentMonth}`;
    document.getElementById('report-summary').innerHTML = `
        <p>Total Income: $${income.toFixed(2)}</p>
        <p>Total Expense: $${expense.toFixed(2)}</p>
        <p>Net Cashflow: $${(income - expense).toFixed(2)}</p>
    `;

    // Logic for Category/Member breakdown... (omitted for brevity, similar to Charting logic)

    window.print();
}

/**
 * MAIN RENDER LOOP
 **/
function render() {
    const allProcessed = getProcessedTransactions();
    
    // 1. Render Dashboard Cards
    const monthTxs = state.transactions.filter(t => t.date.startsWith(state.currentMonth));
    const inc = monthTxs.filter(t => t.amount > 0).reduce((a, b) => a + b.amount, 0);
    const exp = monthTxs.filter(t => t.amount < 0).reduce((a, b) => a + Math.abs(b.amount), 0);
    document.getElementById('stat-income').textContent = `$${inc.toFixed(0)}`;
    document.getElementById('stat-expense').textContent = `$${exp.toFixed(0)}`;
    document.getElementById('stat-balance').textContent = `$${(inc - exp).toFixed(0)}`;
    document.getElementById('stat-budget').textContent = `${Math.round((exp / state.budgets.total) * 100)}%`;

    // 2. Quick Access (Last 5)
    const recentList = document.getElementById('recent-list');
    recentList.innerHTML = state.transactions.slice(-5).reverse().map(t => `
        <li class="transaction-item ${t.amount >= 0 ? 'item-income' : 'item-expense'}">
            <span>${t.note || t.category}</span>
            <span>$${t.amount}</span>
        </li>
    `).join('');

    // 3. Paginated Transaction List
    const listEl = document.getElementById('transaction-list');
    if (allProcessed.length === 0) {
        listEl.innerHTML = '<div class="empty-state">No transactions found matching your criteria.</div>';
    } else {
        const start = (state.currentPage - 1) * state.itemsPerPage;
        const end = start + state.itemsPerPage;
        const pagedData = allProcessed.slice(start, end);

        listEl.innerHTML = pagedData.map(t => `
            <li class="transaction-item ${t.amount >= 0 ? 'item-income' : 'item-expense'}">
                <input type="checkbox" class="item-selection" data-id="${t.id}">
                <div class="item-info">
                    <div>${t.date} - ${t.category} ${t.isRecurring ? '<span class="recurring-badge">Auto</span>' : ''}</div>
                    <small>${t.member} | ${t.note || ''}</small>
                </div>
                <div class="item-amount">$${t.amount.toFixed(2)}</div>
            </li>
        `).join('');
    }

    // 4. Update Pagination UI
    const totalPages = Math.ceil(allProcessed.length / state.itemsPerPage) || 1;
    document.getElementById('page-info').textContent = `Page ${state.currentPage} of ${totalPages}`;
    document.getElementById('prev-page').disabled = state.currentPage === 1;
    document.getElementById('next-page').disabled = state.currentPage === totalPages;
}

// Helpers
function saveToLocalStorage() {
    localStorage.setItem('transactions', JSON.stringify(state.transactions));
    localStorage.setItem('budgets', JSON.stringify(state.budgets));
    localStorage.setItem('rules', JSON.stringify(state.rules));
}

function updateSortIcons() {
    ['date', 'amount', 'category'].forEach(f => {
        const el = document.getElementById(`sort-${f}`);
        if (state.sortBy === f) {
            el.textContent = state.sortDir === 'asc' ? '↑' : '↓';
        } else {
            el.textContent = '↕';
        }
    });
}

// Event Listeners
document.getElementById('export-csv-btn').addEventListener('click', exportToCSV);
document.getElementById('backup-json-btn').addEventListener('click', backupJSON);
document.getElementById('restore-json').addEventListener('change', restoreJSON);
document.getElementById('print-report-btn').addEventListener('click', generatePrintReport);
document.getElementById('prev-page').addEventListener('click', () => { if(state.currentPage > 1) { state.currentPage--; render(); }});
document.getElementById('next-page').addEventListener('click', () => { state.currentPage++; render(); });

// Init
render();
```

### Summary of complexity handled:
1.  **Pagination & Performance:** The `getProcessedTransactions` function handles the heavy sorting and filtering of the entire array, but `render()` only performs DOM manipulation for the 20 records in the current page. This prevents the browser from freezing even with 5,000+ records.
2.  **Printability:** We use a "Print-Only" container. When the user hits print, the CSS hides the app's buttons, navigation, and forms, and reveals the clean report template.
3.  **Robust Export:** The CSV export allows users to cherry-pick columns (Date, Member, etc.), which is essential for users who want to import their data into Excel/Google Sheets for further work.
4.  **Data Safety:** The JSON Backup/Restore allows users to move their data between computers or perform a "hard reset" easily.
