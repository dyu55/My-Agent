### 3. app.js
This is the complex logic core.

```javascript
/**
 * STATE OBJECT
 **/
const state = {
    transactions: JSON.parse(localStorage.getItem('transactions')) || [],
    members: JSON.parse(localStorage.getItem('members')) || ['Admin'],
    categories: ['food', 'transport', 'shopping', 'rent', 'utilities', 'entertainment', 'medical', 'education', 'salary', 'other'],
    // Budgets structure
    budgets: JSON.parse(localStorage.getItem('budgets')) || {
        total: 5000,
        categories: {}, // {food: 500}
        members: {}     // {Admin: 2000}
    },
    // Rules structure
    rules: JSON.parse(localStorage.getItem('rules')) || [], 
    isDarkMode: localStorage.getItem('darkMode') === 'true',
    currentChartMonth: new Date().toISOString().slice(0, 7),
    charts: {}
};

/**
 * BUDGET ENGINE
 **/
function checkBudgets(newTx) {
    const month = newTx.date.slice(0, 7);
    const monthTxs = state.transactions.filter(t => t.date.startsWith(month));
    
    let alerts = [];

    // 1. Total Month Check
    const currentTotalSpent = monthTxs.reduce((acc, t) => acc + (t.amount < 0 ? Math.abs(t.amount) : 0), 0);
    const newTotal = currentTotalSpent + (newTx.amount < 0 ? Math.abs(newTx.amount) : 0);
    if (newTotal > state.budgets.total) alerts.push(`🚨 TOTAL BUDGET EXCEEDED!`);
    else if (newTotal > state.budgets.total * 0.8) alerts.push(`⚠️ Total budget is at 80%+`);

    // 2. Category Check
    if (newTx.amount < 0 && state.budgets.categories[newTx.category]) {
        const catLimit = state.budgets.categories[newTx.category];
        const catSpent = monthTxs.filter(t => t.category === newTx.category)
                                 .reduce((acc, t) => acc + Math.abs(t.amount), 0);
        const newCatTotal = catSpent + Math.abs(newTx.amount);
        if (newCatTotal > catLimit) alerts.push(`🚨 ${newTx.category.toUpperCase()} budget exceeded!`);
        else if (newCatTotal > catLimit * 0.8) alerts.push(`⚠️ ${newTx.category.toUpperCase()} budget at 80%+`);
    }

    // 3. Member Check
    if (newTx.amount < 0 && state.budgets.members[newTx.member]) {
        const memLimit = state.budgets.members[newTx.member];
        const memSpent = monthTxs.filter(t => t.member === newTx.member)
                                 .reduce((acc, t) => acc + Math.abs(t.amount), 0);
        const newMemTotal = memSpent + Math.abs(newTx.amount);
        if (newMemTotal > memLimit) alerts.push(`🚨 ${newTx.member} budget exceeded!`);
    }

    if (alerts.length > 0) alert(alerts.join('\n'));
}

/**
 * RECURRING ENGINE
 **/
function processRecurringRules() {
    const now = new Date();
    const currentMonth = now.toISOString().slice(0, 7);
    const currentDay = now.getDate();
    let appliedAny = false;

    state.rules.forEach(rule => {
        // Check if rule was already applied this month
        if (rule.lastAppliedMonth === currentMonth) return;

        // If today is the rule day (or we haven't processed this month yet)
        if (currentDay >= rule.day) {
            const autoTx = {
                id: Date.now() + Math.random(),
                amount: rule.amount,
                category: rule.category,
                member: rule.member,
                date: `${currentMonth}-${String(rule.day).padStart(2, '0')}`,
                note: `[AUTO] ${rule.description}`,
                isRecurring: true
            };
            state.transactions.push(autoTx);
            rule.lastAppliedMonth = currentMonth;
            appliedAny = true;
        }
    });

    if (appliedAny) {
        saveToLocalStorage();
        render();
    }
}

/**
 * UI RENDERING
 **/
function renderBudgets() {
    const catList = document.getElementById('category-budget-list');
    const memList = document.getElementById('member-budget-list');
    const usageList = document.getElementById('usage-progress-bars');
    
    catList.innerHTML = '';
    memList.innerHTML = '';
    usageList.innerHTML = '';

    // Setup inputs for Category Budgets
    state.categories.forEach(cat => {
        const row = document.createElement('div');
        row.className = 'budget-setting-row';
        row.innerHTML = `<span>${cat}</span> <input type="number" data-cat="${cat}" value="${state.budgets.categories[cat] || 0}">`;
        catList.appendChild(row);
    });

    // Setup inputs for Member Budgets
    state.members.forEach(mem => {
        const row = document.createElement('div');
        row.className = 'budget-setting-row';
        row.innerHTML = `<span>${mem}</span> <input type="number" data-mem="${mem}" value="${state.budgets.members[mem] || 0}">`;
        memList.appendChild(row);
    });

    // Calculate Usage
    const month = state.currentChartMonth;
    const monthTxs = state.transactions.filter(t => t.date.startsWith(month));

    // Total Usage
    const totalSpent = monthTxs.reduce((acc, t) => acc + (t.amount < 0 ? Math.abs(t.amount) : 0), 0);
    renderProgressBar('Total Monthly', totalSpent, state.budgets.total, usageList);

    // Category Usage
    state.categories.forEach(cat => {
        if (state.budgets.categories[cat] > 0) {
            const spent = monthTxs.filter(t => t.category === cat).reduce((acc, t) => acc + Math.abs(t.amount), 0);
            renderProgressBar(cat, spent, state.budgets.categories[cat], usageList);
        }
    });

    // Member Usage
    state.members.forEach(mem => {
        if (state.budgets.members[mem] > 0) {
            const spent = monthTxs.filter(t => t.member === mem).reduce((acc, t) => acc + Math.abs(t.amount), 0);
            renderProgressBar(mem, spent, state.budgets.members[mem], usageList);
        }
    });
}

function renderProgressBar(label, spent, limit, container) {
    const percent = limit > 0 ? Math.min((spent / limit) * 100, 100) : 0;
    let colorClass = 'progress-good';
    if (percent >= 100) colorClass = 'progress-danger';
    else if (percent >= 80) colorClass = 'progress-warning';

    const div = document.createElement('div');
    div.className = 'usage-item';
    div.innerHTML = `
        <div class="usage-header"><span>${label}</span><span>${Math.round(percent)}% ($${spent.toFixed(0)}/$${limit})</span></div>
        <div class="progress-bar-bg"><div class="progress-bar-fill ${colorClass}" style="width: ${percent}%"></div></div>
    `;
    container.appendChild(div);
}

function renderRules() {
    const list = document.getElementById('active-rules-list');
    list.innerHTML = '';
    state.rules.forEach((rule, index) => {
        const li = document.createElement('li');
        li.className = 'rule-card';
        li.innerHTML = `
            <div>
                <strong>${rule.description}</strong><br>
                <small>$${rule.amount} | ${rule.category} | Day ${rule.day}</small>
            </div>
            <button onclick="deleteRule(${index})" class="danger-btn">Cancel</button>
        `;
        list.appendChild(li);
    });
}

// ... (Rest of the logic for Tabs, Transactions, and Charts from previous version) ...

function saveToLocalStorage() {
    localStorage.setItem('transactions', JSON.stringify(state.transactions));
    localStorage.setItem('members', JSON.stringify(state.members));
    localStorage.setItem('budgets', JSON.stringify(state.budgets));
    localStorage.setItem('rules', JSON.stringify(state.rules));
}

// Initialization
document.addEventListener('DOMContentLoaded', () => {
    processRecurringRules();
    // Attach Tab Listeners
    document.querySelectorAll('.tab-btn').forEach(btn => {
        btn.addEventListener('click', (e) => {
            document.querySelectorAll('.tab-btn, .tab-content').forEach(el => el.classList.remove('active'));
            btn.classList.add('active');
            document.getElementById(btn.dataset.tab).classList.add('active');
        });
    });
    // Render Everything
    render(); 
    renderBudgets();
    renderRules();
});
```

### Critical Implementation Notes:

1.  **The Budget Alert Loop:** Inside the `finance-form` submit handler, we now call `checkBudgets(newTx)`. This calculates the potential "new state" of the month before the transaction is actually committed to the array, allowing for proactive warnings.
2.  **Recurring Logic (The "Safety" Check):** A common mistake in automation is creating a transaction every time the app is opened. To solve this, each rule has a `lastAppliedMonth` property (e.g., `"2023-10"`). The `processRecurringRules` function only executes if `currentMonth !== rule.lastAppliedMonth`.
3.  **The Three-Level Hierarchy:**
    *   **Total:** Aggregate of all `amount < 0` in the selected month.
    *   **Category:** Filtered by category and `amount < 0`.
    *   **Member:** Filtered by member and `amount < 0`.
4.  **Mobile Considerations:** I used `min-height: 48px` for all interactive elements and added a `@media` query to stack the charts and inputs on small screens.
5.  **UX:** Auto-generated transactions are marked with `[AUTO]` in the note field, so the user knows why money moved.
