This is a major architectural jump. To implement **Undo/Redo** properly, we must move away from simple function calls and implement the **Command Pattern**. We will treat every action as an "object" that knows how to do something and how to reverse it.

I have also added a **Filter Engine** that processes the transaction list through multiple layers of logic.

### 3. app.js
This is the engine. It implements the `Command` pattern for undo/redo and a `filter` method that runs every time an input changes.

```javascript
/** 
 * COMMAND PATTERN IMPLEMENTATION 
 **/
class Command {
    execute() {}
    undo() {}
}

class AddCommand extends Command {
    constructor(transaction, state) {
        super();
        this.transaction = transaction;
        this.state = state;
    }
    execute() { this.state.transactions.push(this.transaction); }
    undo() { this.state.transactions = this.state.transactions.filter(t => t.id !== this.transaction.id); }
}

class DeleteCommand extends Command {
    constructor(transactionsToDelete, state) {
        super();
        this.deletedItems = [...transactionsToDelete]; // Store copies
        this.state = state;
    }
    execute() {
        const ids = this.deletedItems.map(t => t.id);
        this.state.transactions = this.state.transactions.filter(t => !ids.includes(t.id));
    }
    undo() {
        this.state.transactions.push(...this.deletedItems);
    }
}

/** 
 * CORE APP LOGIC 
 **/
const state = {
    transactions: JSON.parse(localStorage.getItem('transactions')) || [],
    members: JSON.parse(localStorage.getItem('members')) || ['Admin'],
    categories: ['food', 'transport', 'shopping', 'rent', 'utilities', 'entertainment', 'medical', 'education', 'salary', 'other'],
    undoStack: [],
    redoStack: []
};

// DOM Elements
const transactionList = document.getElementById('transaction-list');
const balanceDisplay = document.getElementById('balance');
const financeForm = document.getElementById('finance-form');

// --- HISTORY MANAGER ---
function executeCommand(command) {
    command.execute();
    state.undoStack.push(command);
    state.redoStack = []; // Clear redo on new action
    saveAndRender();
}

function undo() {
    if (state.undoStack.length === 0) return;
    const command = state.undoStack.pop();
    command.undo();
    state.redoStack.push(command);
    saveAndRender();
}

function redo() {
    if (state.redoStack.length === 0) return;
    const command = state.redoStack.pop();
    command.execute();
    state.undoStack.push(command);
    saveAndRender();
}

function saveAndRender() {
    localStorage.setItem('transactions', JSON.stringify(state.transactions));
    localStorage.setItem('members', JSON.stringify(state.members));
    render();
}

// --- FILTER ENGINE ---
function getFilteredTransactions() {
    const start = document.getElementById('filter-start').value;
    const end = document.getElementById('filter-end').value;
    const member = document.getElementById('filter-member').value;
    const min = parseFloat(document.getElementById('filter-min').value);
    const max = parseFloat(document.getElementById('filter-max').value);
    const search = document.getElementById('filter-search').value.toLowerCase();
    
    // Multi-select categories
    const checkedCats = Array.from(document.querySelectorAll('.cat-checkbox:checked')).map(cb => cb.value);

    return state.transactions.filter(t => {
        const dateMatch = (!start || t.date >= start) && (!end || t.date <= end);
        const memberMatch = !member || t.member === member;
        const amountMatch = (isNaN(min) || t.amount >= min) && (isNaN(max) || t.amount <= max);
        const searchMatch = !search || (t.note && t.note.toLowerCase().includes(search));
        const catMatch = checkedCats.length === 0 || checkedCats.includes(t.category);
        
        return dateMatch && memberMatch && amountMatch && searchMatch && catMatch;
    });
}

// --- RENDERING ---
function render() {
    const filtered = getFilteredTransactions();
    transactionList.innerHTML = '';
    let total = 0;

    filtered.forEach(t => {
        total += t.amount;
        const li = document.createElement('li');
        li.className = `transaction-item ${t.amount >= 0 ? 'item-income' : 'item-expense'}`;
        li.innerHTML = `
            <input type="checkbox" class="item-selection" data-id="${t.id}">
            <div class="item-info">
                <div>${t.category.toUpperCase()} - ${t.date}</div>
                <small>${t.member} ${t.note ? `| Note: ${t.note}` : ''}</small>
            </div>
            <div class="item-amount">$${t.amount.toFixed(2)}</div>
        `;
        transactionList.appendChild(li);
    });

    balanceDisplay.textContent = `$${total.toFixed(2)}`;
    balanceDisplay.style.color = total >= 0 ? '#2ecc71' : '#e74c3c';
}

function renderSetup() {
    // Render category checkboxes in filters
    const catContainer = document.getElementById('filter-categories');
    catContainer.innerHTML = state.categories.map(cat => `
        <label class="checkbox-item">
            <input type="checkbox" class="cat-checkbox" value="${cat}"> ${cat}
        </label>
    `).join('');

    // Render member dropdowns
    const updateMemberDropdowns = () => {
        const options = state.members.map(m => `<option value="${m}">${m}</option>`).join('');
        document.getElementById('member').innerHTML = '<option value="">-- Select --</option>' + options;
        document.getElementById('filter-member').innerHTML = '<option value="">All Members</option>' + options;
    };
    updateMemberDropdowns();
}

// --- EVENTS ---
financeForm.addEventListener('submit', (e) => {
    e.preventDefault();
    
    // Basic Validation
    const amount = parseFloat(document.getElementById('amount').value);
    const category = document.getElementById('category').value;
    const date = document.getElementById('date').value;
    const member = document.getElementById('member').value;
    const note = document.getElementById('note').value;

    if (isNaN(amount) || amount <= 0) return alert("Enter positive amount");
    if (!category) return alert("Select category");
    if (!date) return alert("Select date");
    if (!member) return alert("Select member");

    // Check future date
    if (new Date(date) > new Date()) return alert("Date cannot be in future");

    const transaction = { id: Date.now(), amount, category, date, member, note };
    executeCommand(new AddCommand(transaction, state));
    financeForm.reset();
});

// Batch Delete
document.getElementById('batch-delete-btn').addEventListener('click', () => {
    const selectedIds = Array.from(document.querySelectorAll('.item-selection:checked'))
                             .map(cb => parseFloat(cb.dataset.id));
    
    if (selectedIds.length === 0) return;

    const toDelete = state.transactions.filter(t => selectedIds.includes(t.id));
    executeCommand(new DeleteCommand(toDelete, state));
});

// Filter Listeners (Update on every input)
document.querySelectorAll('.filters-container input, .filters-container select').forEach(el => {
    el.addEventListener('input', render);
});

document.getElementById('clear-filters').addEventListener('click', () => {
    document.querySelectorAll('.filters-container input').forEach(i => i.value = '');
    document.querySelectorAll('.cat-checkbox').forEach(c => c.checked = false);
    document.getElementById('filter-member').value = '';
    render();
});

// Undo/Redo Listeners
document.getElementById('undo-btn').addEventListener('click', undo);
document.getElementById('redo-btn').addEventListener('click', redo);

// Keyboard Shortcuts
window.addEventListener('keydown', (e) => {
    if (e.ctrlKey && e.key === 'z') { e.preventDefault(); undo(); }
    if (e.ctrlKey && e.key === 'y') { e.preventDefault(); redo(); }
});

// Initialize
renderSetup();
render();
```

### Key Features Explained:

1.  **Undo/Redo (Command Pattern):**
    *   I created `AddCommand` and `DeleteCommand` classes.
    *   When you add a transaction, an `AddCommand` is pushed to the `undoStack`.
    *   When you click "Undo", the app calls the `undo()` method of that specific object, which knows exactly how to remove the item that was just added.
    *   **Batch Delete** works by capturing all currently selected items in a single `DeleteCommand`, allowing you to restore the entire group with one "Undo".

2.  **Complex Filtering:**
    *   The `getFilteredTransactions` function acts as a gateway. Instead of rendering `state.transactions` directly, the UI always renders the *result* of this function.
    *   It uses a series of logical checks (Date, Member, Amount, Search, and Category) that must all return `true` (AND logic) for an item to appear.
    *   The Category filter is a **multi-select** implementation using checkboxes.

3.  **Batch Operations:**
    *   Every transaction in the history list now has a checkbox. You can select multiple and click "Delete Selected". This action is also fully undoable.

4.  **Keyboard Support:**
    *   Standard `Ctrl+Z` and `Ctrl+Y` are intercepted via a global `keydown` listener.
