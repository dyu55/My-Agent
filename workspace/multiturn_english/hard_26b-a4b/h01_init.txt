To create this project, create a folder on your computer and place these three files inside it. I have included a functional "Add" feature so you can immediately see the list area working when you test it.

### 3. app.js
This file handles the logic: capturing user input, updating the math, and injecting new items into the HTML list.

```javascript
// Select DOM Elements
const financeForm = document.getElementById('finance-form');
const descriptionInput = document.getElementById('description');
const amountInput = document.getElementById('amount');
const transactionList = document.getElementById('transaction-list');
const balanceDisplay = document.getElementById('balance');

// State Management (Our "Database")
let transactions = [];

// Function to update the UI
function updateUI() {
    // 1. Clear the current list
    transactionList.innerHTML = '';

    // 2. If no transactions, show empty message
    if (transactions.length === 0) {
        transactionList.innerHTML = '<li class="empty-msg">No records found. Add your first transaction!</li>';
        balanceDisplay.innerText = `$0.00`;
        return;
    }

    // 3. Calculate Total Balance
    const total = transactions.reduce((acc, item) => acc + item.amount, 0);
    balanceDisplay.innerText = `$${total.toFixed(2)}`;

    // 4. Render each transaction
    transactions.forEach(transaction => {
        const li = document.createElement('li');
        
        // Determine if it's income or expense for styling
        const typeClass = transaction.amount > 0 ? 'transaction-item-income' : 'transaction-item-expense';
        li.classList.add(typeClass);

        li.innerHTML = `
            <span>${transaction.description}</span>
            <span>${transaction.amount > 0 ? '+' : ''}${transaction.amount.toFixed(2)}</span>
        `;

        transactionList.appendChild(li);
    });
}

// Function to handle form submission
function handleFormSubmit(e) {
    e.preventDefault();

    // Get values from input
    const description = descriptionInput.value;
    const amount = parseFloat(amountInput.value);

    // Create transaction object
    const newTransaction = {
        id: Date.now(),
        description: description,
        amount: amount
    };

    // Add to our array
    transactions.push(newTransaction);

    // Reset form and update display
    descriptionInput.value = '';
    amountInput.value = '';
    updateUI();
}

// Event Listeners
financeForm.addEventListener('submit', handleFormSubmit);

// Initialize App
updateUI();
```

### How to run it:
1. Save all three files in one folder.
2. Double-click `index.html`.
3. It will open in your default web browser.
4. **Test it:** Type "Salary" in description, "2000" in amount, and click Add. Then type "Rent" and "-800" and click Add. The list and balance will update automatically.
