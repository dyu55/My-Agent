### 3. app.js
The logic now includes a `ChartManager` to handle lifecycle (creating/destroying) and the theme toggle.

```javascript
/** 
 * STATE & THEME MANAGEMENT 
 **/
const state = {
    transactions: JSON.parse(localStorage.getItem('transactions')) || [],
    members: JSON.parse(localStorage.getItem('members')) || ['Admin'],
    isDarkMode: localStorage.getItem('darkMode') === 'true',
    currentChartMonth: new Date().toISOString().slice(0, 7), // YYYY-MM
    charts: {} // Store Chart.js instances
};

// --- THEME LOGIC ---
function initTheme() {
    if (state.isDarkMode) document.body.classList.add('dark-mode');
    document.getElementById('theme-toggle').addEventListener('click', () => {
        state.isDarkMode = !state.isDarkMode;
        document.body.classList.toggle('dark-mode');
        localStorage.setItem('darkMode', state.isDarkMode);
        render(); // Re-render to update chart colors if necessary
    });
}

/**
 * CHARTING LOGIC
 **/
function updateCharts() {
    const ctxCat = document.getElementById('categoryChart').getContext('2d');
    const ctxMem = document.getElementById('memberChart').getContext('2d');
    const ctxTrend = document.getElementById('trendChart').getContext('2d');

    // 1. Filter data for the selected month
    const monthTransactions = state.transactions.filter(t => t.date.startsWith(state.currentChartMonth));
    
    // 2. Aggregate Category Data (Pie)
    const catTotals = {};
    monthTransactions.forEach(t => {
        if (t.amount < 0) { // Only chart expenses
            catTotals[t.category] = (catTotals[t.category] || 0) + Math.abs(t.amount);
        }
    });

    // 3. Aggregate Member Data (Bar)
    const memTotals = {};
    monthTransactions.forEach(t => {
        if (t.amount < 0) {
            memTotals[t.member] = (memTotals[t.member] || 0) + Math.abs(t.amount);
        }
    });

    // 4. Aggregate 6-Month Trend (Line)
    const trendData = [];
    const trendLabels = [];
    for (let i = 5; i >= 0; i--) {
        const d = new Date();
        d.setMonth(d.getMonth() - i);
        const mStr = d.toISOString().slice(0, 7);
        const mSum = state.transactions
            .filter(t => t.date.startsWith(mStr) && t.amount < 0)
            .reduce((acc, t) => acc + Math.abs(t.amount), 0);
        trendLabels.push(mStr);
        trendData.push(mSum);
    }

    // Helper to destroy old charts to prevent memory leaks
    const destroyChart = (id) => {
        if (state.charts[id]) state.charts[id].destroy();
    };

    // Create/Update Pie Chart
    destroyChart('cat');
    state.charts['cat'] = new Chart(ctxCat, {
        type: 'pie',
        data: {
            labels: Object.keys(catTotals),
            datasets: [{
                data: Object.values(catTotals),
                backgroundColor: ['#ff6384', '#36a2eb', '#cc65fe', '#ffce56', '#2ecc71', '#e74c3c', '#f1c40f', '#9b59b6']
            }]
        },
        options: { responsive: true, maintainAspectRatio: false, plugins: { title: { display: true, text: 'Spending by Category' } } }
    });

    // Create/Update Bar Chart
    destroyChart('mem');
    state.charts['mem'] = new Chart(ctxMem, {
        type: 'bar',
        data: {
            labels: Object.keys(memTotals),
            datasets: [{
                label: 'Spent ($)',
                data: Object.values(memTotals),
                backgroundColor: '#4a90e2'
            }]
        },
        options: { responsive: true, maintainAspectRatio: false, plugins: { title: { display: true, text: 'Spending by Member' } } }
    });

    // Create/Update Line Chart
    destroyChart('trend');
    state.charts['trend'] = new Chart(ctxTrend, {
        type: 'line',
        data: {
            labels: trendLabels,
            datasets: [{
                label: 'Monthly Expenses',
                data: trendData,
                borderColor: '#b58900',
                tension: 0.3,
                fill: true,
                backgroundColor: 'rgba(181, 137, 0, 0.1)'
            }]
        },
        options: { responsive: true, maintainAspectRatio: false, plugins: { title: { display: true, text: '6-Month Spending Trend' } } }
    });
}

// --- INITIALIZATION & EVENT LISTENERS ---

function render() {
    // [Include the same transaction rendering logic from previous steps]
    // ...
    updateCharts();
}

// Month Selector Listener
document.getElementById('chart-month-selector').addEventListener('change', (e) => {
    state.currentChartMonth = e.target.value;
    updateCharts();
});

// Initialize App
window.addEventListener('DOMContentLoaded', () => {
    initTheme();
    
    // Set default month in selector
    const selector = document.getElementById('chart-month-selector');
    selector.value = state.currentChartMonth;

    // [Add existing listeners for Form, Undo, Redo, Members here]
    // ...
    
    render();
});
```

### How to test the new features:
1.  **Theme Toggle:** Click the 🌓 icon. Notice the background turns into a deep teal and the text becomes a soft cream. Refresh the page; it stays dark.
2.  **Responsive Charts:** Resize your browser window. On a mobile-sized window, the charts will stack vertically instead of being side-by-side.
3.  **Analytics:** 
    *   Add several transactions with different categories and members. 
    *   Ensure some have a negative amount (e.g., `-50`) so they show up in the "Spending" charts.
    *   Use the **Month Selector** to change the data. The Pie and Bar charts will update to show only what you spent in that specific month.
    *   The **Line Chart** will always show the last 6 months regardless of the selector (it's a trend tracker).
4.  **Mobile Usability:** All inputs and buttons have a `min-height` of `48px`, making them easy to tap with a thumb.
