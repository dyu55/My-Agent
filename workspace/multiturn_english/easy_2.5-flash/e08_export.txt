*   **Monthly Statistics:** Displays total income, total expenses, and net balance for the current month.
*   **Overall Monthly Budget:** A fixed budget of $5000 (you can easily edit this in the code) with a big red warning if total monthly expenses exceed it.

        // --- Configuration ---
        const ALL_CATEGORIES = [
            // Income
            'Salary', 'Freelance', 'Gifts', 'Interest', 'Bonus', 'Investment',
            // Expenses
            'Rent', 'Mortgage', 'Groceries', 'Utilities', 'Transportation', 'Dining Out', 'Entertainment',
            'Shopping', 'Health', 'Education', 'Insurance', 'Subscriptions', 'Bills', 'Travel', 'Pets',
            'Home Maintenance', 'Personal Care', 'Debt Payments', 'Investments', 'Savings', 'Other'
        ];
        const OVERALL_MONTHLY_BUDGET = 5000; // Your overall monthly budget in dollars

        // --- DOM Elements ---
        const transactionForm = document.getElementById('transactionForm');
        const dateInput = document.getElementById('date');
        const typeInput = document.getElementById('type');
        const categoryInput = document.getElementById('category');
        const amountInput = document.getElementById('amount');
        const noteInput = document.getElementById('note');
        const transactionsList = document.getElementById('transactionsList');
        const totalIncomeElem = document.getElementById('totalIncome');
        const totalExpensesElem = document.getElementById('totalExpenses');
        const netBalanceElem = document.getElementById('netBalance');
        const overallBudgetWarningElem = document.getElementById('overallBudgetWarning');
        const categoryBudgetsContainer = document.getElementById('categoryBudgetsContainer');
        const expenseChartCanvas = document.getElementById('expenseChart');
        let expenseChart; // To hold the Chart.js instance

        // --- Data Storage ---
        let transactions = JSON.parse(localStorage.getItem('transactions')) || [];
        let categoryBudgets = JSON.parse(localStorage.getItem('categoryBudgets')) || {};

        // --- Initialization ---
        document.addEventListener('DOMContentLoaded', () => {
            // Set default date to today
            dateInput.value = new Date().toISOString().split('T')[0];

            // Populate category dropdown
            ALL_CATEGORIES.forEach(cat => {
                const option = document.createElement('option');
                option.value = cat;
                option.textContent = cat;
                categoryInput.appendChild(option);
            });

            // Load and display everything
            loadCategoryBudgets(); // Must be before rendering summary to show current month's spending
            renderTransactions();
            renderSummary();
            renderCategoryBudgets(); // Render the editable budget inputs
            renderExpenseChart();
        });

        // --- Transaction Management ---
        transactionForm.addEventListener('submit', (e) => {
            e.preventDefault();

            const newTransaction = {
                id: Date.now(), // Unique ID for each transaction
                date: dateInput.value,
                type: typeInput.value,
                category: categoryInput.value,
                amount: parseFloat(amountInput.value),
                note: noteInput.value
            };

            transactions.push(newTransaction);
            localStorage.setItem('transactions', JSON.stringify(transactions));

            // Reset form fields
            dateInput.value = new Date().toISOString().split('T')[0]; // Reset to today
            typeInput.value = '';
            categoryInput.value = '';
            amountInput.value = '';
            noteInput.value = '';

            // Re-render everything
            renderTransactions();
            renderSummary();
            renderCategoryBudgets(); // Update spending status next to budget inputs
            renderExpenseChart();
        });

        function deleteTransaction(id) {
            transactions = transactions.filter(t => t.id !== id);
            localStorage.setItem('transactions', JSON.stringify(transactions));
            renderTransactions();
            renderSummary();
            renderCategoryBudgets(); // Update spending status
            renderExpenseChart();
        }

        function renderTransactions() {
            transactionsList.innerHTML = ''; // Clear existing list

            // Sort by date descending
            const sortedTransactions = [...transactions].sort((a, b) => new Date(b.date) - new Date(a.date));

            sortedTransactions.forEach(t => {
                const row = transactionsList.insertRow();
                row.innerHTML = `
                    <td>${t.date}</td>
                    <td class="${t.type.toLowerCase()}">${t.type}</td>
                    <td>${t.category}</td>
                    <td class="${t.type.toLowerCase()}">$${t.amount.toFixed(2)}</td>
                    <td>${t.note}</td>
                    <td><button onclick="deleteTransaction(${t.id})">Delete</button></td>
                `;
            });
        }

        // --- Summary and Budget Calculation ---
        function getCurrentMonthTransactions() {
            const now = new Date();
            const currentYear = now.getFullYear();
            const currentMonth = now.getMonth(); // 0-indexed

            return transactions.filter(t => {
                const transactionDate = new Date(t.date);
                return transactionDate.getFullYear() === currentYear && transactionDate.getMonth() === currentMonth;
            });
        }

        function renderSummary() {
            const currentMonthTransactions = getCurrentMonthTransactions();
            let totalIncome = 0;
            let totalExpenses = 0;

            currentMonthTransactions.forEach(t => {
                if (t.type === 'Income') {
                    totalIncome += t.amount;
                } else if (t.type === 'Expense') {
                    totalExpenses += t.amount;
                }
            });

            const netBalance = totalIncome - totalExpenses;

            totalIncomeElem.textContent = `$${totalIncome.toFixed(2)}`;
            totalExpensesElem.textContent = `$${totalExpenses.toFixed(2)}`;
            netBalanceElem.textContent = `$${netBalance.toFixed(2)}`;
            netBalanceElem.style.color = netBalance >= 0 ? '#388E3C' : '#D32F2F'; // Green for positive, red for negative

            // Overall Budget Warning
            if (totalExpenses > OVERALL_MONTHLY_BUDGET) {
                overallBudgetWarningElem.style.display = 'block';
            } else {
                overallBudgetWarningElem.style.display = 'none';
            }
        }

        // --- Category Budget Management ---
        function loadCategoryBudgets() {
            // Initialize budgets for all categories if they don't exist
            ALL_CATEGORIES.filter(cat => cat !== 'Salary' && cat !== 'Freelance' && cat !== 'Gifts' && cat !== 'Interest' && cat !== 'Bonus' && cat !== 'Investment').forEach(cat => { // Exclude income categories for budget setting
                if (categoryBudgets[cat] === undefined) {
                    categoryBudgets[cat] = 0; // Default budget is 0
                }
            });
            localStorage.setItem('categoryBudgets', JSON.stringify(categoryBudgets));
        }

        function renderCategoryBudgets() {
            categoryBudgetsContainer.innerHTML = '';
            const currentMonthTransactions = getCurrentMonthTransactions();
            const monthlyCategorySpending = {};

            // Calculate current month's spending per category
            currentMonthTransactions.forEach(t => {
                if (t.type === 'Expense') {
                    monthlyCategorySpending[t.category] = (monthlyCategorySpending[t.category] || 0) + t.amount;
                }
            });

            // Filter out income categories and 'Select Category'
            const expenseCategories = ALL_CATEGORIES.filter(cat => {
                // Determine if a category is primarily for expenses. This is a simple heuristic.
                return !['Salary', 'Freelance', 'Gifts', 'Interest', 'Bonus', 'Investment', ''].includes(cat);
            });

            expenseCategories.sort().forEach(cat => {
                const currentBudget = categoryBudgets[cat] || 0;
                const currentSpending = monthlyCategorySpending[cat] || 0;
                const isOverBudget = currentBudget > 0 && currentSpending > currentBudget;

                const budgetItem = document.createElement('div');
                budgetItem.className = 'budget-item';
                budgetItem.innerHTML = `
                    <label for="budget-${cat}">${cat}:</label>
                    <input type="number" id="budget-${cat}" value="${currentBudget.toFixed(2)}" step="0.01" min="0">
                    <span class="budget-category-status ${isOverBudget ? 'over-budget' : 'under-budget'}">
                        Spent: $${currentSpending.toFixed(2)}
                        ${isOverBudget ? ' (OVER)' : ''}
                    </span>
                    ${isOverBudget ? `<div class="category-budget-warning" style="margin-top:5px; flex-basis: 100%; display:block;">
                        🚨 ${cat} spending is over budget!
                    </div>` : ''}
                `;
                categoryBudgetsContainer.appendChild(budgetItem);
            });
        }

        function saveCategoryBudgets() {
            const newBudgets = {};
            // Iterate over all expense categories to capture their input values
            ALL_CATEGORIES.filter(cat => !['Salary', 'Freelance', 'Gifts', 'Interest', 'Bonus', 'Investment', ''].includes(cat)).forEach(cat => {
                const inputElem = document.getElementById(`budget-${cat}`);
                if (inputElem) {
                    newBudgets[cat] = parseFloat(inputElem.value) || 0;
                }
            });

            categoryBudgets = newBudgets;
            localStorage.setItem('categoryBudgets', JSON.stringify(categoryBudgets));
            alert('Category budgets saved!');
            renderCategoryBudgets(); // Re-render to update status based on new budgets
            renderSummary(); // Re-check overall budget if needed (though not directly affected, good practice)
        }

        // --- Charting ---
        function renderExpenseChart() {
            const currentMonthTransactions = getCurrentMonthTransactions();
            const expenseBreakdown = {};

            currentMonthTransactions.forEach(t => {
                if (t.type === 'Expense') {
                    expenseBreakdown[t.category] = (expenseBreakdown[t.category] || 0) + t.amount;
                }
            });

            const labels = Object.keys(expenseBreakdown);
            const data = Object.values(expenseBreakdown);

            const backgroundColors = [
                '#FF6384', '#36A2EB', '#FFCE56', '#4BC0C0', '#9966FF', '#FF9F40',
                '#E7E9ED', '#8D6E63', '#4DD0E1', '#D4E157', '#AFB42B', '#7CB342',
                '#FFA726', '#FFCA28', '#66BB6A', '#26A69A', '#AB47BC', '#7E57C2',
                '#EF5350', '#EC407A', '#78909C', '#BCAAA4', '#FF8A65', '#9CCC65'
            ];

            if (expenseChart) {
                expenseChart.destroy(); // Destroy previous chart instance if it exists
            }

            expenseChart = new Chart(expenseChartCanvas, {
                type: 'pie',
                data: {
                    labels: labels,
                    datasets: [{
                        data: data,
                        backgroundColor: backgroundColors.slice(0, labels.length)
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: {
                            position: 'right',
                            labels: {
                                font: {
                                    size: 14
                                }
                            }
                        },
                        title: {
                            display: true,
                            text: 'Monthly Spending by Category',
                            font: {
                                size: 18
                            },
                            color: '#333'
                        }
                    }
                }
            });
        }

        // --- Export to CSV ---
        function exportToCsv() {
            if (transactions.length === 0) {
                alert('No transactions to export!');
                return;
            }

            const headers = ['Date', 'Type', 'Category', 'Amount', 'Note'];
            let csvContent = headers.join(',') + '\n';

            transactions.forEach(t => {
                const row = [
                    t.date,
                    t.type,
                    `"${t.category.replace(/"/g, '""')}"`, // Handle commas in category names
                    t.amount.toFixed(2),
                    `"${t.note.replace(/"/g, '""')}"` // Handle commas/quotes in notes
                ];
                csvContent += row.join(',') + '\n';
            });

            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.setAttribute('download', 'finance_records.csv');
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }
