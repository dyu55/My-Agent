To add a chart without you having to write complex math, we will use a library called **Chart.js**. It is the most popular tool for creating beautiful charts on the web.

Since we are using a library, I have added a "Script" link at the top of the code that tells your browser to download the Chart.js logic. I have also added a **Pie Chart** because it is the best way to see "slices" of your spending.

To keep the "Cozy" vibe, I've created a custom color palette for the chart slices that matches your sage and coral theme.

<script>
    const MONTHLY_BUDGET = 5000;
    let categoryChart = null; // Global variable to hold the chart instance

    const balanceDisplay = document.getElementById('balance');
    const monthIncomeDisplay = document.getElementById('month-income');
    const monthSpendDisplay = document.getElementById('month-spend');
    const monthLeftDisplay = document.getElementById('month-left');
    const budgetWarning = document.getElementById('budget-warning');
    const list = document.getElementById('list');
    const form = document.getElementById('transaction-form');
    
    const typeInput = document.getElementById('type');
    const amountInput = document.getElementById('amount');
    const categoryInput = document.getElementById('category');
    const dateInput = document.getElementById('date');
    const noteInput = document.getElementById('note');

    dateInput.valueAsDate = new Date();
    let transactions = JSON.parse(localStorage.getItem('transactions')) || [];

    function updateStats() {
        const now = new Date();
        const currentMonth = now.getMonth();
        const currentYear = now.getFullYear();

        const totalBalance = transactions.reduce((acc, item) => {
            return item.type === 'income' ? acc + item.amount : acc - item.amount;
        }, 0);
        balanceDisplay.innerText = `$${totalBalance.toFixed(2)}`;

        const thisMonthTransactions = transactions.filter(t => {
            const tDate = new Date(t.date);
            return tDate.getMonth() === currentMonth && tDate.getFullYear() === currentYear;
        });

        const monthIncome = thisMonthTransactions
            .filter(t => t.type === 'income')
            .reduce((acc, t) => acc + t.amount, 0);

        const monthSpend = thisMonthTransactions
            .filter(t => t.type === 'expense')
            .reduce((acc, t) => acc + t.amount, 0);

        const leftOver = MONTHLY_BUDGET - monthSpend;

        monthIncomeDisplay.innerText = `$${monthIncome.toFixed(2)}`;
        monthSpendDisplay.innerText = `$${monthSpend.toFixed(2)}`;
        monthLeftDisplay.innerText = `$${leftOver.toFixed(2)}`;

        if (monthSpend > MONTHLY_BUDGET) {
            budgetWarning.style.display = 'block';
            monthLeftDisplay.classList.add('text-expense');
        } else {
            budgetWarning.style.display = 'none';
            monthLeftDisplay.classList.remove('text-expense');
        }

        updateChart(thisMonthTransactions);
    }

    function updateChart(thisMonthTransactions) {
        const ctx = document.getElementById('categoryChart').getContext('2d');
        
        // 1. Calculate totals per category for expenses only
        const categoryTotals = {};
        thisMonthTransactions.filter(t => t.type === 'expense').forEach(t => {
            categoryTotals[t.category] = (categoryTotals[t.category] || 0) + t.amount;
        });

        const labels = Object.keys(categoryTotals);
        const data = Object.values(categoryTotals);

        // 2. If no expenses, don't try to draw a chart
        if (labels.length === 0) {
            if (categoryChart) categoryChart.destroy();
            return;
        }

        // 3. Destroy old chart before creating a new one to prevent overlapping
        if (categoryChart) {
            categoryChart.destroy();
        }

        // 4. Create the Pie Chart
        categoryChart = new Chart(ctx, {
            type: 'pie',
            data: {
                labels: labels,
                datasets: [{
                    data: data,
                    backgroundColor: [
                        '#8E9775', '#CD5C5C', '#A2B5BB', '#E9967A', '#6B8E23', '#D1C7B7', '#F0EBE3'
                    ],
                    borderWidth: 2,
                    borderColor: '#FFFFFF'
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        position: 'bottom',
                        labels: { color: '#4A3F35', font: { size: 11 } }
                    }
                }
            }
        });
    }

    function addTransactionDOM(transaction) {
        const item = document.createElement('li');
        item.classList.add('transaction-item', transaction.type);
        item.innerHTML = `
            <div class="item-details">
                <span class="item-main">${transaction.category}</span>
                <span class="item-sub">${transaction.date} ${transaction.note ? '• ' + transaction.note : ''}</span>
            </div>
            <div class="item-amount-group">
                <span class="item-amount">${transaction.type === 'income' ? '+' : '-'}$${transaction.amount.toFixed(2)}</span>
                <span class="delete-btn" onclick="removeTransaction(${transaction.id})">✕</span>
            </div>
        `;
        list.appendChild(item);
    }

    function removeTransaction(id) {
        transactions = transactions.filter(t => t.id !== id);
        localStorage.setItem('transactions', JSON.stringify(transactions));
        init();
    }

    function init() {
        list.innerHTML = '';
        transactions.sort((a, b) => new Date(b.date) - new Date(a.date));
        transactions.forEach(addTransactionDOM);
        updateStats();
    }

    form.addEventListener('submit', (e) => {
        e.preventDefault();
        const transaction = {
            id: Date.now(),
            type: typeInput.value,
            amount: parseFloat(amountInput.value),
            category: categoryInput.value,
            date: dateInput.value,
            note: noteInput.value
        };
        transactions.push(transaction);
        localStorage.setItem('transactions', JSON.stringify(transactions));
        init();
        amountInput.value = '';
        noteInput.value = '';
        dateInput.valueAsDate = new Date();
    });

    function exportData() {
        const dataStr = JSON.stringify(transactions);
        const dataUri = 'data:application/json;charset=utf-8,'+ encodeURIComponent(dataStr);
        const exportFileDefaultName = 'finance-backup.json';
        let linkElement = document.createElement('a');
        linkElement.setAttribute('href', dataUri);
        linkElement.setAttribute('download', exportFileDefaultName);
        linkElement.click();
    }

    function importData(event) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function(e) {
            try {
                const importedTransactions = JSON.parse(e.target.result);
                if (Array.isArray(importedTransactions)) {
                    transactions = importedTransactions;
                    localStorage.setItem('transactions', JSON.stringify(transactions));
                    init();
                    alert("Data imported successfully!");
                }
            } catch (err) { alert("Error reading file."); }
        };
        reader.readAsText(file);
    }
    init();
</script>
</body>

### How this works (The "Magic" part):

1.  **The Chart.js Library:** I added `<script src="https://cdn.jsdelivr.net/npm/chart.js"></script>`. This tells your browser: *"I don't know how to draw a pie chart, go to this website and download the instructions on how to do it."*
2.  **Data Grouping:** In the `updateChart` function, the app now does a bit of math. It looks at all your expenses for the current month and says: *"Okay, you spent $50 on Food, another $20 on Food, and $30 on Transport."* It then sums those up into a total for "Food" ($70) and "Transport" ($30).
3.  **Dynamic Updates:** The chart is linked to the `init()` function. This means the moment you click "Add to Ledger" or delete a transaction, the chart instantly refreshes to show the new data.
4.  **Avoiding "Chart Glitches":** In JavaScript, if you try to draw a new chart on top of an old one, it creates a glitch where the old data flickers when you hover over it. I added `categoryChart.destroy()`, which clears the canvas before drawing the updated chart.

### How to test it:
1. Save and refresh your browser.
2. Add a few **Expense** records with different categories (e.g., $100 for Food, $50 for Transport).
3. You will see the Pie Chart automatically appear and slice itself based on those percentages!
