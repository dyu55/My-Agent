    let categoryChart = null;
    const balanceDisplay = document.getElementById('balance');
    const monthIncomeDisplay = document.getElementById('month-income');
    const monthSpendDisplay = document.getElementById('month-spend');
    const monthLeftDisplay = document.getElementById('month-left');
    const catWarningsDisplay = document.getElementById('category-warnings');
    const settingsContent = document.getElementById('settings-content');
    const list = document.getElementById('list');
    const form = document.getElementById('transaction-form');
    const typeInput = document.getElementById('type');
    const amountInput = document.getElementById('amount');
    const categoryInput = document.getElementById('category');
    const dateInput = document.getElementById('date');
    const noteInput = document.getElementById('note');

    dateInput.valueAsDate = new Date();
    let transactions = JSON.parse(localStorage.getItem('transactions')) || [];
    let categoryBudgets = JSON.parse(localStorage.getItem('categoryBudgets')) || {};

    function initBudgetSettings() {
        const categories = Array.from(categoryInput.options).map(opt => opt.value);
        settingsContent.innerHTML = '';
        categories.forEach(cat => {
            const val = categoryBudgets[cat] || 0;
            const group = document.createElement('div');
            group.classList.add('budget-input-group');
            group.innerHTML = `<label>${cat}</label><input type="number" value="${val}" onchange="updateBudgetLimit('${cat}', this.value)" placeholder="0">`;
            settingsContent.appendChild(group);

    function updateBudgetLimit(cat, value) {
        categoryBudgets[cat] = parseFloat(value) || 0;
        localStorage.setItem('categoryBudgets', JSON.stringify(categoryBudgets));
        updateStats();
    }

    function toggleSettings() {
        const isActive = settingsContent.classList.toggle('active');
        document.getElementById('toggle-icon').innerText = isActive ? '▲' : '▼';
    }

    function updateStats() {
        const now = new Date();
        const currentMonth = now.getMonth();
        const currentYear = now.getFullYear();

        const totalBalance = transactions.reduce((acc, item) => item.type === 'income' ? acc + item.amount : acc - item.amount, 0);
        balanceDisplay.innerText = `$${totalBalance.toFixed(2)}`;

        const thisMonthTransactions = transactions.filter(t => {
            const tDate = new Date(t.date);
            return tDate.getMonth() === currentMonth && tDate.getFullYear() === currentYear;
        });

        const monthIncome = thisMonthTransactions.filter(t => t.type === 'income').reduce((acc, t) => acc + t.amount, 0);
        const monthSpend = thisMonthTransactions.filter(t => t.type === 'expense').reduce((acc, t) => acc + t.amount, 0);
        const totalBudgetLimit = Object.values(categoryBudgets).reduce((a, b) => a + b, 0);

        monthIncomeDisplay.innerText = `$${monthIncome.toFixed(2)}`;
        monthSpendDisplay.innerText = `$${monthSpend.toFixed(2)}`;
        monthLeftDisplay.innerText = `$${(totalBudgetLimit - monthSpend).toFixed(2)}`;

        catWarningsDisplay.innerHTML = '';
        const categoryTotals = {};
        thisMonthTransactions.filter(t => t.type === 'expense').forEach(t => {
            categoryTotals[t.category] = (categoryTotals[t.category] || 0) + t.amount;
        });

        for (const [cat, spent] of Object.entries(categoryTotals)) {
            const limit = categoryBudgets[cat] || 0;
            if (limit > 0 && spent > limit) {
                const warning = document.createElement('div');
                warning.classList.add('cat-warning');
                warning.innerHTML = `<span>⚠️ ${cat} Over Budget!</span> <span>$${spent.toFixed(0)} / $${limit.toFixed(0)}</span>`;
                catWarningsDisplay.appendChild(warning);
            }
        }
        updateChart(thisMonthTransactions);
    }

    function updateChart(thisMonthTransactions) {
        const ctx = document.getElementById('categoryChart').getContext('2d');
        const categoryTotals = {};
        thisMonthTransactions.filter(t => t.type === 'expense').forEach(t => {
            categoryTotals[t.category] = (categoryTotals[t.category] || 0) + t.amount;
        });
        const labels = Object.keys(categoryTotals);
        const data = Object.values(categoryTotals);
        if (labels.length === 0) { if (categoryChart) categoryChart.destroy(); return; }
        if (categoryChart) categoryChart.destroy();

            </div>
        `;
        list.appendChild(item);
    }

    function removeTransaction(id) {
        transactions = transactions.filter(t => t.id !== id);
        localStorage.setItem('transactions', JSON.stringify(transactions));
        init();
    }

    function init() {
        list.innerHTML = '';
        transactions.sort((a, b) => new Date(b.date) - new Date(a.date));
        transactions.forEach(addTransactionDOM);
        updateStats();
    }

    form.addEventListener('submit', (e) => {

        localStorage.setItem('transactions', JSON.stringify(transactions));

    function exportToCSV() {
        if (transactions.length === 0) {
            alert("No transactions to export!");
            return;
        }

        // 1. Create the CSV Header
        let csvContent = "Date,Type,Category,Amount,Note\n";

        // 2. Add each transaction as a row
        transactions.forEach(t => {
            // We wrap fields in quotes so that commas inside the "Note" don't break the spreadsheet
            const row = [
                `"${t.date}"`,
                `"${t.type}"`,
                `"${t.category}"`,
                `"${t.amount}"`,
                `"${t.note || ''}"`
            ].join(",");
            csvContent += row + "\n";
        });

        // 3. Create a downloadable file
        const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement("a");
        link.setAttribute("href", url);
        link.setAttribute("download", `finance_records_${new Date().toISOString().split('T')[0]}.csv`);
        link.style.visibility = 'hidden';
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    }

    function exportData() {
        const data = { transactions, categoryBudgets };
        const dataStr = JSON.stringify(data);
        const dataUri = 'data:application/json;charset=utf-8,'+ encodeURIComponent(dataStr);
        let linkElement = document.createElement('a');
        linkElement.setAttribute('href', dataUri);
        linkElement.setAttribute('download', 'finance-backup.json');
        linkElement.click();
    }

    function importData(event) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function(e) {
            try {
                const imported = JSON.parse(e.target.result);
                if (imported.transactions) {
                    transactions = imported.transactions;
                    categoryBudgets = imported.categoryBudgets || {};
                    localStorage.setItem('transactions', JSON.stringify(transactions));
                    localStorage.setItem('categoryBudgets', JSON.stringify(categoryBudgets));
                    initBudgetSettings();
                    init();
                    alert("Data and budgets restored!");
                }
            } catch (err) { alert("Error reading file."); }
        };
        reader.readAsText(file);
    }

    initBudgetSettings();
    init();

### How the CSV Export works:
1.  **The Header:** I created a string `Date,Type,Category,Amount,Note`. This tells Excel exactly what each column is.
2.  **Handling Commas:** A common problem with CSVs is that if you write a note like *"Lunch, with Sarah"*, the comma in the text makes Excel think it's a new column. To fix this, I wrapped every field in double quotes (`" "`), which tells the spreadsheet software: *"Everything inside these quotes belongs in one single cell."*
3.  **The "Blob":** Since there is no server, the app creates a "Blob" (a temporary piece of data in your browser's memory) and tricks the browser into "downloading" it as a file.
4.  **Dynamic Naming:** The file will be named `finance_records_YYYY-MM-DD.csv` so you can keep organized archives for different tax years.
