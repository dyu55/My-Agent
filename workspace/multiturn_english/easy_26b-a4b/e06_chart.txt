To add charts without writing thousands of lines of math code, we are going to use a "Library." Think of a library as a professional toolset that we "borrow" and plug into your app. For this, we will use **Chart.js**, which is the industry standard for beautiful, simple web charts.

I have added a **Monthly Spending Breakdown** pie chart. It uses the same "warm and cozy" color palette so it doesn't look out of place.

<script>
    const form = document.getElementById('transaction-form');
    const list = document.getElementById('transaction-list');
    const budgetWarning = document.getElementById('budget-warning');
    const totalBalanceDisplay = document.getElementById('total-balance');
    const totalIncomeDisplay = document.getElementById('total-income');
    const totalExpenseDisplay = document.getElementById('total-expense');
    const monthIncomeDisplay = document.getElementById('month-income');
    const monthExpenseDisplay = document.getElementById('month-expense');
    const monthLeftoverDisplay = document.getElementById('month-leftover');

    const MONTHLY_BUDGET = 5000;
    let transactions = JSON.parse(localStorage.getItem('my_finance_data')) || [];
    let myChart = null; // Variable to hold our chart object

    document.getElementById('date').valueAsDate = new Date();

    function updateAllStats() {
        // 1. ALL-TIME MATH
        const allTimeIncome = transactions.filter(t => t.type === 'income').reduce((acc, t) => acc + t.amount, 0);
        const allTimeExpense = transactions.filter(t => t.type === 'expense').reduce((acc, t) => acc + t.amount, 0);
        totalIncomeDisplay.innerText = `$${allTimeIncome.toFixed(0)}`;
        totalExpenseDisplay.innerText = `$${allTimeExpense.toFixed(0)}`;
        totalBalanceDisplay.innerText = `$${(allTimeIncome - allTimeExpense).toFixed(0)}`;

        // 2. MONTHLY MATH
        const now = new Date();
        const monthlyTransactions = transactions.filter(t => {
            const d = new Date(t.date);
            return d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear();
        });

        const mIncome = monthlyTransactions.filter(t => t.type === 'income').reduce((acc, t) => acc + t.amount, 0);
        const mExpense = monthlyTransactions.filter(t => t.type === 'expense').reduce((acc, t) => acc + t.amount, 0);
        monthIncomeDisplay.innerText = `$${mIncome.toFixed(0)}`;
        monthExpenseDisplay.innerText = `$${mExpense.toFixed(0)}`;
        monthLeftoverDisplay.innerText = `$${(mIncome - mExpense).toFixed(0)}`;

        budgetWarning.style.display = (mExpense > MONTHLY_BUDGET) ? 'block' : 'none';

        // 3. TRIGGER CHART UPDATE
        updateChart(monthlyTransactions);
    }

    function updateChart(monthlyTransactions) {
        const chartWrapper = document.getElementById('chart-wrapper');
        
        // Only show chart if there are expenses this month
        const expensesOnly = monthlyTransactions.filter(t => t.type === 'expense');
        if (expensesOnly.length === 0) {
            chartWrapper.style.display = 'none';
            return;
        }
        chartWrapper.style.display = 'flex';

        // Grouping logic: Totaling amounts by category
        const categoryMap = {};
        expensesOnly.forEach(t => {
            categoryMap[t.category] = (categoryMap[t.category] || 0) + t.amount;
        });

        const labels = Object.keys(categoryMap);
        const data = Object.values(categoryMap);

        // Cozy color palette for the chart
        const colors = ['#8d6e63', '#8da47e', '#c68b8b', '#d4a373', '#a3b18a', '#ccd5ae', '#e9edc9'];

        // If a chart already exists, we must destroy it before making a new one
        // This prevents the "flicker" or old data showing up on hover
        if (myChart) {
            myChart.destroy();
        }

        const ctx = document.getElementById('expenseChart').getContext('2d');
        myChart = new Chart(ctx, {
            type: 'pie',
            data: {
                labels: labels,
                datasets: [{
                    data: data,
                    backgroundColor: colors,
                    borderWidth: 2,
                    borderColor: '#ffffff'
                }]
            },
            options: {
                responsive: true,
                plugins: {
                    legend: { position: 'bottom', labels: { font: { family: 'Quicksand', size: 12 } } }
                }
            }
        });
    }

    function renderTable() {
        list.innerHTML = '';
        transactions.sort((a, b) => new Date(b.date) - new Date(a.date));
        transactions.forEach(t => {
            const row = document.createElement('tr');
            const isIncome = t.type === 'income';
            row.innerHTML = `
                <td><small>${t.date.split('-').slice(1).join('/')}</small></td>
                <td><strong>${t.category}</strong><br><small style="color:#999">${t.note || ''}</small></td>
                <td class="${isIncome ? 'amt-income' : 'amt-expense'}">${isIncome ? '+' : '-'}$${t.amount.toFixed(2)}</td>
                <td><button class="delete-btn" onclick="deleteTransaction(${t.id})">✕</button></td>
            `;
            list.appendChild(row);
        });
    }

    function addTransaction(e) {
        e.preventDefault();
        transactions.push({
            id: Date.now(),
            type: document.getElementById('type').value,
            amount: parseFloat(document.getElementById('amount').value),
            category: document.getElementById('category').value,
            date: document.getElementById('date').value,
            note: document.getElementById('note').value
        });
        saveAndRefresh();
        form.reset();
        document.getElementById('date').valueAsDate = new Date();
    }

    function deleteTransaction(id) {
        transactions = transactions.filter(t => t.id !== id);
        saveAndRefresh();
    }

    function saveAndRefresh() {
        localStorage.setItem('my_finance_data', JSON.stringify(transactions));
        updateAllStats();
        renderTable();
    }

    function exportData() {
        const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(transactions));
        const downloadAnchorNode = document.createElement('a');
        downloadAnchorNode.setAttribute("href", dataStr);
        downloadAnchorNode.setAttribute("download", "finance_backup.json");
        document.body.appendChild(downloadAnchorNode);
        downloadAnchorNode.click();
        downloadAnchorNode.remove();
    }

    function importData() { document.getElementById('fileInput').click(); }

    function handleFile(event) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function(e) {
            try {
                const importedData = JSON.parse(e.target.result);
                if (Array.isArray(importedData)) {
                    transactions = importedData;
                    saveAndRefresh();
                    alert("Restored successfully!");
                }
            } catch (err) { alert("Error reading file."); }
        };
        reader.readAsText(file);
    }

    form.addEventListener('submit', addTransaction);
    updateAllStats();
    renderTable();
</script>

### How the Chart Works:

1.  **The Library Import:** In the `<head>`, you'll see `<script src="https://cdn.jsdelivr.net/npm/chart.js"></script>`. This tells the browser to go to the internet and grab the Chart.js engine.
2.  **The `updateChart()` Logic:**
    *   **Filtering:** It looks at your transactions and says: *"Show me only the ones where the type is 'expense' and the month matches today."*
    *   **Grouping:** It creates a `categoryMap`. If you have two "Food" entries ($10 and $20), it combines them into one "Food" entry of $30.
    *   **The "Destroy" Rule:** This is a common programming trap. If you try to draw a new chart on top of an old one, the computer gets confused (you'll see weird glitches when you hover over it). My code calls `myChart.destroy()` before drawing a new one to keep it clean.
3.  **Smart Display:** I added a line that says `if (expensesOnly.length === 0) { chartWrapper.style.display = 'none'; }`. This means if you haven't spent any money this month yet, the empty chart box won't clutter up your beautiful interface.
