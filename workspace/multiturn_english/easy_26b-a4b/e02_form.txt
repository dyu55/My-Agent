<script>
    // JAVASCRIPT: The Brain
    const form = document.getElementById('transaction-form');
    const list = document.getElementById('transaction-list');
    const balanceDisplay = document.getElementById('total-balance');
    const incomeDisplay = document.getElementById('total-income');
    const expenseDisplay = document.getElementById('total-expense');

    // Load data from local storage
    let transactions = JSON.parse(localStorage.getItem('my_finance_data')) || [];

    // Set default date to today
    document.getElementById('date').valueAsDate = new Date();

    function updateDashboard() {
        const amounts = transactions.map(t => t.type === 'income' ? t.amount : -t.amount);
        
        const total = amounts.reduce((acc, item) => (acc += item), 0).toFixed(2);
        const income = transactions
            .filter(t => t.type === 'income')
            .reduce((acc, t) => acc + t.amount, 0).toFixed(2);
        const expense = transactions
            .filter(t => t.type === 'expense')
            .reduce((acc, t) => acc + t.amount, 0).toFixed(2);

        balanceDisplay.innerText = `$${total}`;
        incomeDisplay.innerText = `+$${income}`;
        expenseDisplay.innerText = `-$${expense}`;
    }

    function renderTable() {
        list.innerHTML = '';
        
        // Sort transactions by date (newest first)
        transactions.sort((a, b) => new Date(b.date) - new Date(a.date));

        transactions.forEach(t => {
            const row = document.createElement('tr');
            const isIncome = t.type === 'income';
            
            row.innerHTML = `
                <td>${t.date}</td>
                <td>${t.category}</td>
                <td><small>${t.note || '-'}</small></td>
                <td class="${isIncome ? 'amt-income' : 'amt-expense'}">
                    ${isIncome ? '+' : '-'}$${t.amount.toFixed(2)}
                </td>
                <td>
                    <button class="delete-btn" onclick="deleteTransaction(${t.id})">Delete</button>
                </td>
            `;
            list.appendChild(row);
        });
    }

    function addTransaction(e) {
        e.preventDefault();

        const newTransaction = {
            id: Date.now(), // Unique ID using timestamp
            type: document.getElementById('type').value,
            amount: parseFloat(document.getElementById('amount').value),
            category: document.getElementById('category').value,
            date: document.getElementById('date').value,
            note: document.getElementById('note').value
        };

        transactions.push(newTransaction);
        saveAndRefresh();
        form.reset();
        document.getElementById('date').valueAsDate = new Date(); // Reset date to today
    }

    function deleteTransaction(id) {
        transactions = transactions.filter(t => t.id !== id);
        saveAndRefresh();
    }

    function saveAndRefresh() {
        localStorage.setItem('my_finance_data', JSON.stringify(transactions));
        updateDashboard();
        renderTable();
    }

    form.addEventListener('submit', addTransaction);

    // Initial Load
    updateDashboard();
    renderTable();
</script>

### What changed in this version?

1.  **The Logic (JavaScript):**
    *   **The Object:** Previously, a transaction was just `{text, amount}`. Now it is a much larger object: `{id, type, amount, category, date, note}`.
    *   **The Math:** Instead of just summing numbers, the code now looks at the `type`. If the type is "expense," it treats the number as a negative for the balance calculation.
    *   **Sorting:** I added a line of code that automatically sorts your transactions so the newest ones appear at the top of the table.

2.  **The Visuals (HTML/CSS):**
    *   **The Dashboard:** I added three "Cards" at the top. This gives you an instant summary of your financial health.
    *   **The Form:** I used `display: grid` in CSS. This allows us to put two inputs side-by-side (like Type and Amount), making the form look professional and compact.
    *   **The Table:** We moved from a `<ul>` (list) to a `<table>`. Tables are the standard way to display structured data in web apps.

3.  **User Experience (UX):**
    *   **The Type Selector:** You no longer have to remember to type a minus sign `-` for expenses. You simply select "Expense" from the dropdown, and the app handles the math for you.
    *   **Automatic Date:** The app automatically sets the date field to "Today" so you don't have to click it every time.
