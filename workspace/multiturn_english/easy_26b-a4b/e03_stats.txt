<script>
    const form = document.getElementById('transaction-form');
    const list = document.getElementById('transaction-list');
    const budgetWarning = document.getElementById('budget-warning');
    
    // Elements for All-Time Dashboard
    const totalBalanceDisplay = document.getElementById('total-balance');
    const totalIncomeDisplay = document.getElementById('total-income');
    const totalExpenseDisplay = document.getElementById('total-expense');

    // Elements for Monthly Insights
    const monthIncomeDisplay = document.getElementById('month-income');
    const monthExpenseDisplay = document.getElementById('month-expense');
    const monthLeftoverDisplay = document.getElementById('month-leftover');

    const MONTHLY_BUDGET = 5000;

    let transactions = JSON.parse(localStorage.getItem('my_finance_data')) || [];

    document.getElementById('date').valueAsDate = new Date();

    function updateAllStats() {
        // 1. ALL-TIME CALCULATIONS
        const allTimeIncome = transactions
            .filter(t => t.type === 'income')
            .reduce((acc, t) => acc + t.amount, 0);
        
        const allTimeExpense = transactions
            .filter(t => t.type === 'expense')
            .reduce((acc, t) => acc + t.amount, 0);

        const allTimeBalance = allTimeIncome - allTimeExpense;

        totalIncomeDisplay.innerText = `$${allTimeIncome.toFixed(2)}`;
        totalExpenseDisplay.innerText = `$${allTimeExpense.toFixed(2)}`;
        totalBalanceDisplay.innerText = `$${allTimeBalance.toFixed(2)}`;

        // 2. MONTHLY CALCULATIONS
        const now = new Date();
        const currentMonth = now.getMonth();
        const currentYear = now.getFullYear();

        const monthlyTransactions = transactions.filter(t => {
            const d = new Date(t.date);
            return d.getMonth() === currentMonth && d.getFullYear() === currentYear;
        });

        const mIncome = monthlyTransactions
            .filter(t => t.type === 'income')
            .reduce((acc, t) => acc + t.amount, 0);

        const mExpense = monthlyTransactions
            .filter(t => t.type === 'expense')
            .reduce((acc, t) => acc + t.amount, 0);

        const mLeftover = mIncome - mExpense;

        monthIncomeDisplay.innerText = `$${mIncome.toFixed(2)}`;
        monthExpenseDisplay.innerText = `$${mExpense.toFixed(2)}`;
        monthLeftoverDisplay.innerText = `$${mLeftover.toFixed(2)}`;

        // 3. BUDGET WARNING LOGIC
        if (mExpense > MONTHLY_BUDGET) {
            budgetWarning.style.display = 'block';
        } else {
            budgetWarning.style.display = 'none';
        }
    }

    function renderTable() {
        list.innerHTML = '';
        transactions.sort((a, b) => new Date(b.date) - new Date(a.date));

        transactions.forEach(t => {
            const row = document.createElement('tr');
            const isIncome = t.type === 'income';
            row.innerHTML = `
                <td>${t.date}</td>
                <td>${t.category}</td>
                <td><small>${t.note || '-'}</small></td>
                <td class="${isIncome ? 'amt-income' : 'amt-expense'}">
                    ${isIncome ? '+' : '-'}$${t.amount.toFixed(2)}
                </td>
                <td><button class="delete-btn" onclick="deleteTransaction(${t.id})">Delete</button></td>
            `;
            list.appendChild(row);
        });
    }

    function addTransaction(e) {
        e.preventDefault();
        const newTransaction = {
            id: Date.now(),
            type: document.getElementById('type').value,
            amount: parseFloat(document.getElementById('amount').value),
            category: document.getElementById('category').value,
            date: document.getElementById('date').value,
            note: document.getElementById('note').value
        };
        transactions.push(newTransaction);
        saveAndRefresh();
        form.reset();
        document.getElementById('date').valueAsDate = new Date();
    }

    function deleteTransaction(id) {
        transactions = transactions.filter(t => t.id !== id);
        saveAndRefresh();
    }

    function saveAndRefresh() {
        localStorage.setItem('my_finance_data', JSON.stringify(transactions));
        updateAllStats();
        renderTable();
    }

    form.addEventListener('submit', addTransaction);
    updateAllStats();
    renderTable();
</script>

### What's new in this version?

1.  **The Filter Logic:** Inside `updateAllStats()`, I created a variable called `monthlyTransactions`. This uses a JavaScript method called `.filter()`. It loops through every item you've ever entered and says: *"Is this item's month the same as the current month? Is the year the same? If yes, keep it. If no, ignore it."*
2.  **The Monthly Dashboard:** I added a dark-themed section (`monthly-section`) that specifically calculates the math for the current month only.
3.  **The Budget Alert:**
    *   **Logic:** I set a constant `MONTHLY_BUDGET = 5000`. If the `mExpense` (monthly expense) is greater than that number, the code changes the warning box from `display: none` (invisible) to `display: block` (visible).
    *   **Animation:** I added a CSS `@keyframes shake` animation. When the warning appears, it actually shakes on the screen to grab your attention.
4.  **Separation of Data:** The app now distinguishes between **"All-Time"** (how much money do I have in total?) and **"Monthly"** (how am I doing this month?).
