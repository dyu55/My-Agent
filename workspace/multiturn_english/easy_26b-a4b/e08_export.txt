    // 1. NEW: Export to CSV (Spreadsheet format)
    function exportToCSV() {
        if (transactions.length === 0) {
            alert("No data to export!");
            return;
        }

        // Header row
        let csvContent = "Date,Category,Note,Amount,Type\n";

        // Data rows
        transactions.forEach(t => {
            // We wrap the note in double quotes and escape any existing quotes 
            // so that commas inside the note don't break the spreadsheet columns.
            const safeNote = `"${t.note.replace(/"/g, '""')}"`;
            const row = `${t.date},${t.category},${safeNote},${t.amount.toFixed(2)},${t.type}`;
            csvContent += row + "\n";
        });

        // Create the file and download it
        const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement("a");
        link.setAttribute("href", url);
        link.setAttribute("download", `finance_report_${new Date().toISOString().split('T')[0]}.csv`);
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    }

        const file = event.target.files[0];
        if (!file) return;

        reader.onload = function(e) {

                    transactions = importedData;
                    saveAndRefresh();

                }
            } catch (err) { alert("Error reading file."); }
        };
        reader.readAsText(file);
    }

### What's new for your Taxes:

1.  **`exportToCSV()` Function:**
    *   This function constructs a text string where every transaction is on a new line, and every piece of data is separated by a comma.
    *   **The "Comma Protector":** I added `"${t.note.replace(/"/g, '""')}"`. This is a crucial step. If you write a note like: *Paid for groceries, including milk*, the extra comma would usually confuse Excel and move "including milk" into a new column. This code wraps the note in quotes so Excel treats the whole sentence as one single cell.
    *   **The Download:** It generates a real file with a `.csv` extension and triggers a download in your browser.

2.  **New Button:**
    *   I added a **"📊 Export CSV (Excel)"** button at the top. It's highlighted in a slightly different color so you can find it easily when it's time to do your taxes.

3.  **File Naming:**
    *   The file name is automatically generated with today's date, for example: `finance_report_2023-10-27.csv`. This makes it easy to keep organized in your folders.
