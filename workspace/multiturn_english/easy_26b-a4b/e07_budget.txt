This is a big jump in complexity! We are moving from "tracking data" to "managing rules."

To do this, we need to:
1.  **Create a "Settings" area** where you can type in your limits.
2.  **Save these limits** to your browser's "filing cabinet" (`localStorage`).
3.  **Update the math engine** to not just check the *total* budget, but to loop through every category and check its specific limit.

I have added a new **"Budget Settings"** section. It is designed as a "toggle" (a dropdown) so it doesn't clutter your screen while you are just trying to add expenses.

<script>
    // --- CONFIGURATION & DATA ---
    const CATEGORIES = ['Salary', 'Food', 'Transport', 'Shopping', 'Rent', 'Entertainment', 'Other'];
    const TOTAL_BUDGET_LIMIT = 5000;

    let transactions = JSON.parse(localStorage.getItem('my_finance_data')) || [];
    
    // Load category limits or set defaults
    let budgetLimits = JSON.parse(localStorage.getItem('my_category_limits')) || {
        'Food': 1500, 'Transport': 500, 'Shopping': 1000, 'Rent': 2000, 'Entertainment': 500, 'Other': 300
    };

    let myChart = null;

    // --- DOM ELEMENTS ---
    const form = document.getElementById('transaction-form');
    const list = document.getElementById('transaction-list');
    const alertContainer = document.getElementById('alert-container');
    const totalBalanceDisplay = document.getElementById('total-balance');
    const totalIncomeDisplay = document.getElementById('total-income');
    const totalExpenseDisplay = document.getElementById('total-expense');
    const monthIncomeDisplay = document.getElementById('month-income');
    const monthExpenseDisplay = document.getElementById('month-expense');
    const monthLeftoverDisplay = document.getElementById('month-leftover');

    document.getElementById('date').valueAsDate = new Date();

    // --- SETTINGS LOGIC ---
    function toggleSettings() {
        const content = document.getElementById('settings-content');
        const arrow = document.getElementById('settings-arrow');
        content.classList.toggle('show');
        arrow.innerText = content.classList.contains('show') ? '▲' : '▼';
    }

    function renderSettings() {
        const container = document.getElementById('category-inputs');
        container.innerHTML = '';
        CATEGORIES.forEach(cat => {
            // We only show limits for categories that are traditionally "expenses"
            if(cat === 'Salary') return; 
            
            const row = document.createElement('div');
            row.className = 'setting-row';
            row.innerHTML = `
                <label>${cat} Limit ($)</label>
                <input type="number" class="limit-input" data-category="${cat}" value="${budgetLimits[cat] || 0}">
            `;
            container.appendChild(row);
        });
    }

    function saveBudgets() {
        const inputs = document.querySelectorAll('.limit-input');
        inputs.forEach(input => {
            const cat = input.getAttribute('data-category');
            budgetLimits[cat] = parseFloat(input.value) || 0;
        });
        localStorage.setItem('my_category_limits', JSON.stringify(budgetLimits));
        alert("Budgets saved!");
        updateAllStats();
    }

    // --- CORE LOGIC ---
    function updateAllStats() {
        const now = new Date();
        const currentMonth = now.getMonth();
        const currentYear = now.getFullYear();

        // 1. ALL-TIME MATH
        const allTimeIncome = transactions.filter(t => t.type === 'income').reduce((acc, t) => acc + t.amount, 0);
        const allTimeExpense = transactions.filter(t => t.type === 'expense').reduce((acc, t) => acc + t.amount, 0);
        totalIncomeDisplay.innerText = `$${allTimeIncome.toFixed(0)}`;
        totalExpenseDisplay.innerText = `$${allTimeExpense.toFixed(0)}`;
        totalBalanceDisplay.innerText = `$${(allTimeIncome - allTimeExpense).toFixed(0)}`;

        // 2. MONTHLY MATH
        const monthlyTransactions = transactions.filter(t => {
            const d = new Date(t.date);
            return d.getMonth() === currentMonth && d.getFullYear() === currentYear;
        });

        const mIncome = monthlyTransactions.filter(t => t.type === 'income').reduce((acc, t) => acc + t.amount, 0);
        const mExpense = monthlyTransactions.filter(t => t.type === 'expense').reduce((acc, t) => acc + t.amount, 0);
        
        monthIncomeDisplay.innerText = `$${mIncome.toFixed(0)}`;
        monthExpenseDisplay.innerText = `$${mExpense.toFixed(0)}`;
        monthLeftoverDisplay.innerText = `$${(mIncome - mExpense).toFixed(0)}`;

        // 3. BUDGET & CATEGORY ALERT LOGIC
        alertContainer.innerHTML = ''; // Clear old alerts

        // Total Monthly Budget Alert
        if (mExpense > TOTAL_BUDGET_LIMIT) {
            const div = document.createElement('div');
            div.className = 'alert-box alert-total';
            div.innerText = `⚠️ TOTAL MONTHLY BUDGET EXCEEDED!`;
            alertContainer.appendChild(div);
        }

        // Category Specific Alerts
        const categorySpending = {};
        monthlyTransactions.filter(t => t.type === 'expense').forEach(t => {
            categorySpending[t.category] = (categorySpending[t.category] || 0) + t.amount;
        });

        Object.keys(budgetLimits).forEach(cat => {
            const spent = categorySpending[cat] || 0;
            const limit = budgetLimits[cat];
            if (limit > 0 && spent > limit) {
                const div = document.createElement('div');
                div.className = 'alert-box alert-category';
                div.innerText = `🚨 ${cat} is over limit! ($${spent.toFixed(0)} / $${limit})`;
                alertContainer.appendChild(div);
            }
        });

        // 4. CHART UPDATE
        updateChart(monthlyTransactions);
    }

    function updateChart(monthlyTransactions) {
        const chartWrapper = document.getElementById('chart-wrapper');
        const expensesOnly = monthlyTransactions.filter(t => t.type === 'expense');
        
        if (expensesOnly.length === 0) {
            chartWrapper.style.display = 'none';
            return;
        }
        chartWrapper.style.display = 'flex';

        const categoryMap = {};
        expensesOnly.forEach(t => {
            categoryMap[t.category] = (categoryMap[t.category] || 0) + t.amount;
        });

        const labels = Object.keys(categoryMap);
        const data = Object.values(categoryMap);
        const colors = ['#8d6e63', '#8da47e', '#c68b8b', '#d4a373', '#a3b18a', '#ccd5ae', '#e9edc9'];

        if (myChart) myChart.destroy();

        const ctx = document.getElementById('expenseChart').getContext('2d');
        myChart = new Chart(ctx, {
            type: 'pie',
            data: {
                labels: labels,
                datasets: [{ data: data, backgroundColor: colors, borderWidth: 2, borderColor: '#ffffff' }]
            },
            options: {
                responsive: true,
                plugins: { legend: { position: 'bottom', labels: { font: { family: 'Quicksand', size: 12 } } } }
            }
        });
    }

    // --- STANDARD FUNCTIONS (Add, Delete, Save, Backup) ---
    function renderTable() {
        list.innerHTML = '';
        transactions.sort((a, b) => new Date(b.date) - new Date(a.date));
        transactions.forEach(t => {
            const row = document.createElement('tr');
            const isIncome = t.type === 'income';
            row.innerHTML = `
                <td><small>${t.date.split('-').slice(1).join('/')}</small></td>
                <td><strong>${t.category}</strong><br><small style="color:#999">${t.note || ''}</small></td>
                <td class="${isIncome ? 'amt-income' : 'amt-expense'}">${isIncome ? '+' : '-'}$${t.amount.toFixed(2)}</td>
                <td><button class="delete-btn" onclick="deleteTransaction(${t.id})">✕</button></td>
            `;
            list.appendChild(row);
        });
    }

    function addTransaction(e) {
        e.preventDefault();
        transactions.push({
            id: Date.now(),
            type: document.getElementById('type').value,
            amount: parseFloat(document.getElementById('amount').value),
            category: document.getElementById('category').value,
            date: document.getElementById('date').value,
            note: document.getElementById('note').value
        });
        saveAndRefresh();
        form.reset();
        document.getElementById('date').valueAsDate = new Date();
    }

    function deleteTransaction(id) {
        transactions = transactions.filter(t => t.id !== id);
        saveAndRefresh();
    }

    function saveAndRefresh() {
        localStorage.setItem('my_finance_data', JSON.stringify(transactions));
        updateAllStats();
        renderTable();
    }

    function exportData() {
        const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(transactions));
        const downloadAnchorNode = document.createElement('a');
        downloadAnchorNode.setAttribute("href", dataStr);
        downloadAnchorNode.setAttribute("download", "finance_backup.json");
        document.body.appendChild(downloadAnchorNode);
        downloadAnchorNode.click();
        downloadAnchorNode.remove();
    }

    function importData() { document.getElementById('fileInput').click(); }

    function handleFile(event) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function(e) {
            try {
                const importedData = JSON.parse(e.target.result);
                if (Array.isArray(importedData)) {
                    transactions = importedData;
                    saveAndRefresh();
                    alert("Restored successfully!");
                }
            } catch (err) { alert("Error reading file."); }
        };
        reader.readAsText(file);
    }

    // Initialize
    form.addEventListener('submit', addTransaction);
    renderSettings();
    updateAllStats();
    renderTable();
</script>

</body>
</html>

### What is different now?

1.  **The Settings Engine:**
    *   I added a `CATEGORIES` array at the top of the script. This is the "Master List."
    *   I added `budgetLimits` which is a separate object in your `localStorage`. This keeps your rules separate from your money records.
    *   `renderSettings()`: This function runs when you open the app. It looks at the Master List and automatically builds a list of input boxes for you. This way, if you ever want to add a category like "Pet Supplies," you just add it to that one list in the code, and the UI updates itself.

2.  **The Alert Engine:**
    *   Inside `updateAllStats()`, the code now does a "Double Check."
    *   First, it checks the **Total Budget**.
    *   Second, it creates a temporary map of how much you spent on every category this month.
    *   Then, it compares that map against your `budgetLimits`. If "Food" spent is $1600 and the limit is $1500, it creates a new "Alert Box" in the `alert-container`.

3.  **The UI Improvements:**
    *   **The Settings Accordion:** To prevent the page from looking messy, the budget settings are hidden behind a "Budget Settings" button. Click it to expand, change your numbers, and click "Save Limits."
    *   **Specific Alerts:** Instead of just one big red bar, you now get specific "pills" that tell you exactly which category is causing the problem (e.g., `🚨 Food is over limit! ($1600 / $1500)`).

### How to test it:

2.  Change the "Food" limit to **$10**.
3.  Click **Save Limits**.
4.  Add an expense: **Food | $15**.
5.  You should immediately see a red alert telling you that Food is over limit!
